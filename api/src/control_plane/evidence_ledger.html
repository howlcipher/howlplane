src.control_plane.evidence_ledger

evidence_ledger.py

Durable, history-preserving, append-only evidence ledger with automatic redaction of secrets, credentials, and sensitive data.

The active file rotates into gzip-compressed archive segments once it reaches a size threshold. Rotation never drops history: every read returns the archived segments, oldest first, followed by the active file.

 85def schema_violations(record: Any) -> List[str]:
 86    """Every way ``record`` violates the evidence schema, as ``path: message`` strings."""
 87    errors = sorted(_entry_validator().iter_errors(record), key=lambda e: list(map(str, e.absolute_path)))
 88    out = []
 89    for e in errors:
 90        where = "/".join(map(str, e.absolute_path))
 91        # An enum message would list every known action; the offending value is enough.
 92        message = f"{e.instance!r} is not a recognised value" if e.validator == "enum" else e.message
 93        out.append(f"{where}: {message}" if where else message)
 94    return out

106def redact_sensitive_data(text: str) -> str:
107    """Scans and scrubs secrets, credentials, and PII from text."""
108    if not isinstance(text, str):
109        return text
110    sanitized = text
111    for pattern, replacement in REDACTION_PATTERNS:
112        sanitized = pattern.sub(replacement, sanitized)
113    return sanitized

116def sanitize_value(val: Any) -> Any:
117    """Recursively redacts sensitive strings inside dicts, lists, and strings."""
118    if isinstance(val, str):
119        return redact_sensitive_data(val)
120    if isinstance(val, dict):
121        return {k: sanitize_value(v) for k, v in val.items()}
122    if isinstance(val, list):
123        return [sanitize_value(item) for item in val]
124    return val

213class EvidenceLedger:
214    """Manages the append-only evidence ledger log files.
215
216    ``ledger_file`` is the active file. An append that leaves it at or above
217    ``rotate_bytes`` compresses it into ``<name>.<UTC timestamp>.gz`` beside it,
218    and the next append starts a new active file. ``rotate_bytes=0`` disables
219    rotation.
220
221    Appends and reads hold a shared ``flock`` on the ledger's directory; rotation
222    holds it exclusively. No writer can therefore still be writing to a file
223    that is being archived, and no reader can see the segment list change under
224    it. Locking the directory rather than a lock file beside the ledger leaves
225    nothing behind that the ledger did not already own.
226    """
227
228    def __init__(self, ledger_file: Optional[str] = None, rotate_bytes: int = DEFAULT_ROTATE_BYTES):
229        if ledger_file is None:
230            ledger_file = str(default_ledger_path())
231        if rotate_bytes < 0:
232            raise ValueError(f"rotate_bytes must be 0 (disabled) or positive, got {rotate_bytes}")
233        self.ledger_file = Path(ledger_file)
234        self.ledger_file.parent.mkdir(parents=True, exist_ok=True)
235        self.rotate_bytes = rotate_bytes
236        # <name>.<stamp> is a staged segment, <name>.<stamp>.gz an archive, and
237        # <name>.<stamp>.gz.partial an archive still being written.
238        self._segment_name = re.compile(
239            rf"^{re.escape(self.ledger_file.name)}\.(\d{{8}}T\d{{12}}Z)(\.gz)?(\.partial)?$"
240        )
241
242    @contextmanager
243    def _locked(self, exclusive: bool) -> Iterator[None]:
244        if fcntl is None:
245            yield
246            return
247        fd = os.open(self.ledger_file.parent, os.O_RDONLY)
248        try:
249            fcntl.flock(fd, fcntl.LOCK_EX if exclusive else fcntl.LOCK_SH)
250            yield
251        finally:
252            os.close(fd)  # closing the descriptor releases the lock
253
254    def append_entry(self, entry: EvidenceEntry) -> None:
255        """Validates against the schema, then appends the sanitized entry to the JSON Lines log.
256
257        Nothing is written when validation or serialization fails. A rotation
258        triggered by this append that fails is logged, not raised: the entry is
259        already durable, and an error here would invite a duplicate retry.
260        """
261        record = entry.to_dict()
262        line = json.dumps(record) + "\n"
263        violations = schema_violations(json.loads(line))
264        if violations:
265            _LOG.error("evidence entry %s (%s) violates the schema: %s", entry.entry_id, entry.action, violations)
266            raise EvidenceSchemaError(
267                f"evidence entry {entry.entry_id} (action={entry.action!r}) violates "
268                f"{_SCHEMA_PATH.name}: " + "; ".join(violations)
269            )
270        # One O_APPEND write of the whole line keeps concurrent appenders from
271        # interleaving records; fsync makes the entry durable before returning.
272        data = line.encode("utf-8")
273        with self._locked(exclusive=False):
274            fd = os.open(self.ledger_file, os.O_WRONLY | os.O_APPEND | os.O_CREAT, 0o666)
275            try:
276                written = os.write(fd, data)
277                while written < len(data):
278                    written += os.write(fd, data[written:])
279                os.fsync(fd)
280                size = os.fstat(fd).st_size
281            finally:
282                os.close(fd)
283        if self.rotate_bytes and size >= self.rotate_bytes:
284            try:
285                self._rotate(min_bytes=self.rotate_bytes)
286            except OSError as exc:
287                _LOG.error("could not rotate %s: %s; the next rotation will retry", self.ledger_file, exc)
288
289    def rotate(self) -> Optional[Path]:
290        """Archives the active file now, whatever its size.
291
292        Returns the new archive, or None when there is nothing to archive or the
293        platform has no ``flock``.
294        """
295        return self._rotate(min_bytes=1)
296
297    def _rotate(self, min_bytes: int) -> Optional[Path]:
298        if fcntl is None:
299            return None
300        with self._locked(exclusive=True):
301            self._finish_interrupted_rotations()
302            try:
303                size = self.ledger_file.stat().st_size
304            except FileNotFoundError:
305                return None
306            # Another appender may have rotated while this one waited for the lock.
307            if size < min_bytes:
308                return None
309            staged = self._new_segment_path()
310            os.replace(self.ledger_file, staged)
311            return self._compress(staged)
312
313    def _new_segment_path(self) -> Path:
314        stamp = datetime.now(timezone.utc)
315        archived = self._archives()
316        if archived:
317            # Never sort before an existing archive, even if the clock went back.
318            latest = datetime.strptime(archived[-1][0], _ARCHIVE_STAMP).replace(tzinfo=timezone.utc)
319            stamp = max(stamp, latest + timedelta(microseconds=1))
320        return self.ledger_file.with_name(f"{self.ledger_file.name}.{stamp.strftime(_ARCHIVE_STAMP)}")
321
322    def _compress(self, staged: Path) -> Path:
323        """Compresses a staged segment, then removes it. Caller holds the exclusive lock.
324
325        The archive is complete and durable before the staged copy goes, so a
326        crash at any point leaves each entry in a readable segment exactly once:
327        readers prefer the archive when both exist.
328        """
329        archive = staged.with_name(staged.name + ".gz")
330        partial = staged.with_name(staged.name + ".gz.partial")
331        with open(staged, "rb") as src, open(partial, "wb") as raw:
332            with gzip.GzipFile(filename=staged.name, mode="wb", fileobj=raw, mtime=0) as dst:
333                shutil.copyfileobj(src, dst, 1024 * 1024)
334            raw.flush()
335            os.fsync(raw.fileno())
336        os.replace(partial, archive)
337        _fsync_directory(archive.parent)
338        os.unlink(staged)
339        return archive
340
341    def _finish_interrupted_rotations(self) -> None:
342        """Completes any rotation a crash cut short. Caller holds the exclusive lock."""
343        staged, archived = [], set()
344        for path in self.ledger_file.parent.iterdir():
345            match = self._segment_name.match(path.name)
346            if not match:
347                continue
348            if match.group(3):
349                path.unlink()
350            elif match.group(2):
351                archived.add(match.group(1))
352            else:
353                staged.append((match.group(1), path))
354        for stamp, path in sorted(staged):
355            if stamp in archived:
356                path.unlink()
357            else:
358                self._compress(path)
359
360    def _archives(self) -> List[Tuple[str, Path]]:
361        """Archived segments, oldest first. A finished archive wins over its staged original."""
362        found: Dict[str, Path] = {}
363        if not self.ledger_file.parent.is_dir():
364            return []
365        for path in self.ledger_file.parent.iterdir():
366            match = self._segment_name.match(path.name)
367            if not match or match.group(3):
368                continue
369            if match.group(2) or match.group(1) not in found:
370                found[match.group(1)] = path
371        return sorted(found.items())
372
373    def segment_files(self) -> List[Path]:
374        """Every file holding ledger history, oldest first: archives, then the active file."""
375        with self._locked(exclusive=False):
376            return self._segment_files()
377
378    def _segment_files(self) -> List[Path]:
379        files = [path for _, path in self._archives()]
380        if self.ledger_file.exists():
381            files.append(self.ledger_file)
382        return files
383
384    def _lines(self) -> Iterator[Tuple[str, int, str]]:
385        """Yields (segment name, line number, text) across all history.
386
387        Holds the shared lock until the generator is exhausted or closed, so a
388        rotation cannot move a segment between listing and reading it.
389        """
390        with self._locked(exclusive=False):
391            for path in self._segment_files():
392                opener = gzip.open if path.name.endswith(".gz") else open
393                with opener(path, "rt", encoding="utf-8") as f:
394                    for number, raw in enumerate(f, start=1):
395                        yield path.name, number, raw

397    def read_entries(self, strict: bool = False) -> Tuple[List[EvidenceEntry], List[LedgerDiagnostic]]:
398        """Reads the whole ledger history and reports every record that is not clean.
399
400        Malformed lines cannot be returned and are listed as diagnostics instead of
401        disappearing. Schema-invalid but parseable (for example historical) records are
402        still returned and flagged. With ``strict`` any diagnostic raises
403        :class:`EvidenceCorruptionError`.
404        """
405        entries: List[EvidenceEntry] = []
406        diagnostics: List[LedgerDiagnostic] = []
407        for source, number, raw in self._lines():
408            text = raw.strip()
409            if not text:
410                continue
411            try:
412                record = json.loads(text)
413                entry = EvidenceEntry.from_dict(record)
414            except Exception as exc:
415                diagnostics.append(LedgerDiagnostic(number, "malformed", f"{type(exc).__name__}: {exc}", source))
416                continue
417            entries.append(entry)
418            violations = schema_violations(record)
419            if violations:
420                diagnostics.append(LedgerDiagnostic(number, "schema_invalid", "; ".join(violations), source))
421        if strict and diagnostics:
422            first = diagnostics[0]
423            raise EvidenceCorruptionError(
424                f"{self.ledger_file}: {len(diagnostics)} unclean record(s); first is line "
425                f"{first.line_number} of {first.source} ({first.kind}): {first.reason}"
426            )
427        return entries, diagnostics
428
429    def list_all_entries(self) -> List[EvidenceEntry]:
430        """Reads all parseable entries across the whole history. Skipped malformed lines are logged, never silent."""
431        entries: List[EvidenceEntry] = []
432        skipped = 0
433        for _, _, raw in self._lines():
434            line = raw.strip()
435            if line:
436                try:
437                    entries.append(EvidenceEntry.from_dict(json.loads(line)))
438                except Exception:
439                    skipped += 1
440        if skipped:
441            _LOG.warning("%s: skipped %d malformed evidence line(s); run `howlplane doctor`", self.ledger_file, skipped)
442        return entries
443
444    def get_task_entries(self, task_id: str) -> List[EvidenceEntry]:
445        """Retrieves all evidence entries associated with a specific task_id."""
446        return [e for e in self.list_all_entries() if e.task_id == task_id]

214class EvidenceLedger:
215    """Manages the append-only evidence ledger log files.
216
217    ``ledger_file`` is the active file. An append that leaves it at or above
218    ``rotate_bytes`` compresses it into ``<name>.<UTC timestamp>.gz`` beside it,
219    and the next append starts a new active file. ``rotate_bytes=0`` disables
220    rotation.
221
222    Appends and reads hold a shared ``flock`` on the ledger's directory; rotation
223    holds it exclusively. No writer can therefore still be writing to a file
224    that is being archived, and no reader can see the segment list change under
225    it. Locking the directory rather than a lock file beside the ledger leaves
226    nothing behind that the ledger did not already own.
227    """
228
229    def __init__(self, ledger_file: Optional[str] = None, rotate_bytes: int = DEFAULT_ROTATE_BYTES):
230        if ledger_file is None:
231            ledger_file = str(default_ledger_path())
232        if rotate_bytes < 0:
233            raise ValueError(f"rotate_bytes must be 0 (disabled) or positive, got {rotate_bytes}")
234        self.ledger_file = Path(ledger_file)
235        self.ledger_file.parent.mkdir(parents=True, exist_ok=True)
236        self.rotate_bytes = rotate_bytes
237        # <name>.<stamp> is a staged segment, <name>.<stamp>.gz an archive, and
238        # <name>.<stamp>.gz.partial an archive still being written.
239        self._segment_name = re.compile(
240            rf"^{re.escape(self.ledger_file.name)}\.(\d{{8}}T\d{{12}}Z)(\.gz)?(\.partial)?$"
241        )
242
243    @contextmanager
244    def _locked(self, exclusive: bool) -> Iterator[None]:
245        if fcntl is None:
246            yield
247            return
248        fd = os.open(self.ledger_file.parent, os.O_RDONLY)
249        try:
250            fcntl.flock(fd, fcntl.LOCK_EX if exclusive else fcntl.LOCK_SH)
251            yield
252        finally:
253            os.close(fd)  # closing the descriptor releases the lock

255    def append_entry(self, entry: EvidenceEntry) -> None:
256        """Validates against the schema, then appends the sanitized entry to the JSON Lines log.
257
258        Nothing is written when validation or serialization fails. A rotation
259        triggered by this append that fails is logged, not raised: the entry is
260        already durable, and an error here would invite a duplicate retry.
261        """
262        record = entry.to_dict()
263        line = json.dumps(record) + "\n"
264        violations = schema_violations(json.loads(line))
265        if violations:
266            _LOG.error("evidence entry %s (%s) violates the schema: %s", entry.entry_id, entry.action, violations)
267            raise EvidenceSchemaError(
268                f"evidence entry {entry.entry_id} (action={entry.action!r}) violates "
269                f"{_SCHEMA_PATH.name}: " + "; ".join(violations)
270            )
271        # One O_APPEND write of the whole line keeps concurrent appenders from
272        # interleaving records; fsync makes the entry durable before returning.
273        data = line.encode("utf-8")
274        with self._locked(exclusive=False):
275            fd = os.open(self.ledger_file, os.O_WRONLY | os.O_APPEND | os.O_CREAT, 0o666)
276            try:
277                written = os.write(fd, data)
278                while written < len(data):
279                    written += os.write(fd, data[written:])
280                os.fsync(fd)
281                size = os.fstat(fd).st_size
282            finally:
283                os.close(fd)
284        if self.rotate_bytes and size >= self.rotate_bytes:
285            try:
286                self._rotate(min_bytes=self.rotate_bytes)
287            except OSError as exc:
288                _LOG.error("could not rotate %s: %s; the next rotation will retry", self.ledger_file, exc)
289
290    def rotate(self) -> Optional[Path]:
291        """Archives the active file now, whatever its size.
292
293        Returns the new archive, or None when there is nothing to archive or the
294        platform has no ``flock``.
295        """
296        return self._rotate(min_bytes=1)
297
298    def _rotate(self, min_bytes: int) -> Optional[Path]:
299        if fcntl is None:
300            return None
301        with self._locked(exclusive=True):
302            self._finish_interrupted_rotations()
303            try:
304                size = self.ledger_file.stat().st_size
305            except FileNotFoundError:
306                return None
307            # Another appender may have rotated while this one waited for the lock.
308            if size < min_bytes:
309                return None
310            staged = self._new_segment_path()
311            os.replace(self.ledger_file, staged)
312            return self._compress(staged)
313
314    def _new_segment_path(self) -> Path:
315        stamp = datetime.now(timezone.utc)
316        archived = self._archives()
317        if archived:
318            # Never sort before an existing archive, even if the clock went back.
319            latest = datetime.strptime(archived[-1][0], _ARCHIVE_STAMP).replace(tzinfo=timezone.utc)
320            stamp = max(stamp, latest + timedelta(microseconds=1))
321        return self.ledger_file.with_name(f"{self.ledger_file.name}.{stamp.strftime(_ARCHIVE_STAMP)}")
322
323    def _compress(self, staged: Path) -> Path:
324        """Compresses a staged segment, then removes it. Caller holds the exclusive lock.
325
326        The archive is complete and durable before the staged copy goes, so a
327        crash at any point leaves each entry in a readable segment exactly once:
328        readers prefer the archive when both exist.
329        """
330        archive = staged.with_name(staged.name + ".gz")
331        partial = staged.with_name(staged.name + ".gz.partial")
332        with open(staged, "rb") as src, open(partial, "wb") as raw:
333            with gzip.GzipFile(filename=staged.name, mode="wb", fileobj=raw, mtime=0) as dst:
334                shutil.copyfileobj(src, dst, 1024 * 1024)
335            raw.flush()
336            os.fsync(raw.fileno())
337        os.replace(partial, archive)
338        _fsync_directory(archive.parent)
339        os.unlink(staged)
340        return archive
341
342    def _finish_interrupted_rotations(self) -> None:
343        """Completes any rotation a crash cut short. Caller holds the exclusive lock."""
344        staged, archived = [], set()
345        for path in self.ledger_file.parent.iterdir():
346            match = self._segment_name.match(path.name)
347            if not match:
348                continue
349            if match.group(3):
350                path.unlink()
351            elif match.group(2):
352                archived.add(match.group(1))
353            else:
354                staged.append((match.group(1), path))
355        for stamp, path in sorted(staged):
356            if stamp in archived:
357                path.unlink()
358            else:
359                self._compress(path)
360
361    def _archives(self) -> List[Tuple[str, Path]]:
362        """Archived segments, oldest first. A finished archive wins over its staged original."""
363        found: Dict[str, Path] = {}
364        if not self.ledger_file.parent.is_dir():
365            return []
366        for path in self.ledger_file.parent.iterdir():
367            match = self._segment_name.match(path.name)
368            if not match or match.group(3):
369                continue
370            if match.group(2) or match.group(1) not in found:
371                found[match.group(1)] = path
372        return sorted(found.items())
373
374    def segment_files(self) -> List[Path]:
375        """Every file holding ledger history, oldest first: archives, then the active file."""
376        with self._locked(exclusive=False):
377            return self._segment_files()
378
379    def _segment_files(self) -> List[Path]:
380        files = [path for _, path in self._archives()]
381        if self.ledger_file.exists():
382            files.append(self.ledger_file)
383        return files
384
385    def _lines(self) -> Iterator[Tuple[str, int, str]]:
386        """Yields (segment name, line number, text) across all history.
387
388        Holds the shared lock until the generator is exhausted or closed, so a
389        rotation cannot move a segment between listing and reading it.
390        """
391        with self._locked(exclusive=False):
392            for path in self._segment_files():
393                opener = gzip.open if path.name.endswith(".gz") else open
394                with opener(path, "rt", encoding="utf-8") as f:
395                    for number, raw in enumerate(f, start=1):
396                        yield path.name, number, raw
