155    def __init__(self, base_dir: Union[str, Path]):
156        super().__init__(

162    def save(self, trajectory: ExecutionTrajectory) -> Path:
163        """Atomically persists a trajectory; idempotent on repeated calls."""
164        target = self._path(trajectory.trajectory_id)
165        if target.is_file():
166            try:
167                existing = self.load(trajectory.trajectory_id)
168            except Exception:
169                existing = None
170            if existing is not None:
171                if existing.content_digest == trajectory.content_digest:
172                    return target
173                existing_payload = existing.to_dict()
174                incoming_payload = trajectory.to_dict()
175                for payload in (existing_payload, incoming_payload):
176                    payload.pop("created_at", None)
177                    payload.pop("completed_at", None)
178                    payload.pop("content_digest", None)
179                if existing_payload == incoming_payload:
180                    return target
181        return super().save(trajectory.trajectory_id, trajectory.to_dict())

243def evaluate_experiment(
244    experiment: ReasoningExperiment,
245    baseline_summaries: Optional[List[Dict[str, Any]]] = None,
246    candidate_summaries: Optional[List[Dict[str, Any]]] = None,
247) -> Tuple[str, Dict[str, Any]]:
248    """
249    Deterministically evaluates a reasoning experiment.
250
251    Returns one of the VALID_EXPERIMENT_OUTCOMES plus a detail dict.
252    """
253    baseline = baseline_summaries if baseline_summaries is not None else experiment.baseline_results
254    candidate = candidate_summaries if candidate_summaries is not None else experiment.candidate_results
255
256    if not baseline or not candidate:
257        return "NOT_YET_MEASURABLE", {"reason": "Missing baseline or candidate results."}
258
259    if not experiment.verify_prediction_digest():
260        raise EvaluationError("Prediction digest mismatch: experiment definition changed after execution started.")
261
262    details: Dict[str, Any] = {
263        "baseline_n": len(baseline),
264        "candidate_n": len(candidate),
265        "falsification_failures": [],
266        "metric_comparisons": {},
267        "primary_metric": None,
268    }
269
270    # 1. Evaluate explicit falsification criteria first.
271    for criterion in experiment.falsification_criteria:
272        try:
273            passed, reason = evaluate_falsification_criterion(criterion, baseline, candidate)
274        except CriterionParseError as exc:
275            details["falsification_failures"].append(str(exc))
276            continue
277        if not passed:
278            details["falsification_failures"].append(reason or criterion)
279
280    if details["falsification_failures"]:
281        return "FALSIFIED", details
282
283    # 2. Compare configured metrics.
284    metrics = experiment.metrics or ["verification_pass_rate", "mean_repair_cycles"]
285    primary_metric = metrics[0]
286    details["primary_metric"] = primary_metric
287
288    comparisons: Dict[str, str] = {}
289    for metric in metrics:
290        b = _get_aggregate(baseline, metric)
291        c = _get_aggregate(candidate, metric)
292        if b is None or c is None:
293            comparisons[metric] = "incomparable"
294            continue
295        if metric in _HIGHER_IS_BETTER:
296            if c > b:
297                comparisons[metric] = "better"
298            elif c == b:
299                comparisons[metric] = "equal"
300            else:
301                comparisons[metric] = "worse"
302        elif metric in _LOWER_IS_BETTER:
303            if c < b:
304                comparisons[metric] = "better"
305            elif c == b:
306                comparisons[metric] = "equal"
307            else:
308                comparisons[metric] = "worse"
309        else:
310            comparisons[metric] = "equal" if c == b else "incomparable"
311    details["metric_comparisons"] = comparisons
312
313    # 3. Decide outcome.
314    primary = comparisons.get(primary_metric, "incomparable")
315    if primary == "worse":
316        return "FALSIFIED", details
317
318    non_primary = [comparisons[m] for m in metrics if m != primary_metric]
319    any_worse = any(v == "worse" for v in non_primary)
320    all_better_or_equal = all(v in ("better", "equal") for v in non_primary)
321
322    candidate_n = len(candidate)
323    baseline_n = len(baseline)
324
325    if primary == "better" and all_better_or_equal and candidate_n >= MIN_SAMPLES_FOR_SUPPORTED and baseline_n >= MIN_SAMPLES_FOR_SUPPORTED:
326        return "SUPPORTED", details
327
328    if primary == "better" and not any_worse and candidate_n >= MIN_SAMPLES_FOR_WEAKLY_SUPPORTED:
329        return "WEAKLY_SUPPORTED", details
330
331    if primary == "better" and any_worse:
332        return "INCONCLUSIVE", details
333
334    if primary == "equal":
335        return "INCONCLUSIVE", details
336
337    # Directional signal on primary but insufficient sample.
338    if primary == "better" and candidate_n < MIN_SAMPLES_FOR_SUPPORTED:
339        return "NOT_YET_MEASURABLE", details
340
341    return "INCONCLUSIVE", details

339            new_refs = sorted(set(obs.evidence_refs) - set(existing.evidence_refs))
340            new_fingerprints = sorted(
341                set(obs.evidence_fingerprints) - set(existing.evidence_fingerprints)
342            )
343            if reopen_new_evidence and new_refs and new_fingerprints:
344                existing.evidence_refs = sorted(
345                    set(existing.evidence_refs) | set(obs.evidence_refs)
346                )
347                existing.evidence_fingerprints = sorted(
348                    set(existing.evidence_fingerprints) | set(obs.evidence_fingerprints)
349                )
350                existing.occurrence_count = max(existing.occurrence_count, obs.occurrence_count)
351                existing.reopen(
352                    new_refs,
353                    new_fingerprints,
354                    "new observable evidence fingerprint after prior disposition",
355                )
356                store.save(existing)
357                result.append(existing)
358        # If existing is OPEN or EXPERIMENTING, do not duplicate; the observation is
359        # already being tracked through the normal backlog flow.
360    return result
