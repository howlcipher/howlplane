role_binding.py

Domain-neutral role execution, capability dispatch, and provider binding framework. Allows any domain (software engineering, writing, research) to bind specialized roles (e.g. humanizer, meaning_reviewer, writer, editor) to real underlying model executors/providers while maintaining reviewer independence, policy compliance, timeout budgets, and durable evidence.

131def extract_structured_output(text: str) -> Optional[Dict[str, Any]]:
132    """Extracts JSON or YAML object from model output text or code blocks."""
133    if not text or not text.strip():
134        return None
135
136    clean = text.strip()
137
138    # 1. Try fenced json or yaml code blocks
139    matches = re.findall(r"```(?:yaml|json)?\s*\n([\s\S]*?)\n```", clean)
140    for block in matches:
141        candidate = block.strip()
142        try:
143            parsed = yaml.safe_load(candidate)
144            if isinstance(parsed, dict):
145                return parsed
146        except Exception:
147            try:
148                parsed_json = json.loads(candidate)
149                if isinstance(parsed_json, dict):
150                    return parsed_json
151            except Exception:
152                pass
153
154    # 2. Extract from Codex/CLI transcript if present
155    codex_match = re.search(
156        r"(?:^|\n)codex\s*\n([\s\S]*?)(?:\ntokens used|\Z)",
157        clean,
158        re.IGNORECASE,
159    )
160    if codex_match:
161        sub_text = codex_match.group(1).strip()
162        sub_matches = re.findall(
163            r"```(?:yaml|json)?\s*\n([\s\S]*?)\n```", sub_text
164        )
165        for block in sub_matches:
166            try:
167                parsed = yaml.safe_load(block.strip())
168                if isinstance(parsed, dict):
169                    return parsed
170            except Exception:
171                pass
172        try:
173            parsed = yaml.safe_load(sub_text)
174            if isinstance(parsed, dict):
175                return parsed
176        except Exception:
177            pass
178
179    # 3. Try direct parse of entire text as JSON or YAML
180    try:
181        parsed_json = json.loads(clean)
182        if isinstance(parsed_json, dict):
183            return parsed_json
184    except Exception:
185        pass
186
187    try:
188        parsed_yaml = yaml.safe_load(clean)
189        if isinstance(parsed_yaml, dict):
190            return parsed_yaml
191    except Exception:
192        pass
193
194    # 4. Try JSON substring if braced
195    first_b = clean.find("{")
196    last_b = clean.rfind("}")
197    if first_b != -1 and last_b != -1 and last_b > first_b:
198        try:
199            parsed_json = json.loads(clean[first_b:last_b + 1])
200            if isinstance(parsed_json, dict):
201                return parsed_json
202        except Exception:
203            pass
204
205    return None

208class RoleBindingRegistry:
209    """Central declarative registry mapping domain roles to providers."""
210
211    def __init__(self) -> None:
212        self._bindings: Dict[Tuple[str, str], RoleBinding] = {}
213        self._descriptors: Dict[Tuple[str, str], RoleDescriptor] = {}
214
215    def register_descriptor(self, descriptor: RoleDescriptor) -> None:
216        key = (descriptor.domain.lower(), descriptor.role.lower())
217        self._descriptors[key] = descriptor
218
219    def register_binding(self, binding: RoleBinding) -> None:
220        key = (binding.domain.lower(), binding.role.lower())
221        self._bindings[key] = binding
222
223    def get_binding(self, domain: str, role: str) -> Optional[RoleBinding]:
224        key = (domain.lower(), role.lower())
225        return self._bindings.get(key)
226
227    def get_descriptor(
228        self, domain: str, role: str
229    ) -> Optional[RoleDescriptor]:
230        key = (domain.lower(), role.lower())
231        return self._descriptors.get(key)
232
233    def list_bindings(
234        self, domain: Optional[str] = None
235    ) -> List[RoleBinding]:
236        if domain is None:
237            return list(self._bindings.values())
238        return [
239            b for b in self._bindings.values()
240            if b.domain.lower() == domain.lower()
241        ]
242
243    def clear(self) -> None:
244        self._bindings.clear()
245        self._descriptors.clear()
246
247    def load_from_config(self, config_data: Dict[str, Any]) -> None:
248        """Parses role bindings from config structure."""
249        roles_section = config_data.get("roles", {})
250        if not isinstance(roles_section, dict):
251            return
252
253        for domain_key, domain_val in roles_section.items():
254            if not isinstance(domain_val, dict):
255                continue
256            for role_key, role_spec in domain_val.items():
257                if isinstance(role_spec, str):
258                    # Shorthand: humanizer = "claude_code"
259                    self.register_binding(
260                        RoleBinding(
261                            domain=domain_key,
262                            role=role_key,
263                            provider=role_spec,
264                        )
265                    )
266                elif isinstance(role_spec, dict):
267                    # Full spec: [roles.writing.humanizer] provider = "..."
268                    provider = (
269                        role_spec.get("provider")
270                        or role_spec.get("agent_id")
271                    )
272                    if provider:
273                        self.register_binding(
274                            RoleBinding(
275                                domain=domain_key,
276                                role=role_key,
277                                provider=provider,
278                                model=role_spec.get("model"),
279                                interface=role_spec.get("interface"),
280                                capability=role_spec.get("capability"),
281                                timeout_seconds=role_spec.get(
282                                    "timeout_seconds", 300
283                                ),
284                                metadata=role_spec.get("metadata", {}),
285                            )
286                        )
287
288    def load_from_env(self) -> None:
289        """Loads bindings from HOWLPLANE_ROLE_<DOMAIN>_<ROLE> env vars."""
290        prefix = "HOWLPLANE_ROLE_"
291        for key, value in os.environ.items():
292            if key.startswith(prefix) and value.strip():
293                rest = key[len(prefix):].lower()
294                parts = rest.split("_", 1)
295                if len(parts) == 2:
296                    domain, role = parts
297                    self.register_binding(
298                        RoleBinding(
299                            domain=domain,
300                            role=role,
301                            provider=value.strip(),
302                        )
303                    )

317class RoleDispatcher:
318    """Resolves and executes domain roles through HowlPlane backends."""
319
320    def __init__(
321        self,
322        binding_registry: Optional[RoleBindingRegistry] = None,
323        agent_registry: Optional[AgentRegistry] = None,
324    ) -> None:
325        self.binding_registry = binding_registry or get_default_role_registry()
326        self.agent_registry = agent_registry or AgentRegistry()
327
328    def resolve_provider(
329        self,
330        domain: str,
331        role: str,
332        avoid_provider: Optional[str] = None,
333        preferred_provider: Optional[str] = None,
334    ) -> Tuple[Optional[str], Optional[str], str]:
335        """Resolves target provider with honest independence calculation."""
336        binding = self.binding_registry.get_binding(domain, role)
337        candidate_provider = (
338            preferred_provider or (binding.provider if binding else None)
339        )
340        model_id = binding.model if binding else None
341
342        if not candidate_provider:
343            return None, None, IndependenceStatus.UNAVAILABLE.value
344
345        norm_candidate = AgentBackendRegistry.normalize_agent_id(
346            candidate_provider
347        )
348        norm_avoid = (
349            AgentBackendRegistry.normalize_agent_id(avoid_provider)
350            if avoid_provider
351            else None
352        )
353
354        if not norm_avoid:
355            return (
356                candidate_provider,
357                model_id,
358                IndependenceStatus.NOT_REVIEWED.value,
359            )
360
361        if norm_candidate != norm_avoid:
362            return (
363                candidate_provider,
364                model_id,
365                IndependenceStatus.INDEPENDENT.value,
366            )
367
368        # Candidate matches avoid_provider. Find available alternate.
369        available_agents = self.agent_registry.list_agents(only_available=True)
370        alternate = next(
371            (
372                a.agent_id
373                for a in available_agents
374                if AgentBackendRegistry.normalize_agent_id(a.agent_id)
375                != norm_avoid
376                and AgentBackendRegistry.get_backend(a.agent_id).is_available()
377            ),
378            None,
379        )
380
381        if alternate:
382            return alternate, None, IndependenceStatus.INDEPENDENT.value
383
384        return (
385            candidate_provider,
386            model_id,
387            IndependenceStatus.SAME_PROVIDER.value,
388        )
389
390    def execute(
391        self,
392        request: RoleExecutionRequest,
393        custom_backend: Optional[AgentBackend] = None,
394        backend_resolver: Optional[Callable[[str], AgentBackend]] = None,
395    ) -> RoleExecutionResult:
396        """Executes a domain role request against the resolved backend."""
397        start_time = time.time()
398
399        if custom_backend is not None:
400            provider_id = getattr(
401                custom_backend, "agent_id", "custom_backend"
402            )
403            independence_status = (
404                IndependenceStatus.INDEPENDENT.value
405                if request.avoid_provider != provider_id
406                else IndependenceStatus.SAME_PROVIDER.value
407            )
408            model_id = None
409        else:
410            provider_id, model_id, independence_status = self.resolve_provider(
411                domain=request.domain,
412                role=request.role,
413                avoid_provider=request.avoid_provider,
414                preferred_provider=request.preferred_provider,
415            )
416
417        if not provider_id:
418            return RoleExecutionResult(
419                domain=request.domain,
420                role=request.role,
421                provider="none",
422                model=None,
423                success=False,
424                duration_seconds=0.0,
425                independence_status=IndependenceStatus.UNAVAILABLE.value,
426                error_message=(
427                    f"No executor or provider configured for role "
428                    f"'{request.role}' in domain '{request.domain}'."
429                ),
430                metadata={"request": request.to_dict()},
431            )
432
433        if custom_backend is not None:
434            backend = custom_backend
435        elif backend_resolver is not None:
436            backend = backend_resolver(provider_id)
437        else:
438            backend = AgentBackendRegistry.get_backend(provider_id)
439
440        target_cwd = Path(request.cwd).resolve() if request.cwd else Path.cwd()
441
442        full_prompt = request.prompt
443        if request.system_instruction:
444            full_prompt = f"{request.system_instruction}\n\n{request.prompt}"
445
446        try:
447            agent_res: AgentExecutionResult = backend.execute(
448                task=None,
449                cwd=target_cwd,
450                role=f"{request.domain}:{request.role}",
451                prompt_override=full_prompt,
452                timeout_seconds=request.timeout_seconds,
453            )
454        except Exception as exc:
455            elapsed = round(time.time() - start_time, 3)
456            return RoleExecutionResult(
457                domain=request.domain,
458                role=request.role,
459                provider=provider_id,
460                model=model_id,
461                success=False,
462                duration_seconds=elapsed,
463                independence_status=independence_status,
464                error_message=f"Backend execution exception: {exc}",
465                metadata={"request": request.to_dict()},
466            )
467
468        elapsed = round(time.time() - start_time, 3)
469
470        if not agent_res.success:
471            err = (
472                agent_res.error_message
473                or agent_res.stderr
474                or "Provider execution failed"
475            )
476            return RoleExecutionResult(
477                domain=request.domain,
478                role=request.role,
479                provider=provider_id,
480                model=model_id,
481                success=False,
482                raw_output=agent_res.stdout,
483                duration_seconds=elapsed,
484                independence_status=independence_status,
485                error_message=err,
486                timed_out=agent_res.timed_out,
487                metadata={
488                    "request": request.to_dict(),
489                    "agent_result": agent_res.to_dict(),
490                },
491            )
492
493        raw_output = agent_res.stdout
494        structured = extract_structured_output(raw_output)
495
496        return RoleExecutionResult(
497            domain=request.domain,
498            role=request.role,
499            provider=provider_id,
500            model=model_id,
501            success=True,
502            raw_output=raw_output,
503            structured_output=structured,
504            duration_seconds=elapsed,
505            independence_status=independence_status,
506            error_message=None,
507            timed_out=agent_res.timed_out,
508            metadata={
509                "request": request.to_dict(),
510                "agent_result": agent_res.to_dict(),
511            },
512        )

132def extract_structured_output(text: str) -> Optional[Dict[str, Any]]:
133    """Extracts JSON or YAML object from model output text or code blocks."""
134    if not text or not text.strip():
135        return None
136
137    clean = text.strip()
138
139    # 1. Try fenced json or yaml code blocks
140    matches = re.findall(r"```(?:yaml|json)?\s*\n([\s\S]*?)\n```", clean)
141    for block in matches:
142        candidate = block.strip()
143        try:
144            parsed = yaml.safe_load(candidate)
145            if isinstance(parsed, dict):
146                return parsed
147        except Exception:
148            try:
149                parsed_json = json.loads(candidate)
150                if isinstance(parsed_json, dict):
151                    return parsed_json
152            except Exception:
153                pass
154
155    # 2. Extract from Codex/CLI transcript if present
156    codex_match = re.search(
157        r"(?:^|\n)codex\s*\n([\s\S]*?)(?:\ntokens used|\Z)",
158        clean,
159        re.IGNORECASE,
160    )
161    if codex_match:
162        sub_text = codex_match.group(1).strip()
163        sub_matches = re.findall(
164            r"```(?:yaml|json)?\s*\n([\s\S]*?)\n```", sub_text
165        )
166        for block in sub_matches:
167            try:
168                parsed = yaml.safe_load(block.strip())
169                if isinstance(parsed, dict):
170                    return parsed
171            except Exception:
172                pass
173        try:
174            parsed = yaml.safe_load(sub_text)
175            if isinstance(parsed, dict):
176                return parsed
177        except Exception:
178            pass
179
180    # 3. Try direct parse of entire text as JSON or YAML
181    try:
182        parsed_json = json.loads(clean)
183        if isinstance(parsed_json, dict):
184            return parsed_json
185    except Exception:
186        pass
187
188    try:
189        parsed_yaml = yaml.safe_load(clean)
190        if isinstance(parsed_yaml, dict):
191            return parsed_yaml
192    except Exception:
193        pass
194
195    # 4. Try JSON substring if braced
196    first_b = clean.find("{")
197    last_b = clean.rfind("}")
198    if first_b != -1 and last_b != -1 and last_b > first_b:
199        try:
200            parsed_json = json.loads(clean[first_b:last_b + 1])
201            if isinstance(parsed_json, dict):
202                return parsed_json
203        except Exception:
204            pass
205
206    return None

336        """Resolves target provider with honest independence calculation."""
