src.control_plane.agent_readiness

Provider-neutral agent CLI readiness: what each CLI can verifiably do, kept apart from its capacity.

Three evidence levels, each independently cached:

  • Level 1 (local, free): executable, version, authentication status, advertised models. No prompt is ever sent.
  • Level 2 (bounded live smoke, opt-in): one tiny non-mutating prompt through HowlPlane's own backend invocation, in an empty temporary directory, so a pass proves HowlPlane can drive the CLI unattended -- not merely that the CLI works interactively.
  • Level 3 (capacity): only what a CLI actually reports in structured output, or what a real invocation proved (quota, session, or rate limit). A CLI that does not expose its allowance stays UNKNOWN; numbers are never derived or guessed.

An execution-budget timeout is attempt evidence and never becomes capacity here.

111def _auth_from_text(exit_code: int, output: str) -> tuple[bool | None, str]:
112    lowered = output.lower()
113    if re.search(r"not (?:logged|signed) in|unauthenticated|please (?:log|sign) in", lowered):
114        return False, "CLI reports it is not logged in"
115    if exit_code == 0 and re.search(r"\blogged in\b|signed in", lowered):
116        return True, "CLI reports an active login"
117    return None, "CLI auth status was not conclusive"

209def run_probe(argv: Iterable[str], timeout: float = PROBE_TIMEOUT_SECONDS) -> tuple[int, str]:
210    try:
211        completed = subprocess.run(list(argv), capture_output=True, text=True, timeout=timeout,
212                                   stdin=subprocess.DEVNULL)
213    except subprocess.TimeoutExpired:
214        return -1, "probe timed out"
215    except OSError as error:
216        return 127, str(error)
217    return completed.returncode, ANSI.sub("", completed.stdout + ("\n" + completed.stderr if completed.stderr else ""))

239def probe_local(spec: AgentProbeSpec, at: datetime, hosted: bool = True) -> dict[str, Any]:
240    path = shutil.which(spec.binary)
241    section: dict[str, Any] = {"observed_at": iso(at), "installed": path is not None, "executable": path,
242                               "version": None, "models": {"status": "NOT_INSTALLED", "items": [], "default": None,
243                                                           "default_source": None}, "mutation_capable": None,
244                               "hosted": hosted}
245    if not path:
246        return section
247    code, output = run_probe(spec.version_argv)
248    section["version"] = parse_version(output) if code == 0 else None
249    models = section["models"]
250    if not hosted:
251        models.update({"status": "NOT_PROBED", "detail": LOCAL_ONLY_DETAIL})
252    elif spec.models_argv and spec.parse_models:
253        code, output = run_probe(spec.models_argv)
254        items = spec.parse_models(output) if code == 0 else []
255        models.update({"status": "LISTED" if items else "FAILED", "items": items})
256        section["models_exit_code"] = code
257        if code:
258            section["models_error"] = scrub(output)
259    else:
260        models["status"] = "UNSUPPORTED"
261    if spec.default_model:
262        models["default"], models["default_source"] = spec.default_model()
263    section["mutation_capable"] = _mutation_capable(spec.agent)
264    return section

276def probe_auth(spec: AgentProbeSpec, local: dict[str, Any], at: datetime, hosted: bool = True) -> dict[str, Any]:
277    section: dict[str, Any] = {"observed_at": iso(at), "authenticated": None, "source": None, "detail": None}
278    if not local.get("installed"):
279        section["detail"] = "executable not installed"
280        return section

282        section["detail"] = LOCAL_ONLY_DETAIL
283        return section
284    if spec.auth_argv and spec.parse_auth:
285        code, output = run_probe(spec.auth_argv)
286        section["authenticated"], section["detail"] = spec.parse_auth(code, output)

294                             else "CLI has no auth status command; model listing was not conclusive")
295    else:
296        section["detail"] = "CLI exposes no auth status command"

306def parse_capacity_windows(text: str) -> dict[str, Any] | None:
307    """Structured usage windows, only when a CLI emits them (e.g. a `rate_limits` event).
308
309    Returns remaining percentages keyed by window length; nothing is inferred
310    from elapsed time or from failures.
311    """
312    for line in reversed(text.splitlines()):
313        line = line.strip()
314        if "rate_limits" not in line or not line.startswith("{"):
315            continue
316        try:
317            document = json.loads(line)
318        except ValueError:
319            continue
320        limits = _find_key(document, "rate_limits")
321        if not isinstance(limits, dict):
322            continue
323        found: dict[str, Any] = {}
324        for window in limits.values():
325            if not isinstance(window, dict) or not isinstance(window.get("used_percent"), (int, float)):
326                continue
327            remaining = max(0.0, 100.0 - float(window["used_percent"]))
328            minutes = window.get("window_minutes")
329            if minutes == 300:
330                found["five_hour_remaining_percent"] = remaining
331            elif minutes == 10080:
332                found["weekly_remaining_percent"] = remaining
333            reset = window.get("resets_at")
334            if isinstance(reset, (int, float)):
335                found.setdefault("reset_at", iso(datetime.fromtimestamp(reset, timezone.utc)))
336        if found:
337            return found
338    return None

341def _find_key(document: Any, key: str) -> Any:
342    if isinstance(document, dict):
343        if key in document:
344            return document[key]
345        for value in document.values():
346            hit = _find_key(value, key)
347            if hit is not None:
348                return hit

352def run_smoke(agent: str, timeout: int, at: datetime, workspace: str | None = None) -> dict[str, Any]:
353    """One bounded, non-mutating invocation through HowlPlane's real backend path.
354
355    Without a workspace it runs in a fresh temporary directory (global
356    readiness). With one it runs read-only in that directory, proving whether
357    the agent can work there unattended.
358    """
359    from howlplane.control_plane.agent_execution import AgentBackendRegistry
360    from howlplane.control_plane.synthesis.provider_pool import ProviderPoolManager
361    from howlplane.control_plane.task_spec import TaskSpec
362
363    backend = AgentBackendRegistry.get_backend(agent)
364
365    def invoke(directory: str) -> tuple[Any, float]:
366        task = TaskSpec(task_id="agent-doctor-smoke", repository=directory, objective="Readiness smoke test")
367        started = time.monotonic()
368        outcome = backend.execute(task, directory, role="review", prompt_override=SMOKE_PROMPT, timeout_seconds=timeout)
369        return outcome, round(time.monotonic() - started, 2)
370
371    if workspace:
372        result, latency = invoke(workspace)
373    else:
374        with tempfile.TemporaryDirectory(prefix="howl-smoke-") as scratch:
375            result, latency = invoke(scratch)
376    text = f"{result.stdout}\n{result.stderr}"
377    failure = None if result.success else ProviderPoolManager.classify_result(agent, result).value
378    if result.success and SMOKE_TOKEN in result.stdout:
379        status = "PASS"
380    elif result.success:
381        status, failure = "FAIL", "MALFORMED_OUTPUT"
382    elif failure == "WORKSPACE_TRUST_REQUIRED" or is_workspace_trust_refusal(text, agent):
383        status, failure = "WORKSPACE_TRUST_REQUIRED", "WORKSPACE_TRUST_REQUIRED"
384    elif failure == "EXECUTION_PERMISSION_REQUIRED":
385        status = "BLOCKED_PERMISSION"
386    elif failure == "EXECUTION_BUDGET_EXCEEDED":
387        status = "TIMED_OUT"
388    else:
389        status = "FAIL"
390    return {"observed_at": iso(at), "status": status, "exit_code": result.exit_code, "latency_seconds": latency,
391            "failure_class": failure, "model_reported": parse_reported_model(text),
392            "capacity_windows": parse_capacity_windows(text),
393            "error": scrub(result.stderr.strip() or result.error_message or "") if status != "PASS" else None}

427def record_session_outcome(agent: str, model: str, failure: str | None, at: datetime | None = None,
428                           detail: str = "") -> None:
429    """Feed a real orchestrate outcome back as readiness evidence, at the scope it proves."""
430    if agent not in SPECS or failure == "WORKSPACE_TRUST_REQUIRED" or (
431            failure == "EXECUTION_PERMISSION_REQUIRED" and is_workspace_trust_refusal(detail, agent)):
432        # An untrusted directory says nothing about the agent in trusted ones.
433        return
434    moment = at or now()
435
436    def mutate(record: dict[str, Any]) -> None:
437        if failure is None:
438            record["unattended"] = {"observed_at": iso(moment), "value": True, "source": "orchestrate session"}
439        elif failure == "EXECUTION_PERMISSION_REQUIRED":
440            record["unattended"] = {"observed_at": iso(moment), "value": False, "source": "orchestrate session"}
441        elif failure == "AUTHENTICATION_REQUIRED":
442            record["auth"] = {"observed_at": iso(moment), "authenticated": False, "source": "orchestrate session",
443                              "detail": "a real invocation required authentication"}
444        elif failure in FAILURE_TO_CAPACITY:
445            scoped_model = None if model in ("", "UNKNOWN") else model
446            limits = [item for item in record.get("limits", []) if item.get("model") != scoped_model]
447            limits.append(limit_entry(FAILURE_TO_CAPACITY[failure], "model" if scoped_model else "agent", scoped_model,
448                                      "orchestrate session", moment, reason=failure))
449            record["limits"] = limits
450        # EXECUTION_BUDGET_EXCEEDED and every other failure prove nothing about capacity.
451
452    _update_cache(agent, mutate)

484def binding_refusal(agent: str, refusal: dict[str, Any] | None, policy: str) -> dict[str, Any] | None:
485    """An observed refusal still applies unless this policy invokes the CLI differently than it did then.
486
487    A refusal met without the bypass flag says nothing about an invocation that
488    carries it; a refusal met with the flag stays binding under every policy.

496def workspace_status(agent: str, workspace: str | Path, live: bool = False,
497                     cache: dict[str, Any] | None = None, policy: str | None = None) -> dict[str, Any]:
498    """One agent's readiness for one directory: vendor trust, any refusal observed there, and the policy verdict.
499
500    `state` is the vendor's own verdict (plus binding refusals) and is never
501    rewritten by policy; `effective_state` is whether an unattended invocation
502    can run here under `policy`. A refusal the CLI itself returned outranks the
503    vendor store until preparation or a passing workspace smoke clears it, so a
504    wrong trust model can never re-dispatch into a folder that already refused.
505    """
506    path = str(Path(workspace).expanduser().resolve())
507    policy = policy or workspace_trust.resolve_policy()["policy"]
508    hosted = hosted_probes_allowed()
509    vendor = (workspace_trust.probe(agent, path) if live and hosted and agent in workspace_trust.PROBE_ARGV
510              else workspace_trust.check(agent, path))
511    record = (cache if cache is not None else load_cache()).get(agent) or {}
512    refusal = binding_refusal(agent, (record.get("workspaces") or {}).get(path), policy)
513    smoke = (record.get("workspace_smokes") or {}).get(path)
514    state = workspace_trust.TRUST_REQUIRED if refusal else vendor["state"]
515    verdict = workspace_trust.effective(agent, vendor["state"], policy)
516    if refusal:
517        # The CLI refused even the way this policy invokes it: no policy makes that usable.
518        verdict.update({"effective_state": workspace_trust.TRUST_REQUIRED,
519                        "detail": "the CLI refused this directory under this policy"})
520    return {"agent": agent, "workspace": path, "state": state,
521            "ready": verdict["effective_state"] == workspace_trust.READY,
522            "vendor_state": verdict["vendor_state"] if not refusal else workspace_trust.TRUST_REQUIRED,
523            "policy": policy, "effective_state": verdict["effective_state"], "mechanism": verdict["mechanism"],
524            "policy_detail": verdict["detail"],
525            "vendor": vendor, "observed_refusal": refusal,
526            "live_smoke": None if not smoke else {"status": smoke.get("status"), "verified_at": smoke.get("observed_at"),
527                                                  "failure_class": smoke.get("failure_class")}}

530def workspace_blocked(agent: str, workspace: str | Path | None, policy: str | None = None) -> str | None:
531    """Routing view: why `agent` must not be dispatched into `workspace` under the trust policy, or None."""
532    if not workspace or agent not in workspace_trust.ADAPTERS:
533        return None
534    try:
535        status = workspace_status(agent, workspace, policy=policy)
536    except OSError:
537        return None
538    if status["effective_state"] != workspace_trust.TRUST_REQUIRED:
539        return None
540    source = "refused earlier" if status["observed_refusal"] else status["vendor"]["detail"]
541    return f"workspace trust required for {status['workspace']} ({source}; policy {status['policy']})"

567def summarize(agent: str, record: dict[str, Any], at: datetime | None = None) -> dict[str, Any]:
568    """The routing/report view of one cached record. Absent evidence is UNKNOWN, never AVAILABLE."""
569    moment = at or now()
570    spec = SPECS[agent]
571    local = record.get("local") or {}
572    auth = record.get("auth") or {}
573    smoke = record.get("live_smoke") or {}
574    unattended = record.get("unattended") or {}
575    smoke_age = age_seconds(smoke.get("observed_at"), moment)
576    smoke_status = smoke.get("status", "NOT_RUN")
577    smoke_stale = smoke_age is not None and smoke_age >= TTL_SECONDS["live_smoke"]
578    # Unattended evidence: a real session outranks a smoke; stale evidence reverts to UNKNOWN.
579    unattended_value, unattended_source = None, "no unattended invocation evidence"
580    if fresh(unattended, "unattended", moment):
581        unattended_value, unattended_source = unattended.get("value"), unattended.get("source")
582    elif smoke_status in ("PASS", "BLOCKED_PERMISSION") and fresh(smoke, "unattended", moment):
583        unattended_value = smoke_status == "PASS"
584        unattended_source = "live smoke"
585    limits = active_limits(record, moment)
586    agent_limit = next((item for item in limits if item.get("scope") in ("agent", "provider")), None)
587    windows = smoke.get("capacity_windows") if fresh(smoke, "capacity", moment) else None
588    capacity: dict[str, Any] = {"state": "UNKNOWN", "visibility": "UNKNOWN", "source": None, "scope": None,
589                                "reason": "CLI does not expose remaining allowance",
590                                "five_hour_remaining_percent": None, "weekly_remaining_percent": None,
591                                "reset_at": None, "limits": limits}
592    if windows:
593        capacity.update({"visibility": "VISIBLE", "source": "cli", "reason": "reported by the CLI", **windows})
594        remaining = [value for key, value in windows.items() if key.endswith("_percent")]
595        capacity["state"] = "QUOTA_EXHAUSTED" if remaining and min(remaining) <= 0 else "AVAILABLE"
596    if agent_limit:
597        capacity.update({"state": agent_limit["state"], "scope": agent_limit["scope"], "source": agent_limit["source"],
598                         "reason": agent_limit.get("reason"), "reset_at": agent_limit["expires_at"]})
599    # Stale auth evidence, positive or negative, reverts to UNKNOWN: no permanent blacklist.
600    authenticated = auth.get("authenticated") if fresh(auth, "auth", moment) else None
601    models = local.get("models") or {"status": "UNKNOWN", "items": [], "default": None, "default_source": None}
602    return {
603        "agent": agent, "name": spec.name, "backend": spec.binary,
604        "executable": local.get("executable"), "installed": bool(local.get("installed")),
605        "version": local.get("version"),
606        "authenticated": authenticated, "auth_detail": auth.get("detail"), "auth_source": auth.get("source"),
607        "unattended_execution": unattended_value, "unattended_source": unattended_source,
608        "mutation_capable": local.get("mutation_capable"),
609        "live_smoke": {"status": "STALE" if smoke_stale and smoke_status != "NOT_RUN" else smoke_status,
610                       "last_status": smoke_status, "verified_at": smoke.get("observed_at"),
611                       "age_seconds": None if smoke_age is None else round(smoke_age),
612                       "latency_seconds": smoke.get("latency_seconds"), "exit_code": smoke.get("exit_code"),
613                       "failure_class": smoke.get("failure_class"), "model_reported": smoke.get("model_reported"),
614                       "error": smoke.get("error")},
615        "models": models,
616        "capacity": capacity,
617        "last_verified_at": max(filter(None, (local.get("observed_at"), auth.get("observed_at"),
618                                              smoke.get("observed_at"))), default=None),
619        "workspace_refusals": dict(record.get("workspaces") or {}),
620    }

623def evaluate(agents: Iterable[str] | None = None, live: bool = False, refresh: bool = False,
624             smoke_timeout: int = DEFAULT_SMOKE_TIMEOUT_SECONDS,
625             smoke_runner: Callable[..., dict[str, Any]] | None = None,
626             workspace: str | None = None) -> list[dict[str, Any]]:
627    """Refresh what is stale (or everything with `refresh`), then summarize.
628
629    Without `live` no prompt is sent. With it, at most one smoke runs per
630    eligible agent, and a fresh PASS is reused instead of spending another call.
631    With a `workspace`, the live smoke runs read-only in that directory instead
632    of a temporary one, and its verdict is kept against that directory.
633    """
634    selected = [agent for agent in AGENT_ORDER if agents is None or agent in set(agents)]
635    cache = load_cache()
636    moment = now()
637    hosted = hosted_probes_allowed()
638    for agent in selected:
639        record = cache.setdefault(agent, {})
640        relocated = not local_fresh(SPECS[agent], record.get("local"), moment, hosted)
641        if refresh or relocated:
642            record["local"] = probe_local(SPECS[agent], moment, hosted)
643        if refresh or relocated or not fresh(record.get("auth"), "auth", moment):
644            record["auth"] = probe_auth(SPECS[agent], record["local"], moment, hosted)
645    if live and hosted:
646        due = []
647        for agent in selected:
648            record = cache[agent]
649            if not record["local"].get("installed") or record["auth"].get("authenticated") is False:
650                continue
651            smoke = (record.get("workspace_smokes") or {}).get(workspace) if workspace else record.get("live_smoke")
652            smoke = smoke or {}
653            if refresh or smoke.get("status") != "PASS" or not fresh(smoke, "live_smoke", moment):
654                due.append(agent)
655        runner = smoke_runner or run_smoke
656
657        def smoke_for(agent: str) -> dict[str, Any]:
658            return runner(agent, smoke_timeout, moment, workspace=workspace) if workspace else runner(
659                agent, smoke_timeout, moment)
660
661        with ThreadPoolExecutor(max_workers=max(1, len(due))) as pool:
662            outcomes = dict(zip(due, pool.map(smoke_for, due)))
663        for agent, smoke in outcomes.items():
664            record = cache[agent]
665            if workspace:
666                record.setdefault("workspace_smokes", {})[workspace] = smoke
667                if smoke["status"] == "WORKSPACE_TRUST_REQUIRED":
668                    record.setdefault("workspaces", {})[workspace] = refusal_entry(agent, "review", "live smoke",
669                                                                                   moment)
670                elif smoke["status"] == "PASS":
671                    record.get("workspaces", {}).pop(workspace, None)
672                    # A pass in a real workspace is the strongest unattended evidence there is.
673                    record["unattended"] = {"observed_at": iso(moment), "value": True,
674                                            "source": "live smoke in workspace"}
675            else:
676                record["live_smoke"] = smoke
677            failure = smoke.get("failure_class")
678            if failure in FAILURE_TO_CAPACITY:
679                model = smoke.get("model_reported")
680                record["limits"] = [item for item in record.get("limits", []) if item.get("model") != model] + [
681                    limit_entry(FAILURE_TO_CAPACITY[failure], "model" if model else "agent", model, "live smoke",
682                                moment, reset_at=(smoke.get("capacity_windows") or {}).get("reset_at"), reason=failure)]
683            if failure == "AUTHENTICATION_REQUIRED":
684                record["auth"] = {"observed_at": iso(moment), "authenticated": False, "source": "live smoke",
685                                  "detail": "the smoke invocation required authentication"}
686    for record in cache.values():
687        record["limits"] = active_limits(record, moment)
688    save_cache(cache)
689    return [summarize(agent, cache[agent], moment) for agent in selected]

720def _smoke_text(summary: dict[str, Any]) -> str:
721    smoke = summary["live_smoke"]
722    status = smoke["status"]
723    if status == "NOT_RUN":
724        return "NOT RUN (use --live)"
725    age = smoke.get("age_seconds")
726    when = f" {_duration(age)} ago" if age is not None else ""
727    if smoke["last_status"] == "PASS":
728        return f"{'PASS' if status != 'STALE' else 'STALE (last PASS'}{when}, {smoke['latency_seconds']}s{')' if status == 'STALE' else ''}"
729    if smoke["last_status"] == "BLOCKED_PERMISSION":
730        return f"BLOCKED — permission required{when}"
731    if smoke["last_status"] == "WORKSPACE_TRUST_REQUIRED":
732        return (f"BLOCKED — workspace trust required{when} (the smoke's fresh directory is untrusted;"
733                " check a real workspace with --repo)")
734    return f"{smoke['last_status']}{when} ({smoke.get('failure_class') or 'no detail'})"

742def _capacity_text(capacity: dict[str, Any]) -> list[str]:
743    lines = []
744    if capacity["visibility"] == "UNKNOWN" and capacity["state"] == "UNKNOWN":
745        lines.append("UNKNOWN — CLI does not expose remaining allowance")
746    else:
747        lines.append(f"{capacity['state']} (source: {capacity['source']}"
748                     + (f", until {capacity['reset_at']}" if capacity.get("reset_at") else "") + ")")
749    for key, label in (("five_hour_remaining_percent", "Five-hour"), ("weekly_remaining_percent", "Weekly")):
750        if capacity.get(key) is not None:
751            lines.append(f"{label}: {capacity[key]:.0f}% remaining (reported by CLI)")
752    for item in capacity["limits"]:
753        if item.get("scope") == "model":
754            lines.append(f"Model {item['model']}: {item['state']} until {item['expires_at']} ({item['source']})")
755    return lines

781def render_workspace(report: dict[str, Any]) -> list[str]:
782    scope = report["scope"]
783    lines = ["WORKSPACE READINESS", f"  Workspace:      {report['workspace']}" + ("" if report["exists"] else " (not created yet)"),
784             "  Authorized:     " + (f"YES — {scope['repo_root']} (factory root {scope['factory_root']})"

786    resolved = report.get("workspace_trust_policy") or {}
787    lines.insert(3, f"  Trust policy:   {str(resolved.get('policy', '')).upper()} (from {resolved.get('source')})")
788    for agent, status in report["agents"].items():
789        spec = workspace_trust.adapter(agent)
790        lines.append(f"  {SPECS[agent].name}")
791        lines.append(f"    Vendor trust:        {_workspace_text(status)}")
792        lines.append(f"    Trust policy:        {status['policy'].upper()}")
793        lines.append(f"    Effective workspace: {TRUST_TEXT.get(status['effective_state'], status['effective_state'])}"

806def render(summaries: list[dict[str, Any]], workspace: dict[str, Any] | None = None) -> str:
807    lines = ["HOWLPLANE AGENT DOCTOR", "", "GLOBAL READINESS", ""]
808    for summary in summaries:
809        lines.append(summary["name"])
810        rows: list[tuple[str, str]] = [("Backend", summary["backend"]),
811                                       ("CLI", "AVAILABLE" if summary["installed"] else "NOT INSTALLED")]
812        if summary["installed"]:
813            models = summary["models"]
814            model_text = {"LISTED": f"{len(models['items'])} listed ({', '.join(models['items'][:4])}"
815                                    f"{', …' if len(models['items']) > 4 else ''})",
816                          "UNSUPPORTED": "not listed by this CLI", "FAILED": "listing failed",
817                          "NOT_PROBED": LOCAL_ONLY_DETAIL}.get(models["status"], models["status"])
818            if models.get("default"):
819                model_text += f"; default {models['default']} ({models['default_source']})"
820            rows += [("Version", summary["version"] or "unknown"),
821                     ("Auth", {True: "READY", False: "NOT LOGGED IN", None: "UNKNOWN"}[summary["authenticated"]]
822                      + (f" — {summary['auth_detail']}" if summary["auth_detail"] else "")),
823                     ("Unattended", {True: "YES", False: "INTERACTIVE_ONLY", None: "UNVERIFIED"}[summary["unattended_execution"]]
824                      + (f" ({summary['unattended_source']})" if summary["unattended_execution"] is not None else "")),
825                     ("Edits allowed", yes_no(summary["mutation_capable"])),
826                     ("Models", model_text),
827                     ("Live smoke", _smoke_text(summary))]
828            if summary["live_smoke"].get("model_reported"):
829                rows.append(("Model seen", summary["live_smoke"]["model_reported"]))
830            capacity = _capacity_text(summary["capacity"])
831            rows.append(("Capacity", capacity[0]))
832            rows += [("", extra) for extra in capacity[1:]]
833            rows.append(("Verified", summary["last_verified_at"] or "never"))
834            if workspace and summary["agent"] in workspace["agents"]:
835                status = workspace["agents"][summary["agent"]]
836                rows += [("Vendor trust", TRUST_TEXT.get(status["vendor_state"], status["vendor_state"])),
837                         ("Trust policy", status["policy"].upper()),
838                         ("Workspace", TRUST_TEXT.get(status["effective_state"], status["effective_state"]))]
839                if status.get("mechanism"):
840                    rows.append(("Mechanism", status["mechanism"]))
841        lines += [f"  {label + ':' if label else '':<15} {value}" for label, value in rows]
842        lines.append("")
843    if workspace:
844        lines += render_workspace(workspace)
845    return "\n".join(lines).rstrip() + "\n"

856def factory_readiness(summaries: list[dict[str, Any]], workspace: dict[str, Any] | None = None) -> dict[str, Any]:
857    """Can an unattended campaign run: an autonomous implementer plus an independent auditor?
858
859    With a workspace report, an agent that would meet a trust prompt in the
860    Factory workspace is not a worker there, however ready it is elsewhere.
861    """
862    per_agent = (workspace or {}).get("agents", {})
863
864    def workspace_ok(summary: dict[str, Any]) -> bool:
865        # Effective readiness under the trust policy decides; vendor trust alone never does.
866        status = per_agent.get(summary["agent"])
867        return status is None or status.get("effective_state", status["state"]) != workspace_trust.TRUST_REQUIRED
868
869    def autonomous(summary: dict[str, Any]) -> bool:
870        return (summary["installed"] and summary["authenticated"] is not False
871                and summary["unattended_execution"] is not False and workspace_ok(summary)
872                and summary["capacity"]["state"] not in ("QUOTA_EXHAUSTED", "SESSION_EXHAUSTED", "RATE_LIMITED"))
873    implementers = [s["name"] for s in summaries if autonomous(s) and s["mutation_capable"] is not False]
874    reviewers = [s["name"] for s in summaries if autonomous(s)]
875    independent_audit = any(auditor != implementer for implementer in implementers for auditor in reviewers)
876    verified = [s["name"] for s in summaries if autonomous(s) and s["unattended_execution"] is True]
877    gated = [s["name"] for s in summaries if s["installed"] and not workspace_ok(s)]
878    # Trust-gated agents do not block a campaign others can run, but it is degraded.
879    status = "BLOCKED" if not implementers else (
880        "READY" if independent_audit and set(verified) >= set(reviewers) and not gated else "DEGRADED")
881    return {
882        "status": status,
883        "implementation_capable": implementers,
884        "review_capable": reviewers,
885        "interactive_only": [s["name"] for s in summaries if s["installed"] and s["unattended_execution"] is False],
886        "unavailable": [s["name"] for s in summaries if not s["installed"] or s["authenticated"] is False],
887        "capacity_unknown": [s["name"] for s in summaries if s["installed"] and s["capacity"]["state"] == "UNKNOWN"],
888        "known_exhausted": [f"{s['name']}: {item['state']}" + (f" ({item['model']})" if item.get("model") else "")
889                            for s in summaries for item in s["capacity"]["limits"]],
890        "live_smoke_passed": [s["name"] for s in summaries if s["live_smoke"]["status"] == "PASS"],
891        "workspace": None if workspace is None else workspace["workspace"],
892        "workspace_authorized": None if workspace is None else workspace["authorized"],
893        "workspace_trust_policy": None if workspace is None else (workspace.get("workspace_trust_policy") or {}).get(
894            "policy"),
895        "workspace_trust_bypassed": [SPECS[agent].name for agent, status in per_agent.items()
896                                     if status.get("mechanism") and status.get("effective_state") == workspace_trust.READY
897                                     and status.get("vendor_state") != workspace_trust.READY],
898        # With a workspace, trust there is what gates Factory; a fresh-directory
899        # smoke refusal only describes that throwaway directory.
900        "workspace_trust_required": (gated if workspace is not None else
901                                     [s["name"] for s in summaries
902                                      if s["live_smoke"].get("last_status") == "WORKSPACE_TRUST_REQUIRED"]),
903        "unverified": [name for name in reviewers if name not in verified],
904        "independent_audit_available": independent_audit,
905    }
