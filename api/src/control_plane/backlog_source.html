src.control_plane.backlog_source

backlog_source.py

Read-only selection of the next eligible item from a project's ranked markdown backlogs (bugs.md and improvements.md).

This is the implementation behind the select_next_evidence_backed_task authority action, which existed only as a string literal in authority_profile.py until now. No provider is invoked here and nothing is written: the backlog is parsed, filtered and ranked deterministically, and the caller decides what to do with the result.

Two properties of the real backlogs drive the parsing rules, and both were found by reading HowlFrame's files rather than assuming a shape:

  1. A file can hold several tables. improvements.md carries the live ranked backlog plus two historical "V2"/"V3" tables with a different column count. Parsing every pipe row would offer work shipped months ago as pending, so only the table under the first ## Ranked Backlog heading is read, and parsing stops at the next ##.

  2. Pending is not a prefix. The real status column contains Pending, Pending ⚠️ below floor and Pending — blocked on #88. A startswith test would hand an unattended run an item that is explicitly below the ROI floor -- which the backlog says needs human confirmation before being worked -- or one blocked on another item. Only an exact Pending is eligible.

 40# The heading that introduces a file's live ranked backlog. Only the first
 41# occurrence is read.
 42RANKED_BACKLOG_HEADING = "## Ranked Backlog"

104def _split_row(line: str) -> List[str]:
105    return [cell.strip() for cell in line.split("|")[1:-1]]

108def _parse_score(cell: str) -> Optional[float]:
109    """Reads the leading number out of a `2.5 (5×1÷2)` score cell."""
110    match = _SCORE_PATTERN.match(cell)
111    return float(match.group(1)) if match else None
112
113
114def _parse_title(cell: str) -> tuple:
115    """Returns (title, anchor) for a `[text](#anchor)` or plain cell."""
116    match = _LINK_PATTERN.match(cell)
117    if match:
118        return match.group("text"), match.group("anchor")
119    return cell.strip("*").strip(), None

122def parse_backlog_file(path: Union[str, Path]) -> List[BacklogItem]:
123    """Parses the live ranked table out of one backlog file.
124
125    Only the table under the *first* `## Ranked Backlog` heading is read, and
126    reading stops at the next `##` heading, so historical tables later in the
127    same file are never mistaken for open work.
128    """
129    path = Path(path)
130    text = path.read_text(encoding="utf-8")
131    lines = text.splitlines()
132
133    start = next(
134        (i for i, line in enumerate(lines) if line.startswith(RANKED_BACKLOG_HEADING)),
135        None,
136    )
137    if start is None:
138        raise BacklogParseError(
139            f"{path} has no '{RANKED_BACKLOG_HEADING}' heading, so it has no "
140            f"live ranked table to read."
141        )
142    end = next(
143        (i for i, line in enumerate(lines[start + 1:], start + 1) if line.startswith("## ")),
144        len(lines),
145    )
146
147    kind = "bug" if path.name == "bugs.md" else "improvement"
148    items: List[BacklogItem] = []
149    for line in lines[start:end]:
150        match = _ROW_PATTERN.match(line)
151        if not match:
152            continue
153        cells = _split_row(line)
154        if len(cells) < 3:
155            continue
156        title, anchor = _parse_title(cells[1])
157        items.append(
158            BacklogItem(
159                item_id=cells[0],
160                source_file=path.name,
161                title=title,
162                anchor=anchor,
163                status=cells[2],
164                score=_parse_score(cells[3]) if len(cells) > 3 else None,
165                rationale=cells[-1] if len(cells) > 4 else None,
166                kind=kind,
167            )
168        )
169    return items

172class BacklogSource:
173    """Deterministic, read-only next-item selection over a project's backlogs."""
174
175    def __init__(
176        self,
177        repo_root: Union[str, Path],
178        backlog_files: Sequence[str] = DEFAULT_BACKLOG_FILES,
179        roi_floor: float = 0.5,
180    ):
181        self.repo_root = Path(repo_root).resolve()
182        self.backlog_files = list(backlog_files)
183        # The floor both backlogs document. A row at or above it may be worked
184        # unattended; one below it is flagged in the backlog itself and needs
185        # explicit human confirmation, so it is never selected here.
186        self.roi_floor = roi_floor
187
188    def select(self) -> BacklogSelection:
189        """Reads every configured backlog and ranks what may be worked.
190
191        Bugs are returned ahead of improvements, and within each file rows keep
192        their committed rank order rather than being re-sorted by score: the
193        backlog's own ordering already encodes judgement this code cannot
194        reconstruct.
195        """
196        selection = BacklogSelection()
197        for filename in self.backlog_files:
198            path = self.repo_root / filename
199            if not path.is_file():
200                continue
201            selection.files_read.append(filename)
202            for item in parse_backlog_file(path):
203                reason = self._ineligibility_reason(item)
204                if reason is None:
205                    selection.eligible.append(item)
206                elif item.status.startswith(ELIGIBLE_STATUS):
207                    # Only near-misses are worth recording. Rows that are simply
208                    # Done would bury the interesting exclusions.
209                    selection.excluded.append({
210                        "item_id": item.item_id,
211                        "source_file": item.source_file,
212                        "status": item.status,
213                        "reason": reason,
214                    })
215        return selection
216
217    def _ineligibility_reason(self, item: BacklogItem) -> Optional[str]:
218        """Why this row may not be worked unattended, or None if it may."""
219        if not item.is_eligible:
220            # Covers "Pending ⚠️ below floor" and "Pending — blocked on #88",
221            # which a prefix match would have wrongly admitted.
222            return f"STATUS_NOT_ELIGIBLE:{item.status}"
223        if item.score is not None and item.score < self.roi_floor:
224            return f"BELOW_ROI_FLOOR:{item.score}"
225        return None
226
227    def item_detail(self, item: BacklogItem) -> str:
228        """Returns the item's detail section, which is its problem statement.
229
230        A governed task needs the symptom, root cause and deterministic
231        acceptance the backlog already records; the one-line table rationale is
232        not enough to implement from.
233        """
234        path = self.repo_root / item.source_file
235        if not path.is_file():
236            return ""
237        lines = path.read_text(encoding="utf-8").splitlines()
238        start = next(
239            (i for i, line in enumerate(lines)
240             if line.startswith(f"### {item.item_id}.")),
241            None,
242        )
243        if start is None:
244            return ""
245        end = next(
246            (i for i, line in enumerate(lines[start + 1:], start + 1)
247             if line.startswith("### ")),
248            len(lines),
249        )
250        return "\n".join(lines[start:end]).strip()
