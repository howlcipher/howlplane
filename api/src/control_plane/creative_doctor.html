src.control_plane.creative_doctor

howlplane doctor --creative: preflight for the Dream -> Writer -> Create pipeline.

Run 5 lost time to failures that only surfaced at runtime: a repository virtualenv carrying a stale howl-provider-core (ImportError: cannot import name 'classify_failure'), virtualenvs pinned to an interpreter that had since been upgraded, and no way to tell which component builds were installed. This module finds those problems before a workflow starts.

Every check is read-only. It never installs, upgrades, writes, or prints a credential; command configurations are reported by executable name only. Statuses are PASS, WARN and FAIL, and every non-PASS result carries a concrete remediation.

 89def _git_head(path: Path) -> Optional[str]:
 90    if not (path / ".git").exists():
 91        return None
 92    try:
 93        result = subprocess.run(["git", "-C", str(path), "rev-parse", "HEAD"], capture_output=True,
 94                                text=True, timeout=5, check=False)
 95    except (OSError, subprocess.TimeoutExpired):
 96        return None
 97    return result.stdout.strip() if result.returncode == 0 else None

100def installed_source(dist: Any) -> Dict[str, Any]:
101    """Version plus where it came from: a VCS commit or an editable checkout."""
102    info = _direct_url(dist)
103    source: Dict[str, Any] = {"version": dist.version, "kind": "index", "commit": None, "path": None}
104    if info.get("vcs_info"):
105        source.update(kind="vcs", commit=info["vcs_info"].get("commit_id"))
106    elif info.get("dir_info", {}).get("editable"):
107        url = info.get("url", "")
108        path = Path(url[7:]) if url.startswith("file://") else None
109        source.update(kind="editable", path=str(path) if path else None,
110                      commit=_git_head(path) if path else None)
111    return source

114def core_pin(dist: Any) -> Optional[str]:
115    for requirement in dist.requires or []:
116        match = _PIN.search(requirement)
117        if match:
118            return match.group("sha")
119    return None

122def _missing_symbols(component: str, importer: Callable[[str], Any]) -> List[str]:
123    missing = []
124    for module_name, symbols in REQUIRED_SYMBOLS[component].items():
125        try:
126            module = importer(module_name)
127        except Exception as error:  # noqa: BLE001 - any import failure is the finding
128            missing.append(f"{module_name} ({type(error).__name__}: {str(error)[:160]})")
129            continue
130        missing.extend(f"{module_name}.{s}" for s in symbols if not hasattr(module, s))
131    return missing

134def check_interpreter(python: str, version: tuple, prefix: str, base_prefix: str) -> List[Check]:
135    checks = []
136    shown = ".".join(str(v) for v in version[:3])
137    if tuple(version[:2]) < MIN_PYTHON:
138        checks.append(Check("python", "Python interpreter", FAIL, f"{python} is Python {shown}",
139                            "Use Python 3.11 or newer for the Howl components."))
140    else:
141        checks.append(Check("python", "Python interpreter", PASS, f"{python} (Python {shown})"))
142    if prefix == base_prefix:
143        checks.append(Check("venv", "Virtual environment", WARN, "No virtual environment is active",
144                            "Create one: python -m venv .venv && . .venv/bin/activate"))
145        return checks
146    cfg = Path(prefix) / "pyvenv.cfg"
147    values: Dict[str, str] = {}
148    if cfg.is_file():
149        for line in cfg.read_text(errors="replace").splitlines():
150            if "=" in line:
151                key, value = line.split("=", 1)
152                values[key.strip()] = value.strip()
153    home = values.get("home")
154    recorded = values.get("version_info") or values.get("version")
155    problems = []
156    if home and not Path(home).exists():
157        problems.append(f"pyvenv.cfg home {home} no longer exists")
158    recorded_minor = ".".join((recorded or "").split(".")[:2])
159    if recorded and recorded_minor != ".".join(shown.split(".")[:2]):
160        # site-packages is per minor version: a minor mismatch breaks imports.
161        problems.append(f"venv was created for Python {recorded}, running {shown}")
162    fix = f"Recreate it: python -m venv --clear {prefix} and reinstall the Howl components."
163    if problems:
164        checks.append(Check("venv", "Virtual environment", FAIL,
165                            f"Stale virtualenv {prefix}: " + "; ".join(problems), fix))
166    elif recorded and not shown.startswith(".".join(recorded.split(".")[:3])):
167        checks.append(Check("venv", "Virtual environment", WARN,
168                            f"{prefix} was created for Python {recorded}; running {shown} "
169                            "(patch drift, imports still resolve)", fix))
170    else:
171        checks.append(Check("venv", "Virtual environment", PASS, f"{prefix}"))
172    return checks

175def check_components(lookup: Callable[[str], Any], importer: Callable[[str], Any],
176                     repos_root: Optional[Path]) -> List[Check]:
177    checks: List[Check] = []
178    sources: Dict[str, Dict[str, Any]] = {}
179    for name in COMPONENTS:
180        dist = _dist(name, lookup)
181        if dist is None:
182            checks.append(Check(f"package.{name}", f"{name} installed", FAIL, "Not installed",
183                                f"pip install 'git+https://github.com/howlcipher/{name}.git@main'"))
184            continue
185        source = installed_source(dist)
186        sources[name] = source
187        where = source["commit"][:12] if source["commit"] else source["kind"]
188        missing = _missing_symbols(name, importer)
189        if missing:
190            checks.append(Check(
191                f"package.{name}", f"{name} installed", FAIL,
192                f"{name} {source['version']} ({where}) lacks: {', '.join(missing[:4])}",
193                f"Reinstall {name} from the merged main branch into this interpreter.",
194                data=source))
195        else:
196            checks.append(Check(f"package.{name}", f"{name} installed", PASS,
197                                f"{name} {source['version']} ({where})", data=source))
198        if repos_root is not None and source["kind"] != "editable":
199            head = _git_head(repos_root / name)
200            if head and source["commit"] and head != source["commit"]:
201                checks.append(Check(
202                    f"repo.{name}", f"{name} checkout vs install", WARN,
203                    f"Local checkout {head[:12]} differs from installed {source['commit'][:12]}",
204                    f"pip install -e {repos_root / name}  (or update the checkout)"))
205
206    core = sources.get("howl-provider-core")
207    for consumer in CONSUMERS_OF_CORE:
208        dist = _dist(consumer, lookup)
209        if dist is None or core is None:
210            continue
211        pin = core_pin(dist)
212        core_commit = core.get("commit")
213        if pin is None:
214            continue
215        label = f"{consumer} -> howl-provider-core"
216        if core_commit and core_commit.startswith(pin):
217            checks.append(Check(f"compat.{consumer}.core", label, PASS, f"pin {pin[:12]} installed"))
218        elif _missing_symbols("howl-provider-core", importer):
219            checks.append(Check(
220                f"compat.{consumer}.core", label, FAIL,
221                f"{consumer} requires provider-core {pin[:12]}; installed "
222                f"{(core_commit or core['kind'])[:12]} is missing required symbols",
223                f"pip install 'git+https://github.com/howlcipher/howl-provider-core.git@{pin}'"))
224        else:
225            checks.append(Check(
226                f"compat.{consumer}.core", label, WARN,
227                f"{consumer} pins {pin[:12]}; installed {(core_commit or core['kind'])[:12]} "
228                "differs but exports every required symbol",
229                f"pip install 'git+https://github.com/howlcipher/howl-provider-core.git@{pin}'"))
230    checks.append(check_core_pin_agreement(lookup))
231    checks.append(check_writer_create_contract(importer))
232    return checks

235def check_core_pin_agreement(lookup: Callable[[str], Any]) -> Check:
236    """Consumers pinning different provider-core commits cannot be co-installed.
237
238    pip treats two direct-URL requirements for the same project as a conflict,
239    so a fresh environment fails to resolve even when the pinned trees match.
240    """
241    label = "Shared provider-core pin"
242    pins = {}
243    for consumer in CONSUMERS_OF_CORE:
244        dist = _dist(consumer, lookup)
245        pin = core_pin(dist) if dist is not None else None
246        if pin:
247            pins[consumer] = pin
248    if len(set(pins.values())) <= 1:
249        detail = f"all consumers pin {next(iter(pins.values()))[:12]}" if pins else "no pinned consumers installed"
250        return Check("compat.core_pin_agreement", label, PASS, detail)
251    listing = ", ".join(f"{c}@{p[:12]}" for c, p in sorted(pins.items()))
252    return Check("compat.core_pin_agreement", label, FAIL,
253                 f"Consumers pin different howl-provider-core commits ({listing}); "
254                 "a fresh install cannot resolve them together",
255                 "Align every consumer's howl-provider-core pin to one commit and reinstall.")

258def check_writer_create_contract(importer: Callable[[str], Any]) -> Check:
259    """Create validates Writer packages against a vendored schema; both copies must agree."""
260    label = "Writer -> Create contract"
261    try:
262        writer = importer("howlwriter.schemas")
263        create = importer("howlcreate")
264    except Exception:  # noqa: BLE001
265        return Check("compat.writer.create", label, FAIL, "howlwriter or howlcreate is not importable",
266                     "Install both components from merged main.")
267    name = "howlwriter.copy_package.v1.schema.json"
268    writer_schema = Path(writer.__file__).with_name(name)
269    create_schema = Path(create.__file__).parent / "schemas" / name
270    if not writer_schema.is_file() or not create_schema.is_file():
271        missing = "howlwriter" if not writer_schema.is_file() else "howlcreate"
272        return Check("compat.writer.create", label, FAIL, f"{missing} does not ship {name}",
273                     f"Upgrade {missing} to a build with the native copy-package contract.")
274    if json.loads(writer_schema.read_text()) != json.loads(create_schema.read_text()):
275        return Check("compat.writer.create", label, FAIL,
276                     "Writer and Create copy-package schemas differ",
277                     "Install matching merged builds of howlwriter and howlcreate.")
278    return Check("compat.writer.create", label, PASS, "copy_package/v1 schemas match")

281def _shebang_target(path: str) -> Optional[str]:
282    try:
283        with open(path, "rb") as handle:
284            first = handle.readline(512).decode("utf-8", "replace").strip()
285    except OSError:
286        return None
287    if not first.startswith("#!"):
288        return None
289    parts = first[2:].split()
290    return parts[0] if parts else None

293def check_entrypoints(which: Callable[[str], Optional[str]]) -> List[Check]:
294    checks = []
295    for name, command in CLIS.items():
296        path = which(command)
297        if path is None:
298            checks.append(Check(f"cli.{name}", f"{command} CLI", WARN, "Not on PATH",
299                                f"Activate the environment that has {name}, or run python -m {name}.cli"))
300            continue
301        target = _shebang_target(path)
302        if target and not target.endswith("/env") and not Path(target).exists():
303            checks.append(Check(f"cli.{name}", f"{command} CLI", FAIL,
304                                f"{path} points at missing interpreter {target}",
305                                f"Reinstall {name} into a live environment; the entry point is stale."))
306        else:
307            checks.append(Check(f"cli.{name}", f"{command} CLI", PASS, path))
308    return checks

311def check_policy(env: Mapping[str, str]) -> Check:
312    value = env.get("HOWL_FORBID_LOCAL_INFERENCE", "")
313    if value.lower() in {"1", "true", "yes"}:
314        return Check("policy.local", "HOWL_FORBID_LOCAL_INFERENCE", PASS, "Local inference is forbidden")
315    return Check("policy.local", "HOWL_FORBID_LOCAL_INFERENCE", WARN,
316                 f"Not set (value: {value or 'unset'}); local inference is only denied by default",

320def check_local_providers(env: Mapping[str, str], config_argv: Optional[List[str]]) -> Check:
321    forbid = env.get("HOWL_FORBID_LOCAL_INFERENCE", "").lower() in {"1", "true", "yes"}
322    found = [name for name in _LOCAL_ENV if env.get(name) and (
323        name.startswith(("OLLAMA", "LLAMA", "LMSTUDIO", "LOCALAI", "VLLM")) or _LOCAL_HOST.search(env[name]))]
324    if config_argv and Path(config_argv[0]).name.lower() in _LOCAL_LAUNCHERS:
325        found.append(f"command config launcher {Path(config_argv[0]).name}")
326    if not found:
327        return Check("policy.local_config", "Local model configuration", PASS, "None detected")
328    status = WARN if forbid else FAIL
329    return Check("policy.local_config", "Local model configuration", status,
330                 "Local inference configuration present: " + ", ".join(found)
331                 + (" (blocked by HOWL_FORBID_LOCAL_INFERENCE)" if forbid else ""),
332                 "Unset these variables or remove the local launcher; keep HOWL_FORBID_LOCAL_INFERENCE=1.")

335def check_command_config(path: Optional[Path]) -> tuple:
336    if path is None:
337        return Check("provider.remote", "Remote provider configuration", WARN,
338                     "No --command-config supplied; Dream/Writer/Create model stages cannot run",
339                     "Pass --command-config with a reviewed remote CLI profile."), None
340    try:
341        from howl_provider_core import CommandConfig, ProviderError
342    except ImportError:
343        return Check("provider.remote", "Remote provider configuration", FAIL,
344                     "howl-provider-core is not importable", "Install howl-provider-core."), None
345    try:
346        config = CommandConfig.read(path)
347    except (ProviderError, OSError, TypeError, ValueError) as error:
348        return Check("provider.remote", "Remote provider configuration", FAIL,
349                     f"{path} is invalid: {str(error)[:200]}",
350                     "Fix the profile: explicit argv, remote: true, no shell or local launcher."), None
351    executable = Path(config.argv[0]).name
352    if shutil.which(config.argv[0]) is None:
353        return Check("provider.remote", "Remote provider configuration", FAIL,
354                     f"{path} parses, but executable {executable} is not available",
355                     f"Install or put {executable} on PATH."), list(config.argv)
356    return Check("provider.remote", "Remote provider configuration", PASS,
357                 f"{path} parses; remote command {executable}, adapter {config.adapter or config.output_format}"), \
358        list(config.argv)

361def check_workspace(path: Path) -> Check:
362    target = path
363    while not target.exists() and target != target.parent:
364        target = target.parent
365    if os.access(target, os.W_OK | os.X_OK):
366        return Check("workspace", "Workspace permissions", PASS, f"{path} is writable")
367    return Check("workspace", "Workspace permissions", FAIL, f"{path} is not writable",
368                 "Choose a writable --workspace directory.")

371def run_checks(
372    *,
373    env: Optional[Mapping[str, str]] = None,
374    command_config: Optional[Path] = None,
375    workspace: Optional[Path] = None,
376    repos_root: Optional[Path] = None,
377    lookup: Callable[[str], Any] = metadata.distribution,
378    importer: Callable[[str], Any] = importlib.import_module,
379    which: Callable[[str], Optional[str]] = shutil.which,
380    interpreter: Optional[tuple] = None,
381) -> List[Check]:
382    env = os.environ if env is None else env
383    python, version, prefix, base = interpreter or (
384        sys.executable, tuple(sys.version_info[:3]), sys.prefix, sys.base_prefix)
385    checks = check_interpreter(python, version, prefix, base)
386    checks += check_components(lookup, importer, repos_root)
387    checks += check_entrypoints(which)
388    config_check, argv = check_command_config(command_config)
389    checks.append(config_check)
390    checks.append(check_policy(env))
391    checks.append(check_local_providers(env, argv))
392    checks.append(check_workspace(workspace or Path.cwd()))
393    return checks

396def summarize(checks: Iterable[Check]) -> Dict[str, Any]:
397    rows = [asdict(c) for c in checks]
398    counts = {s: sum(1 for r in rows if r["status"] == s) for s in (PASS, WARN, FAIL)}
399    overall = FAIL if counts[FAIL] else WARN if counts[WARN] else PASS
400    return {"schema": SCHEMA, "status": overall, "counts": counts, "checks": rows}

403def render(report: Dict[str, Any]) -> str:
404    lines = [f"HowlPlane creative preflight: {report['status']}",
405             f"{report['counts'][PASS]} PASS, {report['counts'][WARN]} WARN, {report['counts'][FAIL]} FAIL", ""]
406    for row in report["checks"]:
407        lines.append(f"{row['status']:<4}  {row['label']}: {row['detail']}")
408        if row["status"] != PASS and row["fix"]:
409            lines.append(f"      Suggested action: {row['fix']}")
410    return "\n".join(lines)

413def command(args: argparse.Namespace) -> int:
414    report = summarize(run_checks(
415        command_config=Path(args.command_config).expanduser() if getattr(args, "command_config", None) else None,
416        workspace=Path(args.workspace).expanduser() if getattr(args, "workspace", None) else None,
417        repos_root=Path(args.repos_root).expanduser() if getattr(args, "repos_root", None) else None,
418    ))
419    print(json.dumps(report, indent=2) if getattr(args, "json", False) else render(report))
420    return 1 if report["status"] == FAIL else 0
