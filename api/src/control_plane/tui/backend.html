src.control_plane.tui.backend

The only place howlplane tui touches HowlPlane (ADR 0009).

Backend is the seam between the terminal host and HowlPlane, so tests can substitute a fake that records calls and never reaches a provider. HowlPlaneBackend adds no configuration, routing or authority of its own:

  • inventory and route preview use the shared ProviderPoolManager in process, read only, with readiness probed (TUI-00 finding HP-F2);
  • a run is howlplane work OBJECTIVE --execute --json in a child process with the operator's own environment: the same governed path as the CLI;
  • cancel interrupts that child, then records the cancellation through the existing HumanLifecycleManager.cancel path.

 60class HowlPlaneBackend:
 61    """Delegates every action to existing HowlPlane machinery."""
 62
 63    def __init__(self, target_repo: Path, private_dir: Path):
 64        self.target_repo = Path(target_repo)
 65        self.private_dir = Path(private_dir)
 66
 67    def _pool(self):
 68        from howlplane.control_plane.synthesis.provider_pool import ProviderPoolManager
 69        return ProviderPoolManager.from_config(read_only=True, probe_on_start=True)
 70
 71    def inventory(self) -> Dict[str, Any]:
 72        from howlplane.control_plane.resource_cli import inventory_document
 73        return inventory_document(self._pool())
 74
 75    def route(self, objective: str, resource_id: Optional[str]) -> Dict[str, Any]:
 76        from howlplane.control_plane.cli import create_task_plan
 77        from howlplane.control_plane.project_adapter import ProjectAdapter
 78        pool = self._pool()
 79        args = argparse.Namespace(objective=objective, task_id=None, risk=None, tier=None, agent=resource_id,
 80                                  task_class=None, criteria=None, constraints=None, actions=None, skills=None,
 81                                  preferred_agent=None, repo=str(self.target_repo))
 82        spec, _ = create_task_plan(ProjectAdapter.discover(self.target_repo), self.target_repo, None, args,
 83                                   resource_pool=pool)
 84        return pool.select_resource(spec, role="implementation", explicit_resource_id=resource_id).to_dict()
 85
 86    def start_run(self, objective: str, resource_id: Optional[str], task_id: str) -> RunHandle:
 87        argv = [sys.executable, "-m", "howlplane.control_plane", "work", objective, "--execute", "--json",
 88                "--progress", "always", "--task-id", task_id, "--repo", str(self.target_repo)]
 89        if resource_id:
 90            argv += ["--agent", resource_id]
 91        self.private_dir.mkdir(parents=True, exist_ok=True, mode=0o700)
 92        fd, name = tempfile.mkstemp(prefix=f"run-{task_id}-", suffix=".stderr", dir=self.private_dir)
 93        with os.fdopen(fd, "w", encoding="utf-8") as stderr:
 94            # The operator's own environment: the same authority as typing the command.
 95            process = subprocess.Popen(  # nosec B603 - fixed argv, objective is one verbatim argument
 96                argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=stderr, text=True,
 97                start_new_session=True)
 98        return RunHandle(task_id=task_id, process=process, stderr_path=Path(name))
 99
100    def progress(self, handle: RunHandle) -> Optional[Dict[str, Any]]:
101        path = self.target_repo / ".task_runs" / handle.task_id / "progress.json"
102        try:
103            document = safe_load_json(path) if path.is_file() else None
104        except Exception:
105            return None
106        return document if isinstance(document, dict) else None
107
108    def result(self, handle: RunHandle) -> Dict[str, Any]:
109        stdout = handle.process.stdout.read() if handle.process and handle.process.stdout else ""
110        code = handle.process.wait() if handle.process else None
111        try:
112            document = json.loads(stdout) if stdout.strip() else None
113        except ValueError:
114            document = None
115        if isinstance(document, dict) and document.get("schema") == "howlplane.work_result/v1":
116            return {"event": "work_result", "document": document}
117        error = _last_error_document(handle.stderr_path)
118        if error:
119            return {"event": "failure", "document": error}
120        return {"event": "failure", "document": {
121            "schema": "howlplane.error/v1", "code": "RUN_OUTPUT_UNREADABLE",
122            "message": f"The governed run exited {code} without a work result.",
123            "why": "howlplane work did not print a howlplane.work_result/v1 document.",
124            "next_action": f"Inspect {handle.stderr_path} and the task's run directory.",
125            "command": f"howlplane status --repo {self.target_repo}"}}
126
127    def cancel(self, handle: RunHandle, grace_seconds: float) -> Dict[str, Any]:
128        handle.cancel_requested = True
129        process = handle.process
130        if process and process.poll() is None:
131            for sig in (signal.SIGINT, signal.SIGTERM):
132                try:
133                    os.killpg(process.pid, sig)
134                except ProcessLookupError:
135                    break
136                try:
137                    process.wait(timeout=grace_seconds)
138                    break
139                except subprocess.TimeoutExpired:
140                    continue
141            if process.poll() is None:
142                os.killpg(process.pid, signal.SIGKILL)
143                process.wait()
144        from howlplane.control_plane.human_boundary import HumanLifecycleManager
145        try:
146            outcome = HumanLifecycleManager.cancel(target_repo=str(self.target_repo), task_id=handle.task_id,
147                                                   reason="cancelled from howlplane tui", ledger=None)
148            recorded = f"Recorded as cancelled (exit {outcome.exit_code}); code changes are kept in the working tree."
149        except Exception as exc:  # the stop already happened; say plainly what did not
150            recorded = f"The cancellation could not be recorded: {exc}"
151        return {"schema": "howlplane.error/v1", "code": "RUN_CANCELLED",
152                "message": f"Run {handle.task_id} was stopped at your request.",
153                "why": recorded,
154                "next_action": "Review the working tree; resume or start a new objective.",
155                "command": f"howlplane status --repo {self.target_repo}",
156                "retry_safe": "RECONCILE_FIRST"}

159def _last_error_document(path: Optional[Path]) -> Optional[Dict[str, Any]]:
160    if not path or not path.is_file():
161        return None
162    found = None
163    with path.open(encoding="utf-8", errors="replace") as handle:
164        for line in handle:
165            line = line.strip()
166            if line.startswith("{") and '"howlplane.error/v1"' in line:
167                try:
168                    value = json.loads(line)
169                except ValueError:
170                    continue
171                if isinstance(value, dict):
172                    found = value
173    return found

Base class for protocol classes.

Protocol classes are defined as::

class Proto(Protocol):
    def meth(self) -> int:
        ...

Such classes are primarily used with static type checkers that recognize structural subtyping (static duck-typing).

For example::

class C:
    def meth(self) -> int:
        return 0

def func(x: Proto) -> int:
    return x.meth()

func(C())  # Passes static type check

See PEP 544 for details. Protocol classes decorated with @typing.runtime_checkable act as simple-minded runtime protocols that check only the presence of given attributes, ignoring their type signatures. Protocol classes can be generic, they are defined as::

class GenProto[T](Protocol):
    def meth(self) -> T:
        ...

1771def _no_init_or_replace_init(self, *args, **kwargs):
1772    cls = type(self)
1773
1774    if cls._is_protocol:
1775        raise TypeError('Protocols cannot be instantiated')
1776
1777    # Already using a custom `__init__`. No need to calculate correct
1778    # `__init__` to call. This can lead to RecursionError. See bpo-45121.
1779    if cls.__init__ is not _no_init_or_replace_init:
1780        return
1781
1782    # Initially, `__init__` of a protocol subclass is set to `_no_init_or_replace_init`.
1783    # The first instantiation of the subclass will call `_no_init_or_replace_init` which
1784    # searches for a proper new `__init__` in the MRO. The new `__init__`
1785    # replaces the subclass' old `__init__` (ie `_no_init_or_replace_init`). Subsequent
1786    # instantiation of the protocol subclass will thus use the new
1787    # `__init__` and no longer call `_no_init_or_replace_init`.
1788    for base in cls.__mro__:
1789        init = base.__dict__.get('__init__', _no_init_or_replace_init)
1790        if init is not _no_init_or_replace_init:
1791            cls.__init__ = init
1792            break
1793    else:
1794        # should not happen
1795        cls.__init__ = object.__init__
