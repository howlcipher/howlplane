recovery.py

Deterministic crash recovery and safe-retry classification engine for HowlPlane. Classifies recoverable lifecycle stages, reconciles uncommitted repository deltas, preserves completed reviewer results, invalidates stale reviews on repository drift, and prevents double-execution of consequential bounded actions.

 75    @staticmethod
 76    def _reviewer_disposition(reviews_dir: Path, role: str) -> str:
 77        """Classifies what a reviewer's durable evidence actually says.
 78
 79        Returns one of: completed_clean, completed_with_findings, failed,
 80        invalid, running, pending. File presence alone is not a verdict -- a
 81        zero-byte transcript beside an empty findings list is what a provider
 82        that never answered leaves behind, and it must not read as a passing
 83        review (HOWLFRAM-SLOPFIX-06).
 84        """
 85        result_file = reviews_dir / role / "result.json"
 86        if result_file.is_file():
 87            try:
 88                data = safe_load_json(result_file)
 89                status = (data or {}).get("status")
 90            except Exception:
 91                status = None
 92            if status == "clean":
 93                return "completed_clean"
 94            if status == "findings_detected":
 95                return "completed_with_findings"
 96            if status == "output_invalid":
 97                return "invalid"
 98            if status in ("reviewer_failure", "malformed_output"):
 99                return "failed"
100            if status:
101                return "pending"
102
103        md_file = reviews_dir / f"{role}.md"
104        yaml_file = reviews_dir / f"{role}_findings.yaml"
105        if not (md_file.is_file() and yaml_file.is_file()):
106            return "running" if md_file.is_file() else "pending"
107
108        try:
109            transcript = md_file.read_text(encoding="utf-8")
110        except Exception:
111            return "invalid"
112        if not transcript.strip():
113            return "invalid"
114
115        try:
116            parsed = yaml.safe_load(yaml_file.read_text(encoding="utf-8"))
117        except Exception:
118            return "failed"
119        if isinstance(parsed, dict):
120            parsed = parsed.get("findings", [])
121        if isinstance(parsed, list) and parsed:
122            return "completed_with_findings"
123        return "completed_clean"

125    @classmethod
126    def inspect_task(
127        cls,
128        target_repo: Union[str, Path],
129        task_id: str,
130    ) -> Dict[str, Any]:
131        """
132        Inspects all durable state for a task run to determine recovery classification and actions.
133        """
134        repo_root = Path(target_repo).resolve()
135        run_dir = repo_root / ".task_runs" / task_id
136
137        if not run_dir.is_dir() or not (run_dir / "task.yaml").is_file():
138            return {
139                "task_id": task_id,
140                "exists": False,
141                "error": f"Task run directory not found: {run_dir}",
142                "classification": RetryClassification.HUMAN_DECISION_REQUIRED,
143            }
144
145        try:
146            task_spec = TaskSpec.load_from_file(run_dir / "task.yaml")
147        except Exception as exc:
148            return {
149                "task_id": task_id,
150                "exists": True,
151                "corrupt": True,
152                "error": f"Corrupt task specification: {exc}",
153                "classification": RetryClassification.HUMAN_DECISION_REQUIRED,
154            }
155
156        latest_checkpoint = CheckpointManager.load_latest_checkpoint(run_dir)
157        last_stage = latest_checkpoint.stage if latest_checkpoint else task_spec.current_state
158
159        active_process = ProcessTracker.get_active_process(run_dir)
160        is_process_running = (
161            active_process is not None and active_process.status == "running"
162        )
163
164        # Inspect lock status
165        repo_lock_path = get_repo_lock_path(repo_root)
166        task_lock_path = get_task_lock_path(repo_root, task_id)
167        repo_locked = False
168        task_locked = False
169        lock_owner_pid = None
170
171        repo_lock_state: Optional[str] = None
172        repo_lock_reason: Optional[str] = None
173        repo_lock_task_id: Optional[str] = None
174        if repo_lock_path.exists():
175            try:
176                l_data = safe_load_json(repo_lock_path)
177                # Classified tri-state like the task lock: a recommendation is
178                # only useful if it names a command that can actually work, and
179                # the repository lock blocks recovery just as effectively.
180                state, reason = classify_lock_owner(
181                    l_data.get("pid", 0),
182                    l_data.get("hostname", ""),
183                    l_data.get("process_create_time", 0.0),
184                )
185                repo_lock_state = state.value
186                repo_lock_reason = reason
187                repo_lock_task_id = l_data.get("task_id")
188                if state is not LockOwnerState.STALE:
189                    repo_locked = True
190                    lock_owner_pid = l_data.get("pid")
191            except Exception:
192                pass
193
194        task_lock_state: Optional[str] = None
195        task_lock_reason: Optional[str] = None
196        if task_lock_path.exists():
197            try:
198                t_data = safe_load_json(task_lock_path)
199                state, reason = classify_lock_owner(
200                    t_data.get("pid", 0),
201                    t_data.get("hostname", ""),
202                    t_data.get("process_create_time", 0.0),
203                )
204                task_lock_state = state.value
205                task_lock_reason = reason
206                # A provably dead owner is reclaimed automatically on the next
207                # acquire, so it does not block anything and is not reported as
208                # a held lock.
209                if state is not LockOwnerState.STALE:
210                    task_locked = True
211                    lock_owner_pid = lock_owner_pid or t_data.get("pid")
212            except Exception:
213                pass
214
215        # Inspect completed vs incomplete reviewers
216        completed_reviewers: List[str] = []
217        incomplete_reviewers: List[str] = []
218        route_file = run_dir / "route.json"
219        recommended_reviewers: List[str] = []
220
221        if route_file.is_file():
222            try:
223                r_data = safe_load_json(route_file)
224                recommended_reviewers = r_data.get("recommended_reviewers", [])
225            except Exception:
226                pass
227
228        reviews_dir = run_dir / "reviews"
229        reviewer_dispositions: Dict[str, str] = {}
230        if reviews_dir.is_dir() and recommended_reviewers:
231            for role in recommended_reviewers:
232                disposition = cls._reviewer_disposition(reviews_dir, role)
233                reviewer_dispositions[role] = disposition
234                # Only a review that actually produced a usable verdict counts
235                # as completed. Reporting a dead reviewer's empty transcript as
236                # "Completed Reviews" is how SLOPFIX-06 looked further along
237                # than it was.
238                if disposition in ("completed_clean", "completed_with_findings"):
239                    completed_reviewers.append(role)
240                else:
241                    incomplete_reviewers.append(role)
242
243        # Inspect execution receipt
244        receipt_file = run_dir / "execution_receipt.json"
245        has_local_receipt = receipt_file.is_file()
246        native_receipt_found = False
247
248        if not has_local_receipt and (
249            task_spec.current_state == "awaiting_human"
250            or last_stage == "bounded_execution"
251        ):
252            # Check if HowlChangeOps native receipt exists on disk
253            dec_file = run_dir / "human_decision.json"
254            if dec_file.is_file():
255                try:
256                    dec_data = safe_load_json(dec_file)
257                    dec_id = dec_data.get("changeops_decision_id")
258                    if dec_id:
259                        hco_receipt = (
260                            repo_root
261                            / ".howlchangeops"
262                            / "receipts"
263                            / f"{dec_id}.json"
264                        )
265                        if hco_receipt.is_file():
266                            native_receipt_found = True
267                except Exception:
268                    pass
269
270        # Compute classification
271        if is_process_running:
272            classification = RetryClassification.HUMAN_DECISION_REQUIRED
273            recommendation = (
274                f"Task is actively running (PID {active_process.pid}). Run 'howlplane cancel {task_id}' to cancel."
275            )
276        elif task_spec.current_state == "complete":
277            classification = RetryClassification.ALREADY_COMPLETE
278            recommendation = "Task is already COMPLETE. No recovery needed."
279        elif task_spec.current_state == "cancelled":
280            classification = RetryClassification.HUMAN_DECISION_REQUIRED
281            recommendation = "Task was CANCELLED. Re-run or create a new task."
282        elif native_receipt_found:
283            classification = RetryClassification.NEVER_AUTO_RETRY
284            recommendation = (
285                f"HowlChangeOps action already executed natively. Run 'howlplane resume {task_id}' to reconcile receipt."
286            )
287        elif task_locked and task_lock_state == LockOwnerState.ACTIVE.value:
288            # Recommending resume while a live owner holds the lock sends the
289            # operator at a door that is bolted (HOWLFRAM-SLOPFIX-05).
290            classification = RetryClassification.HUMAN_DECISION_REQUIRED
291            recommendation = (
292                f"Task lock is held by active process PID {lock_owner_pid}. "
293                f"Wait for it to finish, or run 'howlplane cancel {task_id}'."
294            )
295        elif task_locked and task_lock_state == LockOwnerState.AMBIGUOUS.value:
296            classification = RetryClassification.HUMAN_DECISION_REQUIRED
297            recommendation = (
298                f"Task lock ownership is indeterminate ({task_lock_reason}). "
299                f"If that run is definitely gone, run 'howlplane unlock {task_id}' "
300                f"first, then 'howlplane resume {task_id}'."
301            )
302        elif repo_locked and repo_lock_state == LockOwnerState.ACTIVE.value:
303            classification = RetryClassification.HUMAN_DECISION_REQUIRED
304            recommendation = (
305                f"Repository lock is held by active process PID "
306                f"{lock_owner_pid} for task '{repo_lock_task_id}'. Wait for it "
307                f"to finish before resuming."
308            )
309        elif repo_locked and repo_lock_state == LockOwnerState.AMBIGUOUS.value:
310            classification = RetryClassification.HUMAN_DECISION_REQUIRED
311            recommendation = (
312                f"Repository lock ownership is indeterminate "
313                f"({repo_lock_reason}). If that run is definitely gone, run "
314                f"'howlplane unlock {repo_lock_task_id or task_id}' first, then "
315                f"'howlplane resume {task_id}'."
316            )
317        else:
318            classification = classify_stage_retry(last_stage)
319            recommendation = f"Run 'howlplane resume {task_id}' to recover from {last_stage}."
320
321        return {
322            "task_id": task_id,
323            "exists": True,
324            "corrupt": False,
325            "current_state": task_spec.current_state,
326            "last_stage": last_stage,
327            "classification": classification,
328            "is_process_running": is_process_running,
329            "process_info": active_process.to_dict() if active_process else None,
330            "repo_locked": repo_locked,
331            "task_locked": task_locked,
332            "task_lock_state": task_lock_state,
333            "task_lock_reason": task_lock_reason,
334            "lock_owner_pid": lock_owner_pid,
335            "repo_lock_state": repo_lock_state,
336            "completed_reviewers": completed_reviewers,
337            "reviewer_dispositions": reviewer_dispositions,
338            "incomplete_reviewers": incomplete_reviewers,
339            "has_local_receipt": has_local_receipt,
340            "native_receipt_found": native_receipt_found,
341            "recommendation": recommendation,
342        }

344    @classmethod
345    def reconcile_interrupted_implementation(
346        cls,
347        target_repo: Union[str, Path],
348        run_dir: Union[str, Path],
349        task_spec: TaskSpec,
350    ) -> Tuple[bool, Optional[RepositoryDelta], str]:
351        """
352        Reconciles working tree state after an interrupted implementation stage.
353        If actual repository delta exists, records it and avoids duplicate reruns.
354        Returns: (has_valid_delta, delta_obj, summary_reason).
355        """
356        root = Path(target_repo).resolve()
357        r_dir = Path(run_dir).resolve()
358        baseline_file = r_dir / "baseline.json"
359
360        if baseline_file.is_file():
361            try:
362                b_data = safe_load_json(baseline_file)
363                baseline = GitBaseline.from_dict(b_data)
364            except Exception:
365                baseline = capture_baseline(root)
366        else:
367            baseline = capture_baseline(root)
368            atomic_write_json(baseline_file, baseline.to_dict())
369
370        current_delta = capture_delta(root, baseline)
371
372        if current_delta.is_empty:
373            return (
374                False,
375                None,
376                "No working tree modifications found from previous run. Safe to execute implementation.",
377            )
378
379        # Delta exists: persist diff artifacts to avoid rerun
380        impl_dir = r_dir / "implementation"
381        impl_dir.mkdir(parents=True, exist_ok=True)
382        (impl_dir / "diff.patch").write_text(
383            current_delta.diff_content, encoding="utf-8"
384        )
385        (r_dir / "diff.patch").write_text(
386            current_delta.diff_content, encoding="utf-8"
387        )
388
389        return (
390            True,
391            current_delta,
392            f"Recovered implementation delta: {len(current_delta.files_modified)} modified, {len(current_delta.files_added)} added files.",
393        )

395    @classmethod
396    def check_review_validity(
397        cls,
398        target_repo: Union[str, Path],
399        run_dir: Union[str, Path],
400    ) -> Tuple[bool, Optional[str]]:
401        """
402        Verifies that current repository state matches the delta that was reviewed.
403        Returns (is_valid, invalid_reason).
404        """
405        root = Path(target_repo).resolve()
406        r_dir = Path(run_dir).resolve()
407        baseline_file = r_dir / "baseline.json"
408        diff_file = r_dir / "diff.patch"
409
410        if not baseline_file.is_file() or not diff_file.is_file():
411            return False, "Missing baseline or diff artifacts for review validation"
412
413        try:
414            baseline = GitBaseline.from_dict(safe_load_json(baseline_file))
415            current_delta = capture_delta(root, baseline)
416            recorded_diff = diff_file.read_text(encoding="utf-8")
417            if current_delta.diff_content != recorded_diff:
418                return (
419                    False,
420                    "Working tree modifications have drifted since review was conducted",
421                )
422            return True, None
423        except Exception as e:
424            return False, f"Failed to validate review repository state: {e}"
