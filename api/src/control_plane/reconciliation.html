 61    def validate(self) -> None:
 62        if not self.id or not isinstance(self.id, str):
 63            raise ReconciliationValidationError("Finding id must be a non-empty string.")
 64        if self.severity not in VALID_SEVERITIES:
 65            raise ReconciliationValidationError(
 66                f"Finding severity '{self.severity}' invalid. Allowed: {sorted(VALID_SEVERITIES)}"
 67            )
 68        if self.status not in VALID_FINDING_STATUSES:
 69            raise ReconciliationValidationError(
 70                f"Finding status '{self.status}' invalid. Allowed: {sorted(VALID_FINDING_STATUSES)}"
 71            )
 72        if self.redundancy_status not in VALID_REDUNDANCY_STATUSES:
 73            raise ReconciliationValidationError(
 74                f"redundancy_status '{self.redundancy_status}' invalid. Allowed: {sorted(VALID_REDUNDANCY_STATUSES)}"
 75            )
 76
 77        # Rule: Any dismissed blocker or high finding must have a resolution reason
 78        if self.status in ("false_positive", "out_of_scope") and self.severity in ("blocker", "high"):
 79            if not self.resolution_reason or not self.resolution_reason.strip():
 80                raise ReconciliationValidationError(
 81                    f"Finding '{self.id}' with severity '{self.severity}' cannot be marked '{self.status}' "
 82                    "without an explicit non-empty resolution_reason."
 83                )

203    @staticmethod
204    def reconcile(findings: List[ReviewFinding]) -> ReconciliationResult:
205        """
206        Reconciles a collection of review findings into classified categories.
207        Ensures all validations pass and preserves every disagreement.
208        """
209        # Validate all findings first
210        for f in findings:
211            f.validate()
212
213        confirmed: List[ReviewFinding] = []
214        likely: List[ReviewFinding] = []
215        disputed: List[ReviewFinding] = []
216        false_positives: List[ReviewFinding] = []
217        out_of_scope: List[ReviewFinding] = []
218        requires_human: List[ReviewFinding] = []
219
220        unresolved_blockers = 0
221        unresolved_highs = 0
222        duplicate_count = 0
223        overlapping_count = 0
224        findings_by_role: Dict[str, int] = {}
225        unique_findings_by_role: Dict[str, int] = {}
226
227        # Index findings by location and component
228        location_map: Dict[str, List[ReviewFinding]] = {}
229        for f in findings:
230            findings_by_role[f.reviewer_role] = findings_by_role.get(f.reviewer_role, 0) + 1
231            if f.location:
232                loc_key = f.location.strip().lower()
233                location_map.setdefault(loc_key, []).append(f)
234
235        # Detect cross-reviewer duplicates and overlaps
236        processed_pairs = set()
237        for f in findings:
238            if f.location:
239                loc_key = f.location.strip().lower()
240                co_located = location_map.get(loc_key, [])
241                distinct_reviewers = {other.reviewer_role for other in co_located if other.id != f.id}
242
243                if distinct_reviewers:
244                    # Match found across different reviewer roles
245                    other_findings = [other for other in co_located if other.id != f.id]
246                    for other in other_findings:
247                        pair = tuple(sorted([f.id, other.id]))
248                        if pair not in processed_pairs:
249                            processed_pairs.add(pair)
250                            if other.id not in f.related_finding_ids:
251                                f.related_finding_ids.append(other.id)
252                            if f.id not in other.related_finding_ids:
253                                other.related_finding_ids.append(f.id)
254
255                    # Determine redundancy classification
256                    if f.redundancy_status == "independent":
257                        # If first finding for this location, mark overlapping; if subsequent, duplicate
258                        first_for_loc = co_located[0].id == f.id
259                        if first_for_loc:
260                            f.redundancy_status = "overlapping"
261                            overlapping_count += 1
262                        else:
263                            f.redundancy_status = "duplicate"
264                            duplicate_count += 1
265                else:
266                    if f.redundancy_status == "duplicate":
267                        duplicate_count += 1
268                    elif f.redundancy_status == "overlapping":
269                        overlapping_count += 1
270            else:
271                if f.redundancy_status == "duplicate":
272                    duplicate_count += 1
273                elif f.redundancy_status == "overlapping":
274                    overlapping_count += 1
275
276            if f.redundancy_status == "independent":
277                unique_findings_by_role[f.reviewer_role] = unique_findings_by_role.get(f.reviewer_role, 0) + 1
278
279        for f in findings:
280            # Check if status was already set or should be deduced
281            if f.status == "open":
282                # Auto-classification logic
283                loc_matches = location_map.get(f.location.strip().lower(), []) if f.location else []
284                reviewers_at_loc = {m.reviewer_role for m in loc_matches}
285
286                if len(reviewers_at_loc) > 1:
287                    # Multiple distinct reviewers flagged this location
288                    f.status = "confirmed"
289                elif f.severity in ("blocker", "high") and f.category == "security":
290                    f.status = "requires_human_judgment"
291                    f.resolution_reason = "Security-critical finding automatically routed for human verification."
292                elif f.severity in ("blocker", "high"):
293                    f.status = "likely"
294                else:
295                    f.status = "likely"
296
297            # Sort into buckets
298            if f.status == "confirmed":
299                confirmed.append(f)
300            elif f.status == "likely":
301                likely.append(f)
302            elif f.status == "disputed":
303                disputed.append(f)
304            elif f.status == "false_positive":
305                false_positives.append(f)
306            elif f.status == "out_of_scope":
307                out_of_scope.append(f)
308            elif f.status == "requires_human_judgment":
309                requires_human.append(f)
310
311            # Count unresolved severe items
312            if f.status in ("confirmed", "likely", "disputed", "requires_human_judgment"):
313                if f.severity == "blocker":
314                    unresolved_blockers += 1
315                elif f.severity == "high":
316                    unresolved_highs += 1
317
318        return ReconciliationResult(
319            findings=findings,
320            confirmed=confirmed,
321            likely=likely,
322            disputed=disputed,
323            false_positives=false_positives,
324            out_of_scope=out_of_scope,
325            requires_human_judgment=requires_human,
326            unresolved_blockers=unresolved_blockers,
327            unresolved_highs=unresolved_highs,
328            duplicate_count=duplicate_count,
329            overlapping_count=overlapping_count,
330            unique_findings_by_role=unique_findings_by_role,
331            findings_by_role=findings_by_role,
332        )

205    def reconcile(findings: List[ReviewFinding]) -> ReconciliationResult:
206        """
207        Reconciles a collection of review findings into classified categories.
208        Ensures all validations pass and preserves every disagreement.
209        """
210        # Validate all findings first
211        for f in findings:
212            f.validate()
213
214        confirmed: List[ReviewFinding] = []
215        likely: List[ReviewFinding] = []
216        disputed: List[ReviewFinding] = []
217        false_positives: List[ReviewFinding] = []
218        out_of_scope: List[ReviewFinding] = []
219        requires_human: List[ReviewFinding] = []
220
221        unresolved_blockers = 0
222        unresolved_highs = 0
223        duplicate_count = 0
224        overlapping_count = 0
225        findings_by_role: Dict[str, int] = {}
226        unique_findings_by_role: Dict[str, int] = {}
227
228        # Index findings by location and component
229        location_map: Dict[str, List[ReviewFinding]] = {}
230        for f in findings:
231            findings_by_role[f.reviewer_role] = findings_by_role.get(f.reviewer_role, 0) + 1
232            if f.location:
233                loc_key = f.location.strip().lower()
234                location_map.setdefault(loc_key, []).append(f)
235
236        # Detect cross-reviewer duplicates and overlaps
237        processed_pairs = set()
238        for f in findings:
239            if f.location:
240                loc_key = f.location.strip().lower()
241                co_located = location_map.get(loc_key, [])
242                distinct_reviewers = {other.reviewer_role for other in co_located if other.id != f.id}
243
244                if distinct_reviewers:
245                    # Match found across different reviewer roles
246                    other_findings = [other for other in co_located if other.id != f.id]
247                    for other in other_findings:
248                        pair = tuple(sorted([f.id, other.id]))
249                        if pair not in processed_pairs:
250                            processed_pairs.add(pair)
251                            if other.id not in f.related_finding_ids:
252                                f.related_finding_ids.append(other.id)
253                            if f.id not in other.related_finding_ids:
254                                other.related_finding_ids.append(f.id)
255
256                    # Determine redundancy classification
257                    if f.redundancy_status == "independent":
258                        # If first finding for this location, mark overlapping; if subsequent, duplicate
259                        first_for_loc = co_located[0].id == f.id
260                        if first_for_loc:
261                            f.redundancy_status = "overlapping"
262                            overlapping_count += 1
263                        else:
264                            f.redundancy_status = "duplicate"
265                            duplicate_count += 1
266                else:
267                    if f.redundancy_status == "duplicate":
268                        duplicate_count += 1
269                    elif f.redundancy_status == "overlapping":
270                        overlapping_count += 1
271            else:
272                if f.redundancy_status == "duplicate":
273                    duplicate_count += 1
274                elif f.redundancy_status == "overlapping":
275                    overlapping_count += 1
276
277            if f.redundancy_status == "independent":
278                unique_findings_by_role[f.reviewer_role] = unique_findings_by_role.get(f.reviewer_role, 0) + 1
279
280        for f in findings:
281            # Check if status was already set or should be deduced
282            if f.status == "open":
283                # Auto-classification logic
284                loc_matches = location_map.get(f.location.strip().lower(), []) if f.location else []
285                reviewers_at_loc = {m.reviewer_role for m in loc_matches}
286
287                if len(reviewers_at_loc) > 1:
288                    # Multiple distinct reviewers flagged this location
289                    f.status = "confirmed"
290                elif f.severity in ("blocker", "high") and f.category == "security":
291                    f.status = "requires_human_judgment"
292                    f.resolution_reason = "Security-critical finding automatically routed for human verification."
293                elif f.severity in ("blocker", "high"):
294                    f.status = "likely"
295                else:
296                    f.status = "likely"
297
298            # Sort into buckets
299            if f.status == "confirmed":
300                confirmed.append(f)
301            elif f.status == "likely":
302                likely.append(f)
303            elif f.status == "disputed":
304                disputed.append(f)
305            elif f.status == "false_positive":
306                false_positives.append(f)
307            elif f.status == "out_of_scope":
308                out_of_scope.append(f)
309            elif f.status == "requires_human_judgment":
310                requires_human.append(f)
311
312            # Count unresolved severe items
313            if f.status in ("confirmed", "likely", "disputed", "requires_human_judgment"):
314                if f.severity == "blocker":
315                    unresolved_blockers += 1
316                elif f.severity == "high":
317                    unresolved_highs += 1
318
319        return ReconciliationResult(
320            findings=findings,
321            confirmed=confirmed,
322            likely=likely,
323            disputed=disputed,
324            false_positives=false_positives,
325            out_of_scope=out_of_scope,
326            requires_human_judgment=requires_human,
327            unresolved_blockers=unresolved_blockers,
328            unresolved_highs=unresolved_highs,
329            duplicate_count=duplicate_count,
330            overlapping_count=overlapping_count,
331            unique_findings_by_role=unique_findings_by_role,
332            findings_by_role=findings_by_role,
