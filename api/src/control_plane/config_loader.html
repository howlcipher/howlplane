 14class _UniqueKeyLoader(yaml.SafeLoader):
 15    """YAML loader that rejects duplicate configuration keys."""

 18def _construct_unique_mapping(loader, node, deep=False):
 19    mapping = {}
 20    for key_node, value_node in node.value:
 21        key = loader.construct_object(key_node, deep=deep)
 22        if key in mapping:
 23            raise ValueError(f"Duplicate configuration key: {key}")
 24        mapping[key] = loader.construct_object(value_node, deep=deep)
 25    return mapping
 26

 29    yaml.resolver.BaseResolver.DEFAULT_MAPPING_TAG,
 30    _construct_unique_mapping,

 34def _deep_merge(base: dict, override: dict) -> dict:
 35    """Returns a recursive mapping merge without mutating either input."""
 36    merged = dict(base)
 37    for key, value in override.items():
 38        if isinstance(value, dict) and isinstance(merged.get(key), dict):
 39            merged[key] = _deep_merge(merged[key], value)
 40        else:
 41            merged[key] = value
 42    return merged

 45def _resource_local_config(local_data: dict) -> dict:
 46    """Extracts only supported application overrides from operator TOML."""
 47    source = local_data.get("ai_resources", local_data)
 48    supported = (
 49        "operating_mode",
 50        "providers",
 51        "provider_policy",
 52        "semantic_recommendation",
 53        "roles",
 54    )
 55    extracted = {key: source[key] for key in supported if key in source}
 56    if "roles" in local_data and "roles" not in extracted:
 57        extracted["roles"] = local_data["roles"]
 58    # One canonical top-level table, read by `workspace_trust.resolve_policy`.
 59    if "workspace_trust" in local_data:

 61    if "tui" in local_data:

 63    return extracted

309class ConfigLoader:
310    """
311    A class to load and provide access to configuration settings,
312    as well as common directory paths to apply DRY principles.
313    """
314
315    def __init__(self, config_path=None, local_config_path=None):
316        """
317        Initializes the ConfigLoader and calculates common paths.
318        """
319        self.script_dir = str(Path(__file__).resolve().parent)
320        self.repo_root = str(Path(__file__).resolve().parents[3])
321
322        if config_path is None:
323            self.config_path = os.path.join(self.repo_root, "config", "settings.yaml")
324        else:
325            self.config_path = config_path
326
327        yaml_data = {}
328        if os.path.exists(self.config_path):
329            with open(self.config_path, "r") as f:
330                loader = _UniqueKeyLoader(f)
331                try:
332                    yaml_data = loader.get_single_data() or {}
333                finally:
334                    loader.dispose()
335
336        # Kept so `howlplane config` can report which layer supplied a value.
337        self.file_layer = yaml_data
338        self.local_layer = {}
339
340        local_candidate = local_config_path
341        if local_candidate is None and config_path is None:
342            local_candidate = (
343                os.environ.get("HOWLPLANE_LOCAL_CONFIG")
344                or Path.home() / ".config" / "howlplane" / "config.toml"
345            )
346        self.local_config_path = (
347            Path(local_candidate).expanduser() if local_candidate else None
348        )
349        if self.local_config_path and self.local_config_path.is_file():
350            with open(self.local_config_path, "rb") as local_file:
351                local_data = tomllib.load(local_file)
352            self.local_layer = _resource_local_config(local_data)
353            yaml_data = _deep_merge(yaml_data, self.local_layer)
354
355        # Load pydantic settings prioritizing .env over yaml_data
356        self.settings = AppSettings(**yaml_data)
357
358        # Override with Secret Manager if configured
359        secret_mgr = SecretManager()
360        if os.environ.get("USE_AWS_SECRETS_MANAGER") == "true":
361            aws_api_key = secret_mgr.get_secret("GEMINI_API_KEY")
362            if aws_api_key:
363                self.settings.gemini_api_key = aws_api_key
364
365            aws_anthropic_key = secret_mgr.get_secret("ANTHROPIC_API_KEY")
366            if aws_anthropic_key:
367                self.settings.anthropic_api_key = aws_anthropic_key
368
369            aws_webhook = secret_mgr.get_secret("WEBHOOK_SECRET")
370            if aws_webhook:
371                self.settings.server.webhook_secret = aws_webhook
372
373        self.config = self.settings.model_dump()
374
375    def get(self, key, default=None):
376        """
377        Retrieves a value from the configuration using the provided key.
378        """
379        return self.config.get(key, default)

400def get_default_loader() -> "ConfigLoader":
401    """Build the shared loader on first use, not at import.
402
403    Constructing it validates the environment, so doing that at import killed
404    every command (including ``howlplane config validate``) with a raw pydantic
405    traceback before the CLI could explain it. ``default_loader`` and ``config``
406    stay importable for backward compatibility through ``__getattr__``; the
407    first access builds and caches them in module globals, so tests can still
408    replace them with ``monkeypatch.setattr``.
409    """
410    loader = globals().get("default_loader")
411    if loader is None:
412        loader = ConfigLoader()
413        globals()["default_loader"] = loader
414        globals()["config"] = loader.config
415    return loader
