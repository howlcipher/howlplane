git_integration.py

Real git/GitHub integration for governed engineering tasks (#59 Phases 3-6). Replaces the fabricated branch/commit/PR/CI/merge values previously hand-built in synthesis/marathon.py with actual git/gh subprocess calls, each independently verified before the corresponding GitIntegrationRecord flag is set.

Design principles carried over from the rest of the control plane:

  • Never shell=True (mirrors git_baseline.py's _run_git_cmd).
  • Never git add -A / git add . -- only the explicit path list from the orchestrator's already-computed RepositoryDelta is staged.
  • Every step is independently re-verified against observable reality (git ls-remote, gh pr view, git merge-base --is-ancestor) rather than trusting a prior command's exit code alone.
  • merged=True is only ever reachable after verify_remote_main_contains() returns True -- there is no other code path that sets it.

 149def check_gh_auth(
 150    repo_root: Union[str, Path] = ".",
 151    gh_runner: CommandRunner = run_gh,
 152) -> GhAuthStatus:
 153    """
 154    Verifies GitHub CLI authentication by actually invoking `gh auth status`
 155    -- never assumes credentials are usable merely because the `gh` binary
 156    is present on PATH (#59 Phase 3).
 157    """
 158    try:
 159        proc = gh_runner(repo_root, ["auth", "status"], 15)
 160    except (OSError, subprocess.SubprocessError) as exc:
 161        return GhAuthStatus(authenticated=False, detail=f"gh auth status failed to execute: {exc}")
 162
 163    combined = f"{proc.stdout}\n{proc.stderr}"
 164    if proc.returncode != 0 or "Logged in to" not in combined:
 165        return GhAuthStatus(authenticated=False, detail=combined.strip()[:500])
 166
 167    account_match = re.search(r"account\s+(\S+)", combined)
 168    scopes_match = re.search(r"Token scopes:\s*(.+)", combined)
 169    scopes = []
 170    if scopes_match:
 171        scopes = [s.strip().strip("'") for s in scopes_match.group(1).split(",")]
 172
 173    return GhAuthStatus(
 174        authenticated=True,
 175        account=account_match.group(1) if account_match else None,
 176        scopes=scopes,
 177        detail=combined.strip()[:500],
 178    )

 192def _remote_slug(remote_url: str) -> Optional[str]:
 193    """Extracts `owner/repo` from an https or ssh GitHub remote URL."""
 194    m = re.search(r"github\.com[:/](?P<slug>[^/]+/[^/]+?)(\.git)?/?$", remote_url.strip())
 195    return m.group("slug") if m else None

 198def detect_repo_slug(repo_root: Union[str, Path], git_runner: CommandRunner = run_git) -> Optional[str]:
 199    """Reads `owner/repo` from the `origin` remote, or None if undeterminable."""
 200    proc = git_runner(repo_root, ["remote", "get-url", "origin"], 15)
 201    if proc.returncode != 0 or not proc.stdout.strip():
 202        return None
 203    return _remote_slug(proc.stdout.strip())

 206def run_preflight(
 207    repo_root: Union[str, Path],
 208    expected_repo_slug: str,
 209    git_runner: CommandRunner = run_git,
 210    gh_runner: CommandRunner = run_gh,
 211) -> GitPreflightResult:
 212    """
 213    Real git/GitHub preflight (#59 Phase 3): confirms this is a git repo, the
 214    remote points at the expected repository, and GitHub credentials
 215    actually permit PR/CI/merge operations. Never fabricates capability --
 216    any failure produces an actionable `blocked_reason` (e.g.
 217    "BLOCKED_GITHUB_AUTH") instead of proceeding.
 218    """
 219    is_repo_proc = git_runner(repo_root, ["rev-parse", "--is-inside-work-tree"], 15)
 220    is_git_repo = is_repo_proc.returncode == 0 and is_repo_proc.stdout.strip() == "true"
 221    if not is_git_repo:
 222        return GitPreflightResult(
 223            is_git_repo=False, remote_ok=False, remote_url="", expected_repo_slug=expected_repo_slug,
 224            push_capable=False, gh_auth=GhAuthStatus(authenticated=False),
 225            blocked_reason="NOT_A_GIT_REPOSITORY",
 226        )
 227
 228    remote_proc = git_runner(repo_root, ["remote", "get-url", "origin"], 15)
 229    remote_url = remote_proc.stdout.strip() if remote_proc.returncode == 0 else ""
 230    remote_slug = _remote_slug(remote_url) if remote_url else None
 231    remote_ok = remote_slug is not None and remote_slug.lower() == expected_repo_slug.lower()
 232
 233    gh_auth = check_gh_auth(repo_root, gh_runner=gh_runner)
 234
 235    blocked_reason = None
 236    if not remote_ok:
 237        blocked_reason = "REMOTE_MISMATCH"
 238    elif not gh_auth.authenticated:
 239        blocked_reason = "BLOCKED_GITHUB_AUTH"
 240    elif "repo" not in gh_auth.scopes and gh_auth.scopes:
 241        # Some gh auth outputs omit scopes entirely (older gh versions); only
 242        # flag missing 'repo' scope when scopes were actually parsed.
 243        blocked_reason = "BLOCKED_GITHUB_AUTH"
 244
 245    return GitPreflightResult(
 246        is_git_repo=is_git_repo,
 247        remote_ok=remote_ok,
 248        remote_url=remote_url,
 249        expected_repo_slug=expected_repo_slug,
 250        push_capable=remote_ok and gh_auth.authenticated,
 251        gh_auth=gh_auth,
 252        blocked_reason=blocked_reason,
 253    )

 304    def authorizes_merge(self) -> bool:
 305        """Every gate an unattended merge depends on, in one place."""
 306        return (
 307            self.policy is not None
 308            and self.policy.authorizes_merge_gate()
 309            and self.all_required_observed
 310            and self.all_required_terminal
 311            and self.all_required_green
 312            and not self.failed_jobs
 313            and not self.pending_jobs
 314            and not self.timed_out
 315            and bool(self.pr_head_sha)
 316            and self.pr_head_sha == self.checks_head_sha

 350class GitHubCIObserver:
 351    """
 352    Observes real GitHub CI state for a PR, never assuming local test exit 0
 353    means GitHub CI is green (#59 Phase 5).
 354
 355    Required-check discovery reads live repository policy on every call, so a
 356    ruleset that gains or loses a required check is picked up with no code
 357    change (#59.1 Blocker 3). GitHub rulesets are consulted first -- they are
 358    what protects `main` on this repository, and the legacy branch-protection
 359    endpoint 404s under a ruleset -- with legacy branch protection as the
 360    fallback for repositories still using it. When neither can be read the
 361    policy is reported unavailable and the caller must fail closed; a
 362    hard-coded list must never authorize an unattended merge.
 363
 364    A deterministic test or fake mode may inject a known policy via
 365    `policy_override`; production unattended mode leaves it unset and requires
 366    observed policy.
 367    """
 368
 369    def __init__(
 370        self,
 371        gh_runner: CommandRunner = run_gh,
 372        git_runner: CommandRunner = run_git,
 373        policy_override: Optional[RequiredCheckPolicy] = None,
 374    ):
 375        self._gh = gh_runner
 376        self._git = git_runner
 377        self._policy_override = policy_override
 378
 379    def required_check_policy(
 380        self,
 381        repo_root: Union[str, Path],
 382        repo_slug: str,
 383        branch: str = "main",
 384    ) -> RequiredCheckPolicy:
 385        if self._policy_override is not None:
 386            return self._policy_override
 387
 388        contexts = self._contexts_from_rulesets(repo_root, repo_slug, branch)
 389        source = POLICY_SOURCE_RULESET
 390        if contexts is None:
 391            contexts = self._contexts_from_branch_protection(repo_root, repo_slug, branch)
 392            source = POLICY_SOURCE_BRANCH_PROTECTION
 393        if contexts is None:
 394            return RequiredCheckPolicy(
 395                available=False,
 396                source=POLICY_SOURCE_UNAVAILABLE,
 397                reason=CI_POLICY_UNAVAILABLE,
 398                detail=(
 399                    f"neither rulesets nor branch protection could be read for "
 400                    f"{repo_slug}@{branch}"
 401                ),
 402            )
 403        if not contexts:
 404            return RequiredCheckPolicy(
 405                available=True,
 406                contexts=[],
 407                source=source,
 408                reason=NO_REQUIRED_CHECKS_ENFORCED,
 409                detail=f"{source} readable but enforces no required status checks",
 410            )
 411        return RequiredCheckPolicy(available=True, contexts=contexts, source=source)

 413    def _gh_json(self, repo_root: Union[str, Path], endpoint: str) -> Optional[Any]:
 414        """
 415        Reads one `gh api` endpoint as JSON. Returns None when the endpoint
 416        could not be read or parsed at all -- which the caller must treat as
 417        "policy unknown", never as "policy empty".
 418        """
 419        proc = self._gh(repo_root, ["api", endpoint], 30)
 420        if proc.returncode != 0 or not proc.stdout.strip():
 421            return None
 422        try:
 423            return json.loads(proc.stdout)
 424        except json.JSONDecodeError:
 425            return None

 427    def _contexts_from_rulesets(
 428        self, repo_root: Union[str, Path], repo_slug: str, branch: str
 429    ) -> Optional[List[str]]:
 430        """
 431        Reads the rules GitHub itself reports as applying to `branch`, so
 432        repository- and organization-level rulesets are both covered without
 433        HowlPlane re-implementing ruleset applicability. Returns None when the
 434        endpoint could not be read at all (so the caller can try legacy
 435        protection), and [] when it was read but enforces no status checks.
 436
 437        Response shape confirmed against this repository's live API:
 438          [{"type": "required_status_checks",
 439            "parameters": {"required_status_checks": [{"context": "lint"}, ...]}}]
 440        """
 441        rules = self._gh_json(repo_root, f"repos/{repo_slug}/rules/branches/{branch}")
 442        if not isinstance(rules, list):
 443            return None
 444
 445        contexts: List[str] = []
 446        for rule in rules:
 447            if not isinstance(rule, dict) or rule.get("type") != "required_status_checks":
 448                continue
 449            params = rule.get("parameters") or {}
 450            for entry in params.get("required_status_checks") or []:
 451                context = entry.get("context") if isinstance(entry, dict) else entry
 452                if context and context not in contexts:
 453                    contexts.append(context)
 454        return contexts
 455
 456    def _contexts_from_branch_protection(
 457        self, repo_root: Union[str, Path], repo_slug: str, branch: str
 458    ) -> Optional[List[str]]:
 459        """Legacy branch protection, for repositories not using rulesets."""
 460        data = self._gh_json(repo_root, f"repos/{repo_slug}/branches/{branch}/protection")
 461        if not isinstance(data, dict):
 462            return None
 463        required = data.get("required_status_checks") or {}
 464        if not isinstance(required, dict):
 465            return []
 466        return [c for c in (required.get("contexts") or []) if c]

 468    def required_check_names(
 469        self, repo_root: Union[str, Path], repo_slug: str, branch: str = "main"
 470    ) -> List[str]:
 471        """Convenience wrapper; callers gating a merge must use the policy itself."""
 472        return list(self.required_check_policy(repo_root, repo_slug, branch).contexts)

 474    def _pr_head_sha(
 475        self, repo_root: Union[str, Path], pr_number: int
 476    ) -> Optional[str]:
 477        """Return the live PR head SHA, or None when GitHub cannot prove it."""
 478        proc = self._gh(repo_root, ["pr", "view", str(pr_number), "--json", "headRefOid"], 30)

 488        self,
 489        repo_root: Union[str, Path],
 490        pr_number: int,
 491        repo_slug: str,
 492        policy: Optional[RequiredCheckPolicy] = None,
 493    ) -> CIObservation:
 494        policy = policy or self.required_check_policy(repo_root, repo_slug)
 495        required = set(policy.contexts)
 496        # `gh pr checks` is a PR-head view rather than a historical commit
 497        # query. Capture the head both before and after that view. If it moves
 498        # in between, its results are deliberately unusable: a successful
 499        # check for the earlier head must never authorize the later one.
 500        head_before = self._pr_head_sha(repo_root, pr_number)
 501        proc = self._gh(
 502            repo_root,
 503            ["pr", "checks", str(pr_number), "--json", "name,state,bucket,link"],
 504            30,
 505        )
 506        checks: List[Dict[str, str]] = []
 507        if proc.returncode == 0 and proc.stdout.strip():
 508            try:
 509                checks = json.loads(proc.stdout)
 510            except json.JSONDecodeError:
 511                checks = []
 512
 513        head_after = self._pr_head_sha(repo_root, pr_number)
 514        stable_head = head_before if head_before and head_before == head_after else None
 515        observed_at = datetime.now(timezone.utc).isoformat()
 516        for check in checks:
 517            # The GitHub PR checks endpoint reports the current PR head. This
 518            # explicit durable annotation makes that binding auditable.
 519            check["head_sha"] = stable_head or ""
 520            check["observed_at"] = observed_at
 521
 522        observed_names = {c.get("name") for c in checks}
 523        all_required_observed = bool(stable_head) and bool(required) and required.issubset(observed_names)
 524
 525        required_checks = [c for c in checks if c.get("name") in required]
 526        failed_jobs = [c for c in required_checks if classify_check(c) == "failed"]
 527        pending_jobs = [c for c in required_checks if classify_check(c) == "pending"]
 528
 529        all_required_terminal = all_required_observed and not pending_jobs
 530        all_required_green = (
 531            all_required_observed
 532            and all_required_terminal
 533            and not failed_jobs
 534            and all(classify_check(c) == "green" for c in required_checks)
 535        )
 536
 537        return CIObservation(
 538            all_required_observed=all_required_observed,
 539            all_required_green=all_required_green,
 540            all_required_terminal=all_required_terminal,
 541            checks=checks,
 542            failed_jobs=failed_jobs,
 543            pending_jobs=pending_jobs,
 544            policy=policy,
 545            pr_head_sha=head_after,
 546            checks_head_sha=stable_head,
 547            observed_at=observed_at,
 548            raw={
 549                "required": sorted(required),
 550                "pr_number": pr_number,
 551                "policy_source": policy.source,
 552                "policy_reason": policy.reason,
 553                "pr_head_sha_before": head_before,
 554                "pr_head_sha_after": head_after,
 555                "checks_head_sha": stable_head,
 556                "observed_at": observed_at,
 557            },
 558        )

 560    def poll_until_terminal(
 561        self,
 562        repo_root: Union[str, Path],
 563        pr_number: int,
 564        repo_slug: str,
 565        timeout_seconds: int = 900,
 566        poll_interval: int = 30,
 567        sleep_fn: Callable[[float], None] = time.sleep,
 568        clock_fn: Callable[[], float] = time.monotonic,
 569    ) -> CIObservation:
 570        """
 571        Bounded, terminal-aware CI polling (#59 Phase 5, #59.1 Blocker 4).
 572
 573        Previously this returned as soon as every required check *name* had
 574        appeared -- but a QUEUED job already has a name, so polling could
 575        finish before CI had actually run. A required check must now be both
 576        observed and terminal.
 577
 578        Returns early on a terminal failure (no point waiting out the rest),
 579        returns once every required check is terminal, and otherwise keeps
 580        polling until the deadline. On timeout it returns the last observation
 581        with timed_out set, which never authorizes a merge.
 582        """
 583        policy = self.required_check_policy(repo_root, repo_slug)
 584        observation = self.observe_once(repo_root, pr_number, repo_slug, policy=policy)
 585        if not policy.authorizes_merge_gate():
 586            # Nothing to wait for: the policy itself is unknown or enforces no
 587            # required checks. Either way the merge gate is already closed.
 588            return observation
 589        if not observation.pr_head_sha:
 590            # A PR number without a readable live head cannot be bound to CI
 591            # evidence. This is a terminal fail-closed observation, not
 592            # pending CI work, so do not spin until the ordinary 15-minute
 593            # polling deadline.
 594            return observation
 595
 596        deadline = clock_fn() + timeout_seconds
 597        while clock_fn() < deadline:
 598            if observation.failed_jobs or observation.all_required_terminal:
 599                return observation
 600            sleep_fn(poll_interval)
 601            observation = self.observe_once(repo_root, pr_number, repo_slug, policy=policy)
 602
 603        observation.timed_out = not observation.all_required_terminal
 604        return observation

 743    def _dispatch(self, action: ProposedAction, task_id: str) -> Dict[str, Any]:
 744        args = action.arguments or {}
 745        if action.action_type == "create_task_branch":
 746            branch = self.create_task_branch(task_id, args.get("base_branch", "main"))
 747            return {"branch": branch}
 748        if action.action_type == "commit_task_changes":
 749            sha = self.stage_and_commit(
 750                args["paths"], args["message"], args.get("baseline_sha"), args.get("allowed_paths"),
 751            )
 752            return {"commit_sha": sha}
 753        if action.action_type == "push_task_branch":
 754            self.push_branch(args["branch"])
 755            return {"branch": args["branch"], "pushed": True}
 756        if action.action_type == "create_pull_request":
 757            pr_number, pr_url = self.open_pull_request(
 758                args["branch"], args["title"], args["body"], args.get("base", "main")
 759            )
 760            return {"pr_number": pr_number, "pr_url": pr_url}
 761        if action.action_type == "merge_pull_request":
 762            merge_sha = self.merge_pull_request(args["pr_number"])
 763            self._merges_so_far += 1
 764            return {"merge_sha": merge_sha}
 765        if action.action_type == "sync_local_main":
 766            synced = self.sync_local_main()
 767            return {"local_main_synced": synced}
 768        raise GitIntegrationError(f"Unsupported action_type: {action.action_type}")

 891    def create_task_branch(self, task_id: str, base_branch: str = "main") -> str:
 892        branch = f"fix/{task_id}"
 893        fetch = self._git(self.repo_root, ["fetch", "origin", base_branch], 60)
 894        if fetch.returncode != 0:
 895            raise GitIntegrationError(f"git fetch origin {base_branch} failed: {fetch.stderr}")
 896        create = self._git(self.repo_root, ["switch", "-c", branch, f"origin/{base_branch}"], 30)
 897        if create.returncode != 0:
 898            raise GitIntegrationError(f"git switch -c {branch} failed: {create.stderr}")
 899        verify = self._git(self.repo_root, ["rev-parse", "--verify", branch], 15)
 900        if verify.returncode != 0:
 901            raise GitIntegrationError(f"branch '{branch}' does not exist after creation attempt")
 902        return branch

 904    def stage_and_commit(
 905        self,
 906        paths: List[str],
 907        message: str,
 908        baseline_sha: Optional[str] = None,
 909        allowed_paths: Optional[List[str]] = None,
 910    ) -> str:
 911        if not paths:
 912            raise GitIntegrationError("commit_task_changes called with an empty path list -- nothing to stage")
 913        if allowed_paths:
 914            violations = [p for p in paths if p not in allowed_paths]
 915            if violations:
 916                raise GitIntegrationError(
 917                    f"commit_task_changes rejected: path(s) outside task-declared scope {allowed_paths}: {violations}"
 918                )
 919        self._reject_premature_acceptance_merge_claims(paths)
 920        add_proc = self._git(self.repo_root, ["add", "--"] + list(paths), 30)
 921        if add_proc.returncode != 0:
 922            raise GitIntegrationError(f"git add of task-owned paths failed: {add_proc.stderr}")
 923        commit_proc = self._git(self.repo_root, ["commit", "-m", message], 30)
 924        if commit_proc.returncode != 0:
 925            raise GitIntegrationError(f"git commit failed: {commit_proc.stderr}")
 926        sha_proc = self._git(self.repo_root, ["rev-parse", "HEAD"], 15)
 927        if sha_proc.returncode != 0 or not sha_proc.stdout.strip():
 928            raise GitIntegrationError("could not resolve HEAD after commit")
 929        sha = sha_proc.stdout.strip()
 930        if baseline_sha and sha == baseline_sha:
 931            raise GitIntegrationError("HEAD did not move after commit -- commit did not actually happen")
 932        return sha

 934    def _reject_premature_acceptance_merge_claims(self, paths: List[str]) -> None:
 935        """Reject unverified merge success claims before any git subprocess runs."""
 936        for path in paths:
 937            if not LIVE_ACCEPTANCE_JOURNAL_PATTERN.fullmatch(path):
 938                continue
 939            journal_path = self.repo_root / path
 940            try:
 941                content = journal_path.read_text(encoding="utf-8")
 942            except OSError as exc:
 943                raise GitIntegrationError(
 944                    f"commit_task_changes rejected: could not read live acceptance journal '{path}': {exc}"
 945                ) from exc
 946            if PREMATURE_MERGE_SUCCESS_PATTERN.search(content):
 947                raise GitIntegrationError(
 948                    "commit_task_changes rejected: live acceptance journal contains a premature merge success claim"
 949                )

 951    def push_branch(self, branch: str) -> bool:
 952        # Pre-push hooks may run the full deterministic verification suite,
 953        # which takes well over the previous 90-second budget. Allow 300s.
 954        push_proc = self._git(self.repo_root, ["push", "-u", "origin", branch], 300)
 955        if push_proc.returncode != 0:
 956            raise GitIntegrationError(f"git push -u origin {branch} failed: {push_proc.stderr}")
 957        local_sha = self._git(self.repo_root, ["rev-parse", branch], 15).stdout.strip()
 958        ls_remote = self._git(self.repo_root, ["ls-remote", "origin", branch], 30)
 959        remote_sha = ls_remote.stdout.split()[0] if ls_remote.returncode == 0 and ls_remote.stdout.strip() else None
 960        if not remote_sha or remote_sha != local_sha:
 961            raise GitIntegrationError(
 962                f"push reported success but remote branch '{branch}' SHA "
 963                f"('{remote_sha}') does not match local SHA ('{local_sha}')"
 964            )
 965        return True

 967    def open_pull_request(self, branch: str, title: str, body: str, base: str = "main") -> Tuple[int, str]:
 968        create_proc = self._gh(
 969            self.repo_root,
 970            ["pr", "create", "--repo", self.repo_slug, "--base", base, "--head", branch,
 971             "--title", title, "--body", body],
 972            60,
 973        )
 974        if create_proc.returncode != 0:
 975            raise GitIntegrationError(f"gh pr create failed: {create_proc.stderr}")
 976
 977        list_proc = self._gh(
 978            self.repo_root,
 979            ["pr", "list", "--repo", self.repo_slug, "--head", branch, "--json", "number,url"],
 980            30,
 981        )
 982        if list_proc.returncode != 0 or not list_proc.stdout.strip():
 983            raise GitIntegrationError(f"could not confirm PR exists for branch '{branch}' after creation")
 984        try:
 985            prs = json.loads(list_proc.stdout)
 986        except json.JSONDecodeError as exc:
 987            raise GitIntegrationError(f"unparseable PR list response: {exc}") from exc
 988        if not prs:
 989            raise GitIntegrationError(f"gh pr create reported success but no PR found for branch '{branch}'")

 991
 992        view_proc = self._gh(self.repo_root, ["pr", "view", str(pr_number), "--json", "number"], 30)
 993        if view_proc.returncode != 0:
 994            raise GitIntegrationError(f"gh pr view #{pr_number} failed to confirm PR existence: {view_proc.stderr}")

1000    def merge_pull_request(self, pr_number: int) -> str:
1001        if not TASK_BRANCH_PATTERN.match(self._pr_head_branch(pr_number) or ""):
1002            raise GitIntegrationError(f"PR #{pr_number} head branch is not a recognized campaign task branch")
1003
1004        merge_proc = self._gh(
1005            self.repo_root,
1006            ["pr", "merge", str(pr_number), "--repo", self.repo_slug, "--squash", "--delete-branch"],
1007            120,
1008        )
1009        if merge_proc.returncode != 0:
1010            raise GitIntegrationError(f"gh pr merge --squash #{pr_number} failed: {merge_proc.stderr}")
1011
1012        view_proc = self._gh(
1013            self.repo_root,
1014            ["pr", "view", str(pr_number), "--repo", self.repo_slug, "--json", PR_MERGE_FIELDS],
1015            30,
1016        )
1017        if view_proc.returncode != 0 or not view_proc.stdout.strip():
1018            raise GitIntegrationError(
1019                f"could not independently verify merge of PR #{pr_number}: {view_proc.stderr.strip()}"
1020            )
1021        try:
