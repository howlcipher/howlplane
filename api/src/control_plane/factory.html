factory

The persistent supervisory layer above individual governed tasks.

Authorized by the named carve-out in documentation/CONTROL_PLANE.md section 1.1.1 and specified in documentation/adr/0006_persistent_factory_supervisor.md.

This package decides what is worth attempting. It never decides what is permitted: AuthorityProfile, AuthorityEnvelope and HumanBoundaryGate remain the only authority paths, and admitting an item to the portfolio grants it nothing.

253    def transition_to(self, new_state: str, reason: str = "") -> None:
254        """Move to `new_state`, refusing any transition not on the map."""
255        target = _plain(new_state)
256        allowed = WORK_ITEM_TRANSITIONS.get(self.state, set())
257        if target not in allowed:
258            # Rendered with plain values, not enum reprs: the supervisor logs
259            # this from inside a loop where the message is the only context.
260            legal = sorted(_plain(state) for state in allowed)
261            raise InvalidWorkItemTransitionError(
262                f"{self.work_item_id}: cannot move from '{self.state}' to "
263                f"'{target}'. Allowed: {legal or 'none (terminal state)'}."
264            )
265        self.state = target
266        self.updated_at = _now()
267        self.reopening_history.append(
268            {"at": self.updated_at, "to_state": _plain(new_state), "reason": reason}

157def select(
158    work_items: Sequence[WorkItem],
159    window: Sequence[Dict[str, str]],
160    policy: FactoryPolicy,
161    now: Optional[datetime] = None,
162) -> SelectionOutcome:
163    """Pick the next dispatchable item, or explain why nothing was picked.
164
165    `window` is the recent dispatch history, most recent last, each entry
166    carrying at least `repository` and `origin`.
167    """
168    now = now or datetime.now(timezone.utc)
169    counts = policy.counts(window)
170
171    candidates = sorted(
172        (item for item in work_items if item.is_dispatchable),
173        key=lambda i: _sort_key(i, now, policy),
174    )
175    if not candidates:
176        return SelectionOutcome(reason="no_dispatchable_work", no_valuable_work=True)
177
178    # Every candidate is evaluated, not just those ahead of the winner. The
179    # owner asking "why is my self-improvement work not running" needs the
180    # answer even when something else was dispatched this tick, so `withheld`
181    # is the complete set of capped candidates rather than a prefix of it.
182    chosen: Optional[WorkItem] = None
183    withheld: List[Dict[str, str]] = []
184    for item in candidates:
185        breach = _cap_blocking(item, counts, policy)
186        if breach is None:
187            if chosen is None:
188                chosen = item
189            continue
190        withheld.append(
191            {
192                "work_item_id": item.work_item_id,
193                "origin": item.origin,
194                "repository": item.repository,
195                "reason": breach,
196            }
197        )
198
199    if chosen is not None:
200        return SelectionOutcome(item=chosen, reason="selected", withheld=withheld)
201
202    # Everything ready would over-serve a category. Idling here is the point:
203    # the alternative is grinding on whatever is left regardless of balance.
204    return SelectionOutcome(reason="all_candidates_capped", no_valuable_work=True, withheld=withheld)

def work_item_fingerprint(origin: str, repository: str, keys: List[str]) -> str: 
142def work_item_fingerprint(origin: str, repository: str, keys: List[str]) -> str:
143    """Stable identity for a piece of work, independent of when it was seen.
144
145    The repository is part of the identity because backlog item ids are only
146    unique within a file. `BacklogItem.task_id` hardcodes a `HOWLFRAM-` prefix
147    regardless of which repository the row came from, so two repositories can
148    and do produce the same task id -- the fingerprint must not inherit that
149    collision.
150    """
151    canonical = json.dumps(
152        {"origin": str(origin), "repository": repository, "keys": sorted(keys)},
153        sort_keys=True,
154        separators=(",", ":"),
155    )
156    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()[:16]
