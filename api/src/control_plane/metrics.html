277    @staticmethod
278    def calculate(entries: List[EvidenceEntry]) -> PerformanceMetricsSummary:
279        """
280        Parses evidence ledger entries and constructs a PerformanceMetricsSummary
281        with strict first_pass_success semantics.
282        Independent properties are processed independently without mutually exclusive suppression.
283        """
284        summary = PerformanceMetricsSummary()
285        if not entries:
286            return summary
287
288        # Group entries by task
289        tasks: Dict[str, List[EvidenceEntry]] = {}
290        for entry in entries:
291            tasks.setdefault(entry.task_id, []).append(entry)
292
293        summary.total_tasks = len(tasks)
294
295        for task_id, task_entries in tasks.items():
296            task_agents = set()
297            repo_name = None
298            is_task_override = False
299
300            for e in task_entries:
301                if e.agent_id:
302                    task_agents.add(e.agent_id)
303                if e.implementing_agent:
304                    task_agents.add(e.implementing_agent)
305                if e.actual_agent:
306                    task_agents.add(e.actual_agent)
307                if e.repository:
308                    repo_name = e.repository
309                elif isinstance(e.metadata, dict) and "repository" in e.metadata:
310                    repo_name = e.metadata["repository"]
311                if e.is_override is True or (isinstance(e.metadata, dict) and e.metadata.get("is_override") is True):
312                    is_task_override = True
313
314            if repo_name:
315                summary.repositories_exercised[repo_name] = summary.repositories_exercised.get(repo_name, 0) + 1
316
317            if is_task_override:
318                summary.routing_overrides += 1
319            else:
320                summary.routing_recommendations_followed += 1
321
322            for aid in task_agents:
323                if aid not in summary.agent_summaries:
324                    summary.agent_summaries[aid] = AgentMetricSummary(agent_id=aid)
325                summary.agent_summaries[aid].tasks_worked += 1
326
327            # Task level tracking flags
328            has_remediation = False
329            has_verif_failure = False
330            has_confirmed_blocker_or_high = False
331            has_human_intervention = False
332            is_completed = False
333            is_abandoned = False
334            task_defects_caught = 0
335            task_remediation_cycles = 0
336
337            for e in task_entries:
338                aid = e.actual_agent or e.implementing_agent or e.agent_id
339                agent_sum = summary.agent_summaries.get(aid)
340
341                # 1. Track task class
342                if e.task_class and agent_sum:
343                    agent_sum.task_classes_worked[e.task_class] = (
344                        agent_sum.task_classes_worked.get(e.task_class, 0) + 1
345                    )
346
347                # 2. Track orchestration action
348                orch_act = e.orchestration_action or (e.metadata.get("orchestration_action") if isinstance(e.metadata, dict) else None)
349                if orch_act:
350                    summary.orchestration_counts[orch_act] = summary.orchestration_counts.get(orch_act, 0) + 1
351
352                # 3. Track defect types
353                if e.defect_type == "review_caught_defect" or (e.action == "control_plane_defect_caught" and isinstance(e.metadata, dict) and (e.metadata.get("reviewer_role") or "review" in e.metadata.get("failure_mode", ""))):
354                    summary.review_caught_defects += 1
355                elif e.defect_type == "verification_caught_defect" or (e.action == "verification_executed" and e.result == "failed"):
356                    summary.verification_caught_defects += 1
357                elif e.defect_type == "boundary_caught_risk" or (e.action in ("boundary_checked", "boundary_triggered") and isinstance(e.metadata, dict) and e.metadata.get("boundary_triggered")):
358                    summary.boundary_caught_risks += 1
359
360                # 4. Track repository hygiene / slop metrics
361                is_slop_check = e.action == "slop_check_executed" or (
362                    e.action == "verification_executed"
363                    and (
364                        (e.command and "slopslint" in e.command)
365                        or (isinstance(e.metadata, dict) and e.metadata.get("category") == "repository_hygiene")
366                    )
367                )
368                if is_slop_check:
369                    summary.slop_checks_run += 1
370                    if e.result == "failed":
371                        summary.slop_checks_failed += 1
372
373                if e.action == "debt_acceptance_requested" or (
374                    e.action in ("boundary_triggered", "boundary_checked")
375                    and isinstance(e.metadata, dict)
376                    and "slop_debt_acceptance" in (e.metadata.get("boundary_triggered") or [])
377                ):
378                    summary.debt_acceptance_requests += 1
379
380                if e.action == "debt_acceptance_approved" or (
381                    e.action == "human_decision"
382                    and e.result in ("approved", "accepted")
383                    and isinstance(e.metadata, dict)
384                    and "slop_debt_acceptance" in (e.metadata.get("boundary_triggers") or [])
385                ):
386                    summary.debt_acceptance_approved += 1
387
388                if e.action == "debt_acceptance_rejected" or (
389                    e.action == "human_decision"
390                    and e.result in ("rejected", "denied")
391                    and isinstance(e.metadata, dict)
392                    and "slop_debt_acceptance" in (e.metadata.get("boundary_triggers") or [])
393                ):
394                    summary.debt_acceptance_rejected += 1
395
396                if e.action == "hygiene_policy_tightened":
397                    summary.hygiene_policies_tightened += 1
398
399                if e.action in ("hygiene_policy_weakening_requested", "hygiene_policy_weakened"):
400                    summary.hygiene_policies_weakened += 1
401
402                if e.action == "hygiene_ceiling_lowered":
403                    summary.hygiene_ceilings_lowered += 1
404
405                if e.action == "hygiene_ceiling_raise_attempted":
406                    summary.hygiene_ceiling_raise_attempts += 1
407
408                if e.action == "hygiene_tombstone_added":
409                    summary.hygiene_tombstones_added += 1
410
411                if e.action == "hygiene_tombstone_removed":
412                    summary.hygiene_tombstones_removed += 1
413
414                if e.action == "hygiene_provider_mismatch" or (
415                    isinstance(e.metadata, dict) and e.metadata.get("failure_mode") == "hygiene_provider_mismatch"
416                ):
417                    summary.hygiene_provider_mismatches += 1
418
419                if e.action == "slop_finding_remediated":
420                    summary.slop_findings_remediated += 1
421
422                # 5. Track control_plane_caught_defect & failure modes
423                if e.control_plane_caught_defect is True or e.action == "control_plane_defect_caught" or (e.action == "verification_executed" and e.result == "failed"):
424                    summary.control_plane_caught_defects += 1
425                    task_defects_caught += 1
426                    if agent_sum:
427                        agent_sum.control_plane_defects_caught += 1
428                    if isinstance(e.metadata, dict) and "failure_mode" in e.metadata:
429                        fmode = e.metadata["failure_mode"]
430                        summary.common_failure_modes[fmode] = summary.common_failure_modes.get(fmode, 0) + 1
431                        if fmode in ("duplication_regression", "ceiling_regression"):
432                            summary.duplication_regressions_caught += 1
433                        elif fmode == "orphan_surface":
434                            summary.orphans_caught += 1
435                        elif fmode == "stale_tombstone":
436                            summary.stale_tombstones_caught += 1
437                        elif fmode == "contract_claim_gap":
438                            summary.claim_gaps_caught += 1
439
440                # 5. Track remediation cycles
441                if e.remediation_cycles and e.remediation_cycles > 0:
442                    has_remediation = True
443                    task_remediation_cycles = max(task_remediation_cycles, e.remediation_cycles)
444
445                if e.action == "remediation_started":
446                    has_remediation = True
447                elif e.action == "remediation_completed":
448                    has_remediation = True
449                    task_remediation_cycles += 1
450                    summary.rework_cycles_total += 1
451                    if agent_sum:
452                        agent_sum.remediation_cycles += 1
453                elif e.action == "reconciliation_completed":
454                    if e.result and "remediated" in e.result.lower():
455                        has_remediation = True
456                        task_remediation_cycles += 1
457
458                # 6. Track reviewer stats & findings
459                if e.action == "review_submitted" or e.reviewing_agents:
460                    reviewers_list = e.reviewing_agents or (
461                        e.metadata.get("reviewers") if isinstance(e.metadata, dict) else None
462                    )
463                    if reviewers_list:
464                        for r_role in reviewers_list:
465                            if r_role not in summary.reviewer_summaries:
466                                summary.reviewer_summaries[r_role] = ReviewerMetricSummary(role_id=r_role)
467                            summary.reviewer_summaries[r_role].reviewer_runs += 1
468
469                if e.findings_summary:
470                    total_f = e.findings_summary.get("total", 0)
471                    blockers = e.findings_summary.get("blocker", 0)
472                    highs = e.findings_summary.get("high", 0)
473                    confirmed = e.findings_summary.get("confirmed", 0)
474
475                    summary.total_review_findings += total_f
476                    summary.blocker_findings += blockers
477                    summary.high_findings += highs
478                    summary.confirmed_review_findings += confirmed
479
480                    if agent_sum:
481                        agent_sum.review_findings_received += total_f
482                        agent_sum.confirmed_findings_received += confirmed
483                        agent_sum.blocker_findings_received += blockers
484
485                    if blockers > 0 or highs > 0:
486                        has_confirmed_blocker_or_high = True
487
488                if isinstance(e.metadata, dict) and "reviewers_breakdown" in e.metadata:
489                    rb = e.metadata["reviewers_breakdown"]
490                    for r_role, r_stats in rb.items():
491                        if r_role not in summary.reviewer_summaries:
492                            summary.reviewer_summaries[r_role] = ReviewerMetricSummary(role_id=r_role)
493                        r_sum = summary.reviewer_summaries[r_role]
494                        r_sum.findings_total += r_stats.get("findings_total", 0)
495                        r_sum.confirmed_findings += r_stats.get("confirmed", 0)
496                        r_sum.likely_findings += r_stats.get("likely", 0)
497                        r_sum.false_positives += r_stats.get("false_positives", 0)
498                        r_sum.disputed_findings += r_stats.get("disputed", 0)
499                        r_sum.blockers_found += r_stats.get("blockers", 0)
500                        r_sum.high_findings_found += r_stats.get("highs", 0)
501                        r_sum.unique_findings += r_stats.get("unique", 0)
502                        r_sum.findings_that_triggered_remediation += r_stats.get("triggered_remediation", 0)
503                        r_sum.findings_that_prevented_bad_completion += r_stats.get("prevented_bad_completion", 0)
504
505                # 7. Track verification
506                if e.action == "verification_executed":
507                    if e.result == "failed":
508                        has_verif_failure = True
509                        summary.verification_failures += 1
510                        if agent_sum:
511                            agent_sum.verification_failures += 1
512                        fmode = "verification_failure"
513                        summary.common_failure_modes[fmode] = summary.common_failure_modes.get(fmode, 0) + 1
514
515                # 8. Track human decision
516                if e.action == "human_decision" or e.human_decision is not None:
517                    has_human_intervention = True
518                    summary.human_interventions += 1
519                    if agent_sum:
520                        agent_sum.human_interventions += 1
521
522                # 9. Track task closed
523                if e.action == "task_closed":
524                    if e.result == "complete":
525                        is_completed = True
526                        if agent_sum:
527                            agent_sum.tasks_completed += 1
528                    elif e.result in ("failed", "abandoned"):
529                        is_abandoned = True
530                        if agent_sum:
531                            agent_sum.tasks_abandoned += 1
532
533            if has_remediation:
534                summary.tasks_requiring_remediation += 1
535
536            if is_completed:
537                summary.completed_tasks += 1
538                is_first_pass = (
539                    not has_remediation
540                    and not has_verif_failure
541                    and not has_confirmed_blocker_or_high
542                    and not has_human_intervention
543                    and task_remediation_cycles == 0
544                )
545                if is_first_pass:
546                    summary.first_pass_successes += 1
547                    for aid in task_agents:
548                        if aid in summary.agent_summaries:
549                            summary.agent_summaries[aid].first_pass_successes += 1
550            elif is_abandoned:
551                summary.abandoned_tasks += 1
552
553        # Compute marginal value for each reviewer
554        for r_role, r_sum in summary.reviewer_summaries.items():
555            r_sum.marginal_value = (
556                r_sum.unique_findings
557                + r_sum.findings_that_triggered_remediation
558                + r_sum.findings_that_prevented_bad_completion
559                + r_sum.blockers_found
560            )
561
562        return summary

340                agent_sum = summary.agent_summaries.get(aid)
341
342                # 1. Track task class
343                if e.task_class and agent_sum:
344                    agent_sum.task_classes_worked[e.task_class] = (
345                        agent_sum.task_classes_worked.get(e.task_class, 0) + 1
346                    )
347
348                # 2. Track orchestration action
349                orch_act = e.orchestration_action or (e.metadata.get("orchestration_action") if isinstance(e.metadata, dict) else None)
350                if orch_act:
351                    summary.orchestration_counts[orch_act] = summary.orchestration_counts.get(orch_act, 0) + 1
352
353                # 3. Track defect types
354                if e.defect_type == "review_caught_defect" or (e.action == "control_plane_defect_caught" and isinstance(e.metadata, dict) and (e.metadata.get("reviewer_role") or "review" in e.metadata.get("failure_mode", ""))):
355                    summary.review_caught_defects += 1
356                elif e.defect_type == "verification_caught_defect" or (e.action == "verification_executed" and e.result == "failed"):
357                    summary.verification_caught_defects += 1
358                elif e.defect_type == "boundary_caught_risk" or (e.action in ("boundary_checked", "boundary_triggered") and isinstance(e.metadata, dict) and e.metadata.get("boundary_triggered")):
359                    summary.boundary_caught_risks += 1
360
361                # 4. Track repository hygiene / slop metrics
362                is_slop_check = e.action == "slop_check_executed" or (
363                    e.action == "verification_executed"
364                    and (
365                        (e.command and "slopslint" in e.command)
366                        or (isinstance(e.metadata, dict) and e.metadata.get("category") == "repository_hygiene")
367                    )
368                )
369                if is_slop_check:
370                    summary.slop_checks_run += 1
371                    if e.result == "failed":
372                        summary.slop_checks_failed += 1
373
374                if e.action == "debt_acceptance_requested" or (
375                    e.action in ("boundary_triggered", "boundary_checked")
376                    and isinstance(e.metadata, dict)
377                    and "slop_debt_acceptance" in (e.metadata.get("boundary_triggered") or [])
378                ):
379                    summary.debt_acceptance_requests += 1
380
381                if e.action == "debt_acceptance_approved" or (
382                    e.action == "human_decision"
383                    and e.result in ("approved", "accepted")
384                    and isinstance(e.metadata, dict)
385                    and "slop_debt_acceptance" in (e.metadata.get("boundary_triggers") or [])
386                ):
387                    summary.debt_acceptance_approved += 1
388
389                if e.action == "debt_acceptance_rejected" or (
390                    e.action == "human_decision"
391                    and e.result in ("rejected", "denied")
392                    and isinstance(e.metadata, dict)
393                    and "slop_debt_acceptance" in (e.metadata.get("boundary_triggers") or [])
394                ):
395                    summary.debt_acceptance_rejected += 1
396
397                if e.action == "hygiene_policy_tightened":
398                    summary.hygiene_policies_tightened += 1
399
400                if e.action in ("hygiene_policy_weakening_requested", "hygiene_policy_weakened"):
401                    summary.hygiene_policies_weakened += 1
402
403                if e.action == "hygiene_ceiling_lowered":
404                    summary.hygiene_ceilings_lowered += 1
405
406                if e.action == "hygiene_ceiling_raise_attempted":
407                    summary.hygiene_ceiling_raise_attempts += 1
408
409                if e.action == "hygiene_tombstone_added":
410                    summary.hygiene_tombstones_added += 1
411
412                if e.action == "hygiene_tombstone_removed":
413                    summary.hygiene_tombstones_removed += 1
414
415                if e.action == "hygiene_provider_mismatch" or (
416                    isinstance(e.metadata, dict) and e.metadata.get("failure_mode") == "hygiene_provider_mismatch"
417                ):
418                    summary.hygiene_provider_mismatches += 1
419
420                if e.action == "slop_finding_remediated":
421                    summary.slop_findings_remediated += 1
422
423                # 5. Track control_plane_caught_defect & failure modes
424                if e.control_plane_caught_defect is True or e.action == "control_plane_defect_caught" or (e.action == "verification_executed" and e.result == "failed"):
425                    summary.control_plane_caught_defects += 1
426                    task_defects_caught += 1
427                    if agent_sum:
428                        agent_sum.control_plane_defects_caught += 1
429                    if isinstance(e.metadata, dict) and "failure_mode" in e.metadata:
430                        fmode = e.metadata["failure_mode"]
431                        summary.common_failure_modes[fmode] = summary.common_failure_modes.get(fmode, 0) + 1
432                        if fmode in ("duplication_regression", "ceiling_regression"):
433                            summary.duplication_regressions_caught += 1
434                        elif fmode == "orphan_surface":
435                            summary.orphans_caught += 1
436                        elif fmode == "stale_tombstone":
437                            summary.stale_tombstones_caught += 1
438                        elif fmode == "contract_claim_gap":
439                            summary.claim_gaps_caught += 1
440
441                # 5. Track remediation cycles
442                if e.remediation_cycles and e.remediation_cycles > 0:
443                    has_remediation = True
444                    task_remediation_cycles = max(task_remediation_cycles, e.remediation_cycles)
445
446                if e.action == "remediation_started":
447                    has_remediation = True
448                elif e.action == "remediation_completed":
449                    has_remediation = True
450                    task_remediation_cycles += 1
451                    summary.rework_cycles_total += 1
452                    if agent_sum:
453                        agent_sum.remediation_cycles += 1
454                elif e.action == "reconciliation_completed":
455                    if e.result and "remediated" in e.result.lower():
456                        has_remediation = True
457                        task_remediation_cycles += 1
458
459                # 6. Track reviewer stats & findings
460                if e.action == "review_submitted" or e.reviewing_agents:
461                    reviewers_list = e.reviewing_agents or (
462                        e.metadata.get("reviewers") if isinstance(e.metadata, dict) else None
463                    )
464                    if reviewers_list:
465                        for r_role in reviewers_list:
466                            if r_role not in summary.reviewer_summaries:
467                                summary.reviewer_summaries[r_role] = ReviewerMetricSummary(role_id=r_role)
468                            summary.reviewer_summaries[r_role].reviewer_runs += 1
469
470                if e.findings_summary:
471                    total_f = e.findings_summary.get("total", 0)
472                    blockers = e.findings_summary.get("blocker", 0)
473                    highs = e.findings_summary.get("high", 0)
474                    confirmed = e.findings_summary.get("confirmed", 0)
475
476                    summary.total_review_findings += total_f
477                    summary.blocker_findings += blockers
478                    summary.high_findings += highs
479                    summary.confirmed_review_findings += confirmed
480
481                    if agent_sum:
482                        agent_sum.review_findings_received += total_f
483                        agent_sum.confirmed_findings_received += confirmed
484                        agent_sum.blocker_findings_received += blockers
485
486                    if blockers > 0 or highs > 0:
487                        has_confirmed_blocker_or_high = True
488
489                if isinstance(e.metadata, dict) and "reviewers_breakdown" in e.metadata:
490                    rb = e.metadata["reviewers_breakdown"]
491                    for r_role, r_stats in rb.items():
492                        if r_role not in summary.reviewer_summaries:
493                            summary.reviewer_summaries[r_role] = ReviewerMetricSummary(role_id=r_role)
494                        r_sum = summary.reviewer_summaries[r_role]
495                        r_sum.findings_total += r_stats.get("findings_total", 0)
496                        r_sum.confirmed_findings += r_stats.get("confirmed", 0)
497                        r_sum.likely_findings += r_stats.get("likely", 0)
498                        r_sum.false_positives += r_stats.get("false_positives", 0)
499                        r_sum.disputed_findings += r_stats.get("disputed", 0)
500                        r_sum.blockers_found += r_stats.get("blockers", 0)
501                        r_sum.high_findings_found += r_stats.get("highs", 0)
502                        r_sum.unique_findings += r_stats.get("unique", 0)
503                        r_sum.findings_that_triggered_remediation += r_stats.get("triggered_remediation", 0)
504                        r_sum.findings_that_prevented_bad_completion += r_stats.get("prevented_bad_completion", 0)
505
506                # 7. Track verification
507                if e.action == "verification_executed":
508                    if e.result == "failed":
509                        has_verif_failure = True
510                        summary.verification_failures += 1
511                        if agent_sum:
512                            agent_sum.verification_failures += 1
513                        fmode = "verification_failure"
514                        summary.common_failure_modes[fmode] = summary.common_failure_modes.get(fmode, 0) + 1
515
516                # 8. Track human decision
517                if e.action == "human_decision" or e.human_decision is not None:
518                    has_human_intervention = True
519                    summary.human_interventions += 1
520                    if agent_sum:
521                        agent_sum.human_interventions += 1
522
523                # 9. Track task closed
524                if e.action == "task_closed":
525                    if e.result == "complete":
526                        is_completed = True
527                        if agent_sum:
528                            agent_sum.tasks_completed += 1
529                    elif e.result in ("failed", "abandoned"):
530                        is_abandoned = True
531                        if agent_sum:
532                            agent_sum.tasks_abandoned += 1
533
534            if has_remediation:
535                summary.tasks_requiring_remediation += 1
536
537            if is_completed:
538                summary.completed_tasks += 1
539                is_first_pass = (
540                    not has_remediation
541                    and not has_verif_failure
542                    and not has_confirmed_blocker_or_high
543                    and not has_human_intervention
544                    and task_remediation_cycles == 0
545                )
546                if is_first_pass:
547                    summary.first_pass_successes += 1
548                    for aid in task_agents:
549                        if aid in summary.agent_summaries:
550                            summary.agent_summaries[aid].first_pass_successes += 1
551            elif is_abandoned:
552                summary.abandoned_tasks += 1
553
554        # Compute marginal value for each reviewer
555        for r_role, r_sum in summary.reviewer_summaries.items():
556            r_sum.marginal_value = (
557                r_sum.unique_findings
558                + r_sum.findings_that_triggered_remediation
559                + r_sum.findings_that_prevented_bad_completion
560                + r_sum.blockers_found
561            )
562
563        return summary

Parses evidence ledger entries and constructs a PerformanceMetricsSummary with strict first_pass_success semantics. Independent properties are processed independently without mutually exclusive suppression.
