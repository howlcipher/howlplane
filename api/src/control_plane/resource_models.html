 52class ProviderFailureClass(str, Enum):
 53    """Central normalized provider and engineering failure taxonomy."""
 54
 55    QUOTA_EXHAUSTED = "QUOTA_EXHAUSTED"
 56    SESSION_LIMIT = "SESSION_LIMIT"
 57    RATE_LIMITED = "RATE_LIMITED"
 58    AUTHENTICATION_REQUIRED = "AUTHENTICATION_REQUIRED"
 59    PROVIDER_UNAVAILABLE = "PROVIDER_UNAVAILABLE"
 60    TRANSPORT_UNAVAILABLE = "TRANSPORT_UNAVAILABLE"
 61    MISSING_EXECUTABLE = "MISSING_EXECUTABLE"
 62    ENGINEERING_FAILURE = "ENGINEERING_FAILURE"
 63    VERIFICATION_FAILURE = "VERIFICATION_FAILURE"
 64    MALFORMED_OUTPUT = "MALFORMED_OUTPUT"
 65    CAPABILITY_FAILURE = "CAPABILITY_FAILURE"
 66    POLICY_FAILURE = "POLICY_FAILURE"
 67    # The provider launched and reasoned, but was denied a tool the task
 68    # required and therefore produced no work. Distinct from MISSING_EXECUTABLE
 69    # (never launched), AUTHENTICATION_REQUIRED (launched but not signed in),
 70    # and TRANSPORT_UNAVAILABLE (launched but could not reach the provider).
 71    EXECUTION_PERMISSION_REQUIRED = "EXECUTION_PERMISSION_REQUIRED"
 72    # This control plane enforced its own per-attempt wall-clock budget and
 73    # killed the provider. That is a fact about our deadline, not about the
 74    # provider: the transport was never observed to fail, so this must not be
 75    # reported as TRANSPORT_UNAVAILABLE and must not mark the resource
 76    # unreachable. A provider stopped this way may still have left a usable
 77    # repository delta, which is governed as a candidate rather than trusted
 78    # (HOWLFRAM-SLOPFIX-05).
 79    EXECUTION_BUDGET_EXCEEDED = "EXECUTION_BUDGET_EXCEEDED"
 80    # The provider remained alive but produced no meaningful semantic progress
 81    # or task-attributable repository change during the watchdog window.
 82    # This is distinct from the absolute execution budget: it is evidence that
 83    # this invocation is unusable, not evidence that the provider is offline.
 84    PROVIDER_STALLED = "PROVIDER_STALLED"
 85    UNKNOWN = "UNKNOWN"

107    """Safe adapter readiness result that never requires generation.
108
109    `status` answers only "can this executable be launched". Whether the
110    launched provider can actually exercise the mutation tools a task needs is
111    a separate question, reported by `unattended_mutation_capable`, so a
112    provider is never described as READY for work it cannot perform

114    select a provider that could start but not edit.

147class ResourceSelectionDecision(DataClassSerializationMixin):
148    """Stable evidence packet for every deterministic selection attempt."""
149
150    task_class: str
151    role: str
152    required_capabilities: List[str]
153    status: ResourceSelectionStatus
154    registered_resources: List[ResourceIdentity] = field(default_factory=list)
155    eligible_resources: List[ResourceIdentity] = field(default_factory=list)
156    exclusions: List[ResourceExclusion] = field(default_factory=list)
157    selected: Optional[ResourceIdentity] = None
158    selection_policy: str = "adaptive_capacity"
159    economic_policy: Dict[str, Any] = field(default_factory=dict)
160    cognitive_recommendation: Optional[CognitiveRecommendation] = None
161    capacity_before: Dict[str, str] = field(default_factory=dict)
162    explicit_override: bool = False
163    blocked_reason: Optional[str] = None
164    diversity_achieved: Optional[bool] = None
165    schema: str = RESOURCE_SELECTION_SCHEMA_VERSION
166
167    def exclusion_for(self, resource_id: str) -> ResourceExclusion:
168        """Returns one resource's exclusion or raises a precise lookup error."""
169        for exclusion in self.exclusions:
170            if exclusion.resource_id == resource_id:
171                return exclusion
172        raise KeyError(f"No exclusion was recorded for resource '{resource_id}'.")
173
174    def blocked_outcome(self) -> Dict[str, Any]:
175        """Returns the structured public failure contract."""
176        return {
177            "status": "BLOCKED",
178            "reason": self.blocked_reason or "NO_ELIGIBLE_AI_RESOURCE",
179            "role": self.role,
180            "exclusions": [item.to_dict() for item in self.exclusions],
181            "schema": self.schema,
182        }
