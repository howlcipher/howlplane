 83def assert_no_execution_authority(payload: Any) -> None:
 84    """
 85    Hard negative authority assertion.
 86
 87    Ensures that speculative outputs (from HowlDream, HowlFrame assessment, or
 88    HowlCreate) never claim execution authority, contain self-approvals, or attempt
 89    to authorize actions downstream in HowlPlane executor or HowlChangeOps.
 90    """
 91    if isinstance(payload, dict):
 92        # Check authority block
 93        auth = payload.get("authority")
 94        if isinstance(auth, dict):
 95            if auth.get("executable") is True:
 96                raise AuthorityEscalationError(
 97                    "Exploration output claimed executable=True; rejected"
 98                )
 99            if auth.get("type") not in ("ADVISORY", None):
100                raise AuthorityEscalationError(
101                    f"Exploration output claimed authority type {auth.get('type')}; rejected"
102                )
103
104        # Check self-approval
105        if payload.get("approved") is True or payload.get("self_approved") is True:
106            raise AuthorityEscalationError(
107                "Exploration output claimed self-approval; rejected"
108            )
109
110        # Check execution capability in development results
111        if payload.get("execution_capability") is True:
112            raise AuthorityEscalationError(
113                "Development result claimed execution capability; rejected"
114            )
115
116        # Recursively check sub-dictionaries and lists
117        for v in payload.values():
118            assert_no_execution_authority(v)
119    elif isinstance(payload, list):
120        for item in payload:
121            assert_no_execution_authority(item)

378    def evaluate_candidate_with_howlframe(
379        self,
380        candidate_handoff: Union[Dict[str, Any], Path, str],
381        tmp_dir: Optional[Path] = None,
382    ) -> Dict[str, Any]:
383        """
384        Executes HowlFrame candidate_evaluator bytecode against a candidate handoff.
385        Emits a structured howl.assessment/v1 payload.
386        """
387        bc_path = self.resolve_candidate_evaluator_bc()
388        howlframe_bin = shutil.which("howlframe") or os.environ.get("HOWLFRAME_BIN")
389
390        if not howlframe_bin or not bc_path or not bc_path.is_file():
391            # In-process fallback evaluator conforming to candidate_evaluator.howl rules
392            return self._fallback_evaluate_candidate(candidate_handoff)
393
394        # Write candidate to temporary file if passed as dict
395        if isinstance(candidate_handoff, dict):
396            import tempfile
397            temp_f = tempfile.NamedTemporaryFile(
398                mode="w", suffix=".json", delete=False, dir=str(tmp_dir) if tmp_dir else None
399            )
400            json.dump(candidate_handoff, temp_f)
401            temp_f.close()
402            cand_path = Path(temp_f.name)
403            should_cleanup = True
404        else:
405            cand_path = Path(candidate_handoff)
406            should_cleanup = False
407
408        try:
409            res = subprocess.run(
410                [
411                    str(howlframe_bin),
412                    "-run-bc",
413                    "-allow-caps",
414                    "filesystem",
415                    str(bc_path),
416                    str(cand_path),
417                ],
418                capture_output=True,
419                text=True,
420                timeout=15.0,
421                check=False,
422            )
423            if res.returncode == 0 and res.stdout.strip():
424                data = json.loads(res.stdout.strip())
425                assert_no_execution_authority(data)
426                return data
427            else:
428                return self._fallback_evaluate_candidate(candidate_handoff)
429        except Exception:
430            return self._fallback_evaluate_candidate(candidate_handoff)
431        finally:
432            if should_cleanup and cand_path.is_file():
433                cand_path.unlink()

435    def _fallback_evaluate_candidate(
436        self, candidate_handoff: Union[Dict[str, Any], Path, str]
437    ) -> Dict[str, Any]:
438        """In-process evaluation mirroring candidate_evaluator.howl rules."""
439        if isinstance(candidate_handoff, (str, Path)):
440            with open(candidate_handoff, "r", encoding="utf-8") as f:
441                data = json.load(f)
442        else:
443            data = candidate_handoff
444
445        # Enforce negative authority assertion
446        try:
447            assert_no_execution_authority(data)
448        except AuthorityEscalationError as exc:
449            return {
450                "schema": "howl.assessment/v1",
451                "disposition": "REJECT",
452                "authority": {"type": "ADVISORY", "executable": False},
453                "explanation": f"Authority escalation attempt rejected: {exc}",
454            }
455
456        cand = data.get("candidate", {})
457        contradictions = cand.get("critical_contradictions", 0)
458        status = cand.get("status", "")
459        unresolved = cand.get("unresolved_assumptions", [])
460
461        if contradictions > 0 or status == "CONTRADICTED":
462            disposition = "REJECT"
463            reason = f"Candidate contradicted by NIGHTMARE analysis ({contradictions} conflicts)"
464        elif status == "RESOLVED":
465            disposition = "ACCEPT_FOR_DEVELOPMENT"
466            reason = "Candidate resolved invariants and grounded by WAKE evidence"
467        elif len(unresolved) > 0 or status == "EXPLORED":
468            disposition = "ACCEPT_FOR_DEVELOPMENT"
469            reason = "Candidate has explored potential and bounded assumptions"
470        else:
471            disposition = "UNRESOLVED"
472            reason = "Evidence is inconclusive; candidate deferred"
473
474        return {
475            "schema": "howl.assessment/v1",
476            "candidate_id": cand.get("id", data.get("candidate_id", "unknown")),
477            "parent_dream_id": data.get("parent_dream_id", ""),
478            "disposition": disposition,
479            "authority": {"type": "ADVISORY", "executable": False},
480            "explanation": reason,
481        }

483    def promote_candidate_to_howlcreate(
484        self,
485        candidate_handoff: Union[Dict[str, Any], Path, str],
486        assessment: Union[Dict[str, Any], Path, str],
487    ) -> Dict[str, Any]:
488        """
489        Dispatches accepted candidate to HowlCreate deliberate sandbox development.
490        Ensures that output is purely advisory and non-executable.
491        """
492        assert_no_execution_authority(candidate_handoff)
493        assert_no_execution_authority(assessment)
494
495        try:
496            from howlcreate.engine.candidate_ingestion import develop_candidate
497
498            dev_result = develop_candidate(candidate_handoff, assessment)
499            dev_dict = (
500                dev_result if isinstance(dev_result, dict) else dev_result.model_dump()
501            )
502            assert_no_execution_authority(dev_dict)
503            return dev_dict
504        except ImportError:
505            # Fallback when howlcreate is not installed in current env
506            if isinstance(assessment, (str, Path)):
507                with open(assessment, "r", encoding="utf-8") as f:
508                    assess_data = json.load(f)
509            else:
510                assess_data = assessment
511
512            if assess_data.get("disposition") != "ACCEPT_FOR_DEVELOPMENT":
513                raise ValueError("Cannot promote candidate not ACCEPT_FOR_DEVELOPMENT")
514
515            return {
516                "schema_version": "howl.development_result/v1",
517                "disposition": "ACCEPTED",
518                "execution_authority": "NONE",
519                "prototype_plan": "Fallback sandbox prototype plan",
520                "test_plan": ["verify invariants in sandbox"],
521                "authority": {"type": "ADVISORY", "executable": False},

524    def dispatch_exploration(
525        self,
526        objective: str,
527        context: Optional[ProjectContext] = None,
528        budget: Optional[ExplorationBudget] = None,
529        policy: Optional[ExplorationPolicy] = None,
530        repo_dir: Optional[Path] = None,
531        current_depth: int = 0,
532        run_howlframe: bool = True,
533        run_howlcreate: bool = True,
534    ) -> ExplorationRunResult:
535        """
536        Dispatches a governed exploration cycle through HowlDream, HowlFrame, and HowlCreate.
537        """
538        t0 = time.time()
539        active_policy = policy or self.policy
540        active_budget = budget or ExplorationBudget()
541
542        # 1. Policy Gate
543        if active_policy == ExplorationPolicy.NEVER:
544            return ExplorationRunResult(
545                status="SKIPPED",
546                policy=active_policy.value,
547                objective=objective,
548                error_message="Exploration policy NEVER disallows running HowlDream.",
549                duration_seconds=round(time.time() - t0, 3),
550            )
551        elif active_policy == ExplorationPolicy.SUGGEST:
552            return ExplorationRunResult(
553                status="SKIPPED",
554                policy=active_policy.value,
555                objective=objective,
556                error_message="Exploration policy SUGGEST: exploration suggested but not executed.",
557                duration_seconds=round(time.time() - t0, 3),
558            )
559        elif active_policy == ExplorationPolicy.AUTOMATIC_WITH_BUDGET:
560            if active_budget.max_candidates > 10 or active_budget.max_tokens > 4096:
561                return ExplorationRunResult(
562                    status="SKIPPED",
563                    policy=active_policy.value,
564                    objective=objective,
565                    error_message=(
566                        f"Exploration policy AUTOMATIC_WITH_BUDGET exceeded budget ceiling: "
567                        f"candidates={active_budget.max_candidates} (max 10), "
568                        f"tokens={active_budget.max_tokens} (max 4096)"
569                    ),
570                    duration_seconds=round(time.time() - t0, 3),
571                )
572
573        # 2. Circuit Breaker Gate
574        if current_depth >= self.breakers.max_depth:
575            return ExplorationRunResult(
576                status="SKIPPED",
577                policy=active_policy.value,
578                objective=objective,
579                circuit_breaker_triggered=True,
580                error_message=(
581                    f"Circuit breaker triggered: depth {current_depth} >= "
582                    f"max_depth {self.breakers.max_depth}"
583                ),
584                duration_seconds=round(time.time() - t0, 3),
585            )
586
587        if self._dream_run_count >= self.breakers.max_dream_runs:
588            return ExplorationRunResult(
589                status="SKIPPED",
590                policy=active_policy.value,
591                objective=objective,
592                circuit_breaker_triggered=True,
593                error_message=(
594                    f"Circuit breaker triggered: dream run count {self._dream_run_count} >= "
595                    f"max_dream_runs {self.breakers.max_dream_runs}"
596                ),
597                duration_seconds=round(time.time() - t0, 3),
598            )
599
600        # 3. Availability Check
601        if not self.is_howldream_available():
602            return ExplorationRunResult(
603                status="UNAVAILABLE",
604                policy=active_policy.value,
605                objective=objective,
606                error_message="HowlDream engine is not available or installed.",

def assert_no_execution_authority(payload: Any) -> None: 
 84def assert_no_execution_authority(payload: Any) -> None:
 85    """
 86    Hard negative authority assertion.
 87
 88    Ensures that speculative outputs (from HowlDream, HowlFrame assessment, or
 89    HowlCreate) never claim execution authority, contain self-approvals, or attempt
 90    to authorize actions downstream in HowlPlane executor or HowlChangeOps.
 91    """
 92    if isinstance(payload, dict):
 93        # Check authority block
 94        auth = payload.get("authority")
 95        if isinstance(auth, dict):
 96            if auth.get("executable") is True:
 97                raise AuthorityEscalationError(
 98                    "Exploration output claimed executable=True; rejected"
 99                )
100            if auth.get("type") not in ("ADVISORY", None):
101                raise AuthorityEscalationError(
102                    f"Exploration output claimed authority type {auth.get('type')}; rejected"
103                )
104
105        # Check self-approval
106        if payload.get("approved") is True or payload.get("self_approved") is True:
107            raise AuthorityEscalationError(
108                "Exploration output claimed self-approval; rejected"
109            )
110
111        # Check execution capability in development results
112        if payload.get("execution_capability") is True:
113            raise AuthorityEscalationError(
114                "Development result claimed execution capability; rejected"
115            )
116
117        # Recursively check sub-dictionaries and lists
118        for v in payload.values():
119            assert_no_execution_authority(v)
120    elif isinstance(payload, list):
121        for item in payload:
122            assert_no_execution_authority(item)

593                circuit_breaker_triggered=True,
594                error_message=(
595                    f"Circuit breaker triggered: dream run count {self._dream_run_count} >= "
596                    f"max_dream_runs {self.breakers.max_dream_runs}"
597                ),
598                duration_seconds=round(time.time() - t0, 3),
599            )
600
601        # 3. Availability Check
602        if not self.is_howldream_available():
603            return ExplorationRunResult(
604                status="UNAVAILABLE",
605                policy=active_policy.value,
606                objective=objective,
607                error_message="HowlDream engine is not available or installed.",
608                duration_seconds=round(time.time() - t0, 3),
609            )
610
611        self._dream_run_count += 1
612        work_dir = repo_dir or Path.cwd()
613
614        try:
615            run_dir, raw_envelope = self.provider.explore(
616                objective=objective,
617                budget=active_budget,
618                work_dir=work_dir,
619            )
620            assert_no_execution_authority(raw_envelope)
621            envelope_path = run_dir / "exploration_envelope.json"
622            if not envelope_path.is_file():
623                envelope_path.write_text(json.dumps(raw_envelope, indent=2), encoding="utf-8")
624
625            candidates = raw_envelope.get("candidates", [])
626            assessments: List[Dict[str, Any]] = []
627            development_results: List[Dict[str, Any]] = []
628
629            for cand in candidates:
630                # 4. HowlFrame Invariant Evaluation
631                if run_howlframe:
632                    assess = self.evaluate_candidate_with_howlframe(cand, tmp_dir=work_dir)
633                    assert_no_execution_authority(assess)
634                    assessments.append(assess)
635
636                    # 5. HowlCreate Sandbox Development if accepted
637                    if (
638                        run_howlcreate
639                        and assess.get("disposition") == "ACCEPT_FOR_DEVELOPMENT"
640                    ):
641                        dev_res = self.promote_candidate_to_howlcreate(cand, assess)
642                        assert_no_execution_authority(dev_res)
643                        development_results.append(dev_res)
644
645            dag_dict = raw_envelope.get("descent_dag") or raw_envelope.get("lineage_dag")
646
647            # Record to EvidenceLedger if configured
648            if self.ledger:
649                entry = EvidenceEntry(
650                    source="howldream",
651                    claim_type="exploration",
652                    content={
653                        "objective": objective,
654                        "candidates_count": len(candidates),
655                        "accepted_count": len(development_results),
656                        "envelope_path": str(envelope_path),
657                        "authority": "ADVISORY_ONLY",
658                    },
659                    status="SUCCESS",
660                )
661                self.ledger.record(entry)
662
663            status = "SUCCESS" if len(candidates) > 0 else "NO_CANDIDATES"
664            return ExplorationRunResult(
665                status=status,
666                policy=active_policy.value,
667                objective=objective,
668                envelope_path=str(envelope_path),
669                run_dir=str(envelope_path.parent),
670                candidates=candidates,
671                assessments=assessments,
672                development_results=development_results,
673                lineage_trace=dag_dict,
674                duration_seconds=round(time.time() - t0, 3),
675            )
676
677        except AuthorityEscalationError as exc:
678            return ExplorationRunResult(
679                status="REJECTED",
680                policy=active_policy.value,
681                objective=objective,
682                error_message=f"Authority violation: {exc}",
683                duration_seconds=round(time.time() - t0, 3),
684            )
685        except Exception as exc:
686            return ExplorationRunResult(
687                status="FAILED",
688                policy=active_policy.value,
689                objective=objective,
690                error_message=str(exc),
691                duration_seconds=round(time.time() - t0, 3),
692            )
