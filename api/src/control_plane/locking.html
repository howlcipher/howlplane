locking.py

Lightweight filesystem repository and task lifecycle locks for HowlPlane. Prevents concurrent mutation workflows from destroying git attribution or racing reviews, and protects task lifecycle operations against concurrent invocation. Provides safe stale lock detection without requiring Redis, daemons, or databases.

 63def get_process_create_time(pid: int) -> float:
 64    """Extracts process creation timestamp on Linux /proc or falls back to current time."""
 65    try:
 66        stat_path = Path(f"/proc/{pid}/stat")
 67        if stat_path.is_file():
 68            return stat_path.stat().st_mtime
 69    except Exception:
 70        pass
 71    return time.time()

 74class LockOwnerState(str, Enum):
 75    """What can actually be established about a lock owner from this machine.
 76
 77    The middle state is the one that matters. Previously anything not provably
 78    dead was reported as alive, which let a lock written on another host block
 79    the documented recovery path forever while claiming to be "held by active
 80    process" -- a statement about a PID whose liveness had never been checked
 81    (HOWLFRAM-SLOPFIX-05).
 82    """
 83
 84    ACTIVE = "ACTIVE"        # owner is running: fail closed, never steal
 85    STALE = "STALE"          # owner is provably gone: safe to reclaim
 86    AMBIGUOUS = "AMBIGUOUS"  # cannot be established here: requires a human
 87
 88
 89def classify_lock_owner(
 90    pid: int,
 91    hostname: str,
 92    expected_create_time: Optional[float] = None,
 93) -> Tuple[LockOwnerState, str]:
 94    """Classifies a lock owner as ACTIVE, STALE, or AMBIGUOUS, with a reason."""
 95    current_host = socket.gethostname()
 96    if hostname != current_host:
 97        return (
 98            LockOwnerState.AMBIGUOUS,
 99            f"Lock was written on host '{hostname}' (this host is "
100            f"'{current_host}'); liveness of PID {pid} cannot be established "
101            f"from here",
102        )
103
104    if pid <= 0:
105        return LockOwnerState.STALE, "Invalid PID <= 0"
106
107    try:
108        os.kill(pid, 0)
109    except OSError as err:
110        if err.errno == errno.ESRCH:
111            return (
112                LockOwnerState.STALE,
113                f"Process PID {pid} is no longer running (ESRCH)",
114            )
115        if err.errno == errno.EPERM:
116            return (
117                LockOwnerState.AMBIGUOUS,
118                f"PID {pid} exists but runs under another user account, so it "
119                f"cannot be confirmed as this task's owner",
120            )
121        return (
122            LockOwnerState.AMBIGUOUS,
123            f"Process check returned unexpected OSError {err}",
124        )
125
126    if expected_create_time and expected_create_time > 0:
127        actual_create_time = get_process_create_time(pid)
128        if abs(actual_create_time - expected_create_time) > 10.0:
129            return (
130                LockOwnerState.STALE,
131                f"PID {pid} was recycled by operating system (mismatched start time)",
132            )
133
134    return LockOwnerState.ACTIVE, f"Process PID {pid} is actively running"

137def get_systemd_main_pid(unit_name: str) -> int:
138    """Return the MainPID for a systemd user unit, or 0 if unavailable."""
139    try:
140        shown = subprocess.run(
141            ["systemctl", "--user", "show", unit_name, "--property=MainPID", "--value"],
142            text=True,
143            stdout=subprocess.PIPE,
144            stderr=subprocess.DEVNULL,
145            timeout=3,
146            check=False,
147        )
148        return int(shown.stdout.strip() or "0")
149    except (OSError, ValueError, subprocess.TimeoutExpired):
150        return 0

153def is_process_record_active(record: Dict[str, Any]) -> bool:
154    """Check whether a process record corresponds to a live process.
155
156    Used by both the campaign discovery path and the service status path so the
157    liveness predicate is not duplicated between convenience commands and the
158    long-running process supervisor.
159    """
160    if record.get("status") == "stopped":
161        return False
162    backend = record.get("backend")
163    unit_name = record.get("unit_name")
164    hostname = record.get("hostname", "")
165    if hostname and hostname != socket.gethostname():
166        return False
167    if backend == "systemd" and unit_name:
168        try:
169            result = subprocess.run(
170                ["systemctl", "--user", "is-active", "--quiet", unit_name],
171                timeout=3,
172                check=False,
173            )
174            if result.returncode != 0:
175                return False
176            main_pid = get_systemd_main_pid(unit_name)
177            if main_pid <= 0:
178                return False
179            rec_pid = record.get("pid", 0)
180            create_time = record.get("process_create_time", 0.0) if rec_pid == main_pid else None
181            alive, _ = is_process_alive(main_pid, hostname, create_time)
182            return alive
183        except (OSError, ValueError, subprocess.TimeoutExpired):
184            return False
185    pid = record.get("pid", 0)
186    create_time = record.get("process_create_time", 0.0)
187    alive, _ = is_process_alive(pid, hostname, create_time)
188    return alive

191def is_process_alive(
192    pid: int,
193    hostname: str,
194    expected_create_time: Optional[float] = None,
195) -> Tuple[bool, str]:
196    """Reports whether a lock owner must be treated as running.
197
198    Kept as the single liveness predicate for callers that only need a boolean,
199    and built on classify_lock_owner so there is one authority. AMBIGUOUS counts
200    as alive here: automatic paths must never assume an unverifiable owner is
201    gone. Reclaiming one is a deliberate human act -- see reclaim_lock.
202    """
203    state, reason = classify_lock_owner(pid, hostname, expected_create_time)
204    return state is not LockOwnerState.STALE, reason

226
227def reclaim_lock(lock_path: Union[str, Path]) -> LockReclamation:
228    """Reclaims a lock whose owner is gone or unverifiable, for a human caller.
229
230    Fails closed on an ACTIVE owner: a running process is never displaced, no
231    matter who asks. STALE and AMBIGUOUS owners can be reclaimed, because the
232    first is provably gone and the second is precisely the case that had no
233    recovery path at all -- `ai resume` recommended resuming a run whose lock it
234    could never take (HOWLFRAM-SLOPFIX-05). Reclaiming an AMBIGUOUS lock stays a
235    deliberate act by a person, never something an automatic path does.
236    """
237    path = Path(lock_path)
238    if not path.exists():
239        raise LockError(f"No lock file at '{path}'.")
240    try:
241        existing = LockMetadata.from_dict(safe_load_json(path))
242    except Exception as err:
243        raise LockError(f"Lock file at '{path}' is unreadable: {err}") from err
244
245    state, reason = classify_lock_owner(
246        existing.pid, existing.hostname, existing.process_create_time
247    )
248    if state is LockOwnerState.ACTIVE:
249        raise LockError(
250            f"Refusing to reclaim: lock for task '{existing.task_id}' is held by "
251            f"active process PID {existing.pid} ({existing.command}). {reason}. "
252            f"Stop that process first, or use `howlplane cancel {existing.task_id}`."
253        )
254
255    record = LockReclamation(
256        task_id=existing.task_id,
257        lock_path=str(path),
258        owner_state=state.value,
259        reason=reason,
260        owner_pid=existing.pid,
261        owner_hostname=existing.hostname,
262        owner_command=existing.command,
263        owner_started_at=existing.started_at,
264        reclaimed_by_pid=os.getpid(),
265        reclaimed_by_hostname=socket.gethostname(),
266    )
267    path.unlink()
268    return record

290@dataclass
291class LockOwnership:
292    """One lock-holding lifecycle's claim on a canonical lock path.
293
294    A governed run reaches the same lock through more than one component:
295    `HumanLifecycleManager.resume` takes the task lock, then hands control to
296    the orchestrator, which needs that same lock to do the work. Those are two
297    `TaskLock` objects over one lock file in one process, and tracking "do I
298    hold this?" on the instance made the second one fail -- the documented
299    recovery path could never recover anything (HOWLFRAM-SLOPFIX-06).
300
301    Ownership is therefore a token, not a process identity. A component that was
302    handed this token belongs to the lifecycle that already owns the lock and
303    may re-enter it; a component that was not gets the same refusal as any other
304    caller. `os.getpid()` is deliberately *not* the test: two unrelated
305    operations sharing a process must not share authority.
306    """
307
308    lineage_id: str
309    lock_path: str
310    task_id: str
311    operation: str
312    depth: int = 1
313
314
315# Canonical lock path -> the lifecycle currently holding it in this process.
316# Guarded because a lock may be taken from a worker thread.
317_OWNERSHIP_REGISTRY: Dict[Path, LockOwnership] = {}
318_OWNERSHIP_GUARD = threading.RLock()
319
320
321def _registry_key(lock_path: Union[str, Path]) -> Path:
322    """Canonical identity of a lock file, independent of how it was spelled."""
323    return Path(lock_path).resolve()
324
325
326def reset_lock_ownership_registry() -> None:
327    """Drops all in-process ownership records. For test isolation only."""
328    with _OWNERSHIP_GUARD:
329        _OWNERSHIP_REGISTRY.clear()

332class _BaseFileLock:
333    """Base mutual-exclusion file lock with stale process reclamation."""
334
335    lock_type = "generic_lock"
336    error_cls: Type[LockError] = LockError
337
338    def __init__(
339        self,
340        repo_root: Union[str, Path],
341        task_id: str,
342        lock_path: Path,
343        command: str,
344        operation: str,
345    ):
346        self.repo_root = Path(repo_root).resolve()
347        self.task_id = task_id
348        self.lock_path = lock_path
349        self.command = command
350        self.operation = operation
351        self._acquired = False
352        self._ownership: Optional[LockOwnership] = None
353        self._pending_ownership: Optional[LockOwnership] = None
354
355    def join(self, ownership: Optional[LockOwnership]) -> "_BaseFileLock":
356        """Declares which lifecycle this lock belongs to, for `with` use."""
357        self._pending_ownership = ownership
358        return self
359
360    @property
361    def ownership(self) -> Optional[LockOwnership]:
362        """The lifecycle token to hand to components that must re-enter this lock."""
363        return self._ownership
364
365    def acquire(self, ownership: Optional[LockOwnership] = None) -> bool:
366        """Takes the lock, or joins the lifecycle that already holds it.
367
368        Passing the `ownership` token returned by an outer acquisition of the
369        same lock re-enters it and increments its depth; the lock is released
370        for real only when every holder in the lineage has released. Passing
371        nothing, or a token from a different lineage, fails closed exactly as a
372        foreign caller would.
373        """
374        if self._acquired:
375            return True
376
377        key = _registry_key(self.lock_path)
378        label = self.lock_type.replace("_", " ").title()
379
380        with _OWNERSHIP_GUARD:
381            held = _OWNERSHIP_REGISTRY.get(key)
382            if held is not None:
383                if ownership is not None and ownership.lineage_id == held.lineage_id:
384                    held.depth += 1
385                    self._ownership = held
386                    self._acquired = True
387                    return True
388                raise self.error_cls(
389                    f"{label} lock already held on task '{held.task_id}' by this "
390                    f"process for operation '{held.operation}'. A component of "
391                    f"that operation may re-enter it by passing its ownership "
392                    f"token; an unrelated operation may not."
393                )
394
395        my_pid = os.getpid()
396        my_host = socket.gethostname()
397        my_ctime = get_process_create_time(my_pid)
398
399        if self.lock_path.exists():
400            try:
401                data = safe_load_json(self.lock_path)
402                existing = LockMetadata.from_dict(data)
403            except Exception:
404                existing = None
405
406            if existing:
407                if existing.pid == my_pid and existing.hostname == my_host:
408                    # Our PID, but no lifecycle in this process claims it: the
409                    # file outlived the run that wrote it and the OS handed the
410                    # PID back to us. Never silently adopt it.
411                    raise self.error_cls(
412                        f"{label} lock already held on task '{existing.task_id}'."
413                    )
414
415                state, reason = classify_lock_owner(
416                    existing.pid, existing.hostname, existing.process_create_time
417                )
418                if state is LockOwnerState.ACTIVE:
419                    raise self.error_cls(
420                        f"{label} lock held by active process "
421                        f"PID {existing.pid} ({existing.command}) for task '{existing.task_id}' "
422                        f"started at {existing.started_at}. Concurrent operation blocked."
423                    )
424                if state is LockOwnerState.AMBIGUOUS:
425                    # Never silently steal a lock we cannot prove is dead, and
426                    # never claim its owner is active when we never checked.
427                    raise self.error_cls(
428                        f"{label} lock ownership is INDETERMINATE for task "
429                        f"'{existing.task_id}': PID {existing.pid} ({existing.command}) "
430                        f"on host '{existing.hostname}', started at {existing.started_at}. "
431                        f"{reason}. If that run is definitely gone, reclaim the lock "
432                        f"explicitly with `howlplane unlock {existing.task_id}`."
433                    )
434                # Provably gone: reclaim automatically, as before.
435                try:
436                    self.lock_path.unlink()
437                except Exception:
438                    pass
439            else:
440                try:
441                    self.lock_path.unlink()
442                except Exception:
443                    pass
444
445        meta = LockMetadata(
446            task_id=self.task_id,
447            pid=my_pid,
448            hostname=my_host,
449            lock_type=self.lock_type,
450            operation=self.operation,
451            command=self.command,
452            process_create_time=my_ctime,
453        )
454
455        content = meta.to_json()
456        try:
457            self.lock_path.parent.mkdir(parents=True, exist_ok=True)
458            fd = os.open(
459                str(self.lock_path),
460                os.O_CREAT | os.O_EXCL | os.O_WRONLY,
461                0o644,
462            )
463            with os.fdopen(fd, "w", encoding="utf-8") as f:
464                f.write(content)
465            own = LockOwnership(
466                lineage_id=uuid.uuid4().hex,
467                lock_path=str(self.lock_path),
468                task_id=self.task_id,
469                operation=self.operation,
470            )
471            with _OWNERSHIP_GUARD:
472                _OWNERSHIP_REGISTRY[_registry_key(self.lock_path)] = own
473            self._ownership = own
474            self._acquired = True
475            return True
476        except FileExistsError:
477            raise self.error_cls(
478                f"Contention on '{self.lock_path}'. Another operation acquired the lock concurrently."
479            )
480
481    def release(self) -> None:
482        """Releases this holder's claim, unlinking only when the last one goes.
483
484        Every acquisition has a matching release, including a re-entrant one:
485        an inner holder decrements the lineage depth and leaves the file in
486        place for the outer holder that is still working.
487        """
488        if not self._acquired:
489            return
490
491        own = self._ownership
492        self._ownership = None
493        if own is not None:
494            with _OWNERSHIP_GUARD:
495                own.depth -= 1
496                if own.depth > 0:
497                    self._acquired = False
498                    return
499                _OWNERSHIP_REGISTRY.pop(_registry_key(self.lock_path), None)
500
501        my_pid = os.getpid()
502        my_host = socket.gethostname()
503        if self.lock_path.exists():
504            try:
505                data = safe_load_json(self.lock_path)
506                existing = LockMetadata.from_dict(data)
507                if existing.pid == my_pid and existing.hostname == my_host:
508                    self.lock_path.unlink()
509            except Exception:
510                try:
511                    self.lock_path.unlink()
512                except Exception:
513                    pass
514        self._acquired = False
515
516    def __enter__(self):
517        self.acquire(self._pending_ownership)
518        return self
519
520    def __exit__(self, exc_type, exc_val, exc_tb):
521        self.release()

def classify_lock_owner( pid: int, hostname: str, expected_create_time: Optional[float] = None) -> Tuple[LockOwnerState, str]: 
 90def classify_lock_owner(
 91    pid: int,
 92    hostname: str,
 93    expected_create_time: Optional[float] = None,
 94) -> Tuple[LockOwnerState, str]:
 95    """Classifies a lock owner as ACTIVE, STALE, or AMBIGUOUS, with a reason."""
 96    current_host = socket.gethostname()
 97    if hostname != current_host:
 98        return (
 99            LockOwnerState.AMBIGUOUS,
100            f"Lock was written on host '{hostname}' (this host is "
101            f"'{current_host}'); liveness of PID {pid} cannot be established "
102            f"from here",
103        )
104
105    if pid <= 0:
106        return LockOwnerState.STALE, "Invalid PID <= 0"
107
108    try:
109        os.kill(pid, 0)
110    except OSError as err:
111        if err.errno == errno.ESRCH:
112            return (
113                LockOwnerState.STALE,
114                f"Process PID {pid} is no longer running (ESRCH)",
115            )
116        if err.errno == errno.EPERM:
117            return (
118                LockOwnerState.AMBIGUOUS,
119                f"PID {pid} exists but runs under another user account, so it "
120                f"cannot be confirmed as this task's owner",
121            )
122        return (
123            LockOwnerState.AMBIGUOUS,
124            f"Process check returned unexpected OSError {err}",
125        )
126
127    if expected_create_time and expected_create_time > 0:
128        actual_create_time = get_process_create_time(pid)
129        if abs(actual_create_time - expected_create_time) > 10.0:
130            return (
131                LockOwnerState.STALE,
132                f"PID {pid} was recycled by operating system (mismatched start time)",
133            )
134
135    return LockOwnerState.ACTIVE, f"Process PID {pid} is actively running"

def is_process_record_active(record: Dict[str, Any]) -> bool: 
154def is_process_record_active(record: Dict[str, Any]) -> bool:
155    """Check whether a process record corresponds to a live process.
156
157    Used by both the campaign discovery path and the service status path so the
158    liveness predicate is not duplicated between convenience commands and the
159    long-running process supervisor.
160    """
161    if record.get("status") == "stopped":
162        return False
163    backend = record.get("backend")
164    unit_name = record.get("unit_name")
165    hostname = record.get("hostname", "")
166    if hostname and hostname != socket.gethostname():
167        return False
168    if backend == "systemd" and unit_name:
169        try:
170            result = subprocess.run(
171                ["systemctl", "--user", "is-active", "--quiet", unit_name],
172                timeout=3,
173                check=False,
174            )
175            if result.returncode != 0:
176                return False
177            main_pid = get_systemd_main_pid(unit_name)
178            if main_pid <= 0:
179                return False
180            rec_pid = record.get("pid", 0)
181            create_time = record.get("process_create_time", 0.0) if rec_pid == main_pid else None
182            alive, _ = is_process_alive(main_pid, hostname, create_time)
183            return alive
184        except (OSError, ValueError, subprocess.TimeoutExpired):
185            return False
186    pid = record.get("pid", 0)
187    create_time = record.get("process_create_time", 0.0)
188    alive, _ = is_process_alive(pid, hostname, create_time)
189    return alive

def reclaim_lock( lock_path: Union[str, pathlib.Path]) -> LockReclamation: 
228def reclaim_lock(lock_path: Union[str, Path]) -> LockReclamation:
229    """Reclaims a lock whose owner is gone or unverifiable, for a human caller.
230
231    Fails closed on an ACTIVE owner: a running process is never displaced, no
232    matter who asks. STALE and AMBIGUOUS owners can be reclaimed, because the
233    first is provably gone and the second is precisely the case that had no
234    recovery path at all -- `ai resume` recommended resuming a run whose lock it
235    could never take (HOWLFRAM-SLOPFIX-05). Reclaiming an AMBIGUOUS lock stays a
236    deliberate act by a person, never something an automatic path does.
237    """
238    path = Path(lock_path)
239    if not path.exists():
240        raise LockError(f"No lock file at '{path}'.")
241    try:
242        existing = LockMetadata.from_dict(safe_load_json(path))
243    except Exception as err:
244        raise LockError(f"Lock file at '{path}' is unreadable: {err}") from err
245
246    state, reason = classify_lock_owner(
247        existing.pid, existing.hostname, existing.process_create_time
248    )
249    if state is LockOwnerState.ACTIVE:
250        raise LockError(
251            f"Refusing to reclaim: lock for task '{existing.task_id}' is held by "
252            f"active process PID {existing.pid} ({existing.command}). {reason}. "
253            f"Stop that process first, or use `howlplane cancel {existing.task_id}`."
254        )
255
256    record = LockReclamation(
257        task_id=existing.task_id,
258        lock_path=str(path),
259        owner_state=state.value,
260        reason=reason,
261        owner_pid=existing.pid,
262        owner_hostname=existing.hostname,
263        owner_command=existing.command,
264        owner_started_at=existing.started_at,
265        reclaimed_by_pid=os.getpid(),
266        reclaimed_by_hostname=socket.gethostname(),
267    )
268    path.unlink()
269    return record
